    15→/**
    16→ * Pico call engine — peer-to-peer WebRTC mesh.
    17→ *
    18→ * Design goals:
    19→ *  - Media never touches a server: every audio/video packet travels directly
    20→ *    between participants (mesh topology). The signaling socket only carries
    21→ *    SDP/ICE/chat text.
    22→ *  - Renegotiation-free quality control: resolution/fps changes ride on
    23→ *    applyConstraints(), bitrate changes on setParameters(), screen share on
    24→ *    replaceTrack(). No expensive offer/answer churn.
    25→ *  - Opus DTX + 24 kbps cap: silence costs (almost) nothing on the wire.
    26→ *  - Per-link adaptive bitrate: getStats() every 3s classifies each
    27→ *    connection (RTT + packet loss) and caps outgoing video per link, so one
    28→ *    weak participant never drags the whole call down.
    29→ *  - Perfect-negotiation pattern (polite/impolite) for glare-free offers.
    30→ */
    31→
    32→const ICE_SERVERS: RTCIceServer[] = [
    33→  { urls: ['stun:stun.l.google.com:19302', 'stun:stun1.l.google.com:19302'] },
    34→  { urls: ['stun:stun.cloudflare.com:3478'] },
    35→]
    36→
    37→const AUDIO_SENDER_BITRATE = 24_000

    59→/** Add Opus DTX + low bitrate cap to SDP — near-zero cost for silence. */
    60→function mungeOpus(sdp: string): string {
    61→  const lines = sdp.split('\r\n')
    62→  const opusPTs = new Set<string>()
    63→  for (const line of lines) {
    64→    const m = line.match(/^a=rtpmap:(\d+) opus\/48000/i)
    65→    if (m) opusPTs.add(m[1])
    66→  }
    67→  const out = lines.map((line) => {
    68→    const m = line.match(/^a=fmtp:(\d+)(.*)$/)
    69→    if (m && opusPTs.has(m[1])) {
    70→      let params = m[2] ?? ''
    71→      if (!/usedtx/i.test(params)) params += ';usedtx=1'
    72→      if (!/maxaveragebitrate/i.test(params)) params += ';maxaveragebitrate=24000'
    73→      return `a=fmtp:${m[1]}${params}`
    74→    }
    75→    return line
    76→  })
    77→  return out.join('\r\n')
    78→}
    79→
    80→function classifyLink(rttSec: number | null, lossFrac: number | null): NetworkLevel {
    81→  const rtt = rttSec ?? 0.4
    82→  const loss = lossFrac ?? 0.05
    83→  if (rtt > 0.5 || loss > 0.08) return 0
    84→  if (rtt > 0.3 || loss > 0.05) return 1
    85→  if (rtt > 0.15 || loss > 0.02) return 2
    86→  return 3
    87→}

    93→export class CallEngine {
    94→  private socket: Socket | null = null
    95→  private roomId = ''
    96→  private myId = ''
    97→  private localStream: MediaStream | null = null
    98→  private cameraTrack: MediaStreamTrack | null = null
    99→  private micTrack: MediaStreamTrack | null = null
   100→  private screenTrack: MediaStreamTrack | null = null
   101→  private pcs = new Map<string, RTCPeerConnection>()
   102→  private sendersMap = new Map<string, { audio?: RTCRtpSender; video?: RTCRtpSender }>()
   103→  private pendingIce = new Map<string, RTCIceCandidateInit[]>()
   104→  private links = new Map<string, LinkInfo>()
   105→  private statsTimer: number | null = null
   106→  private makingOffer = new Map<string, boolean>()
   107→  private ignoreOffer = new Map<string, boolean>()
   108→  private preset: QualityPreset = 'balanced'
   109→  private adaptiveProfile: Exclude<QualityPreset, 'auto'> = 'balanced'
   110→  private badCount = 0
   111→  private goodCount = 0
   112→  private started = false
   113→  private testMode = false
   114→  private mediaMode: MediaMode = 'full'
   115→  private prevBytes = new Map<string, number>()
   116→
   117→  async start(opts: StartOptions): Promise<{ mode: MediaMode }> {
   118→    if (this.started) await this.leave(true)
   119→    this.started = true
   120→    this.roomId = opts.roomId
   121→    this.preset = opts.preset
   122→    this.adaptiveProfile = opts.preset === 'auto' ? 'balanced' : opts.preset
   123→    this.testMode = !!opts.testMode
   124→
   125→    const store = useCallStore.getState()
   126→    store.setSelf({ name: opts.name })
   127→    store.setPreset(opts.preset)
   128→    store.setEffectiveLabel(this.effectiveLabel())
   129→
   130→    // ---- Signaling ----
   131→    const socket = io('/?XTransformPort=3003', {
   132→      path: '/',
   133→      transports: ['websocket', 'polling'],
   134→      reconnection: true,
   135→      reconnectionAttempts: Infinity,
   136→      reconnectionDelay: 1000,
   137→      reconnectionDelayMax: 4000,
   138→      timeout: 8000,
   139→    })
   140→    this.socket = socket
   141→    // Dev/debug handle (harmless in prod, invaluable for live diagnostics)
   142→    if (typeof window !== 'undefined') {
   143→      ;(window as unknown as { __pico?: unknown }).__pico = { engine: this, socket }
   144→    }
   145→    socket.on('connect', () => {
   146→      useCallStore.getState().setSignalingConnected(true)
   147→      if (this.myId) this.rejoin()
   148→    })
   149→    socket.on('disconnect', () => useCallStore.getState().setSignalingConnected(false))
   150→    socket.on('connect_error', () => useCallStore.getState().setSignalingConnected(false))
   151→
   152→    // ---- Local media ----
   153→    this.localStream = opts.initialStream ?? null
   154→    if (this.localStream) {
   155→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   156→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   157→      this.mediaMode = this.cameraTrack ? 'full' : this.micTrack ? 'audio' : 'none'
   158→      // Register the handed-over preview stream so tiles can attach to it
   159→      putStream('self', this.localStream)
   160→    }
   161→    if (!this.localStream || (!this.cameraTrack && !this.micTrack)) {
   162→      this.mediaMode = await this.acquireMedia()
   163→    }
   164→    this.applyInitialTrackState()
   165→    trackAudio('self', this.localStream)
   166→
   167→    // ---- Room ----
   168→    // The ack races a timeout so a dead signaling server can never hang the UI.
   169→    await new Promise<void>((resolve) => {
   170→      let settled = false
   171→      const finish = () => {
   172→        if (!settled) {
   173→          settled = true
   174→          resolve()
   175→        }
   176→      }
   177→      const guard = window.setTimeout(finish, 12000)
   178→      socket.emit(
   179→        'room:join',
   180→        { roomId: this.roomId, name: opts.name },
   181→        (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; reason?: string }) => {
   182→          window.clearTimeout(guard)
   183→          if (!res?.ok) {
   184→            useCallStore.getState().addChat({
   185→              id: 'sys-' + Date.now(),
   186→              from: 'system',
   187→              name: 'System',
   188→              text: res?.reason ?? 'Could not join the room',
   189→              ts: Date.now(),
   190→              system: true,
   191→            })
   192→          }
   193→          this.myId = res?.self?.id ?? socket.id!
   194→          useCallStore.getState().setSelf({ id: this.myId })
   195→          if (res?.peers) this.reconcilePeers(res.peers)
   196→          finish()
   197→        },
   198→      )
   199→    })
   200→
   201→    // ---- Events ----
   202→    socket.on('peer:joined', ({ peer }: { peer: { id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean } }) => {
   203→      this.ensurePeer(peer.id, peer.name, peer.micOn, peer.camOn, peer.sharing)
   204→      useCallStore.getState().addChat({
   205→        id: 'sys-' + Math.random(),
   206→        from: 'system',
   207→        name: 'System',
   208→        text: `${peer.name} joined`,
   209→        ts: Date.now(),
   210→        system: true,
   211→      })
   212→    })
   213→    socket.on('peer:left', ({ peerId }: { peerId: string }) => this.dropPeer(peerId))
   214→    socket.on('room:peers', ({ peers }: { peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   215→      // Only meaningful after a reconnect — reconcile full presence
   216→      if (this.myId) this.reconcilePeers(peers)
   217→    })
   218→    socket.on('peer:state', ({ peerId, micOn, camOn, sharing }: { peerId: string; micOn: boolean; camOn: boolean; sharing: boolean }) => {
   219→      const store = useCallStore.getState()
   220→      store.patchPeer(peerId, { micOn, camOn, sharing })
   221→      // Follow whoever is presenting, so a remote screen never hides in a small tile
   222→      if (sharing && store.spotlight !== peerId) store.setSpotlight(peerId)
   223→      else if (!sharing && store.spotlight === peerId) store.setSpotlight(null)
   224→    })
   225→    socket.on('chat:message', (msg: { id: string; from: string; name: string; text: string; ts: number }) => {
   226→      useCallStore.getState().addChat(msg)
   227→    })
   228→
   229→    socket.on('rtc:offer', async ({ from, sdp }: { from: string; sdp: string }) => {
   230→      try {
   231→        const pc = this.pcs.get(from)
   232→        if (!pc) return
   233→        const polite = this.myId < from
   234→        const offerCollision = this.makingOffer.get(from) || pc.signalingState !== 'stable'
   235→        this.ignoreOffer.set(from, !polite && offerCollision)
   236→        if (this.ignoreOffer.get(from)) return
   237→        await pc.setRemoteDescription({ type: 'offer', sdp: mungeOpus(sdp) })
   238→        await this.flushIce(from)
   239→        const answer = await pc.createAnswer()
   240→        answer.sdp = mungeOpus(answer.sdp ?? '')
   241→        await pc.setLocalDescription(answer)
   242→        this.socket?.emit('rtc:answer', { to: from, sdp: pc.localDescription!.sdp })
   243→      } catch {
   244→        /* negotiation raced — next round will fix it */
   245→      }
   246→    })
   247→
   248→    socket.on('rtc:answer', async ({ from, sdp }: { from: string; sdp: string }) => {
   249→      try {
   250→        const pc = this.pcs.get(from)
   251→        if (!pc || pc.signalingState !== 'have-local-offer') return
   252→        await pc.setRemoteDescription({ type: 'answer', sdp: mungeOpus(sdp) })
   253→        await this.flushIce(from)
   254→      } catch {
   255→        /* ignore */
   256→      }
   257→    })
   258→
   259→    socket.on('rtc:ice', ({ from, candidate }: { from: string; candidate: RTCIceCandidateInit }) => {
   260→      const pc = this.pcs.get(from)
   261→      if (!pc) return
   262→      if (pc.remoteDescription) {
   263→        pc.addIceCandidate(candidate).catch(() => {
   264→          if (!this.ignoreOffer.get(from)) {
   265→            this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   266→          }
   267→        })
   268→      } else {
   269→        this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   270→      }
   271→    })
   272→
   273→    // ---- Stats loop ----
   274→    this.statsTimer = window.setInterval(() => void this.pollStats(), STATS_INTERVAL_MS)
   275→
   276→    return { mode: this.mediaMode }
   277→  }
   278→
   279→  /* ---------------- media ---------------- */
   280→
   281→  private async acquireMedia(): Promise<MediaMode> {
   282→    if (this.testMode) {
   283→      this.localStream = await createTestStream()
   284→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   285→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   286→      putStream('self', this.localStream)
   287→      return 'full'
   288→    }
   289→    const audio: MediaTrackConstraints = {
   290→      echoCancellation: true,
   291→      noiseSuppression: true,
   292→      autoGainControl: true,
   293→      channelCount: 1,
   294→    }
   295→    const video = this.videoConstraints(this.resolveProfile())
   296→    // Full quality first
   297→    try {
   298→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio, video })
   299→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   300→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   301→      this.attachLocalStream()
   302→      return 'full'
   303→    } catch {
   304→      /* fall through */
   305→    }
   306→    // Audio only
   307→    try {
   308→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio })
   309→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   310→      this.cameraTrack = null
   311→      this.attachLocalStream()
   312→      return 'audio'
   313→    } catch {
   314→      /* fall through */
   315→    }
   316→    // Listener mode
   317→    this.localStream = null
   318→    return 'none'
   319→  }
   320→
   321→  private videoConstraints(profile: QualityProfile): MediaTrackConstraints {
   322→    return {
   323→      width: { ideal: profile.width },
   324→      height: { ideal: profile.height },
   325→      frameRate: { ideal: profile.frameRate, max: profile.frameRate },
   326→      facingMode: 'user',
   327→    }
   328→  }
   329→
   330→  private attachLocalStream() {
   331→    if (!this.localStream) return
   332→    putStream('self', this.localStream)
   333→  }

   335→  private applyInitialTrackState() {
   336→    const s = useCallStore.getState().self
   337→    if (this.cameraTrack) this.cameraTrack.enabled = s.camOn
   338→    if (this.micTrack) this.micTrack.enabled = s.micOn
   339→    if (!s.camOn) void this.setVideoSendersActive(false)
   340→  }

   342→  /* ---------------- peers ---------------- */
   343→
   344→  private ensurePeer(
   345→    peerId: string,
   346→    name: string,
   347→    micOn = true,
   348→    camOn = true,
   349→    sharing = false,
   350→  ) {
   351→    const store = useCallStore.getState()
   352→    if (!store.peers[peerId]) {
   353→      store.upsertPeer({ id: peerId, name, micOn, camOn, sharing, level: 3, hasStream: false, streamVersion: 0 })
   354→    }
   355→    if (this.pcs.has(peerId)) return
   356→
   357→    const pc = new RTCPeerConnection({
   358→      iceServers: ICE_SERVERS,
   359→      iceCandidatePoolSize: 4,
   360→      bundlePolicy: 'max-bundle',
   361→      rtcpMuxPolicy: 'require',
   362→    })
   363→    this.pcs.set(peerId, pc)
   364→    this.links.set(peerId, { lastCap: null })
   365→    this.makingOffer.set(peerId, false)
   366→    this.ignoreOffer.set(peerId, false)
   367→
   368→    // Local media into this connection
   369→    const senders: { audio?: RTCRtpSender; video?: RTCRtpSender } = {}
   370→    if (this.localStream) {
   371→      for (const track of this.localStream.getTracks()) {
   372→        try {
   373→          const sender = pc.addTrack(track, this.localStream)
   374→          if (track.kind === 'audio') senders.audio = sender
   375→          else senders.video = sender
   376→        } catch {
   377→          /* ignore */
   378→        }
   379→      }
   380→    } else {
   381→      const at = pc.addTransceiver('audio', { direction: 'recvonly' })
   382→      const vt = pc.addTransceiver('video', { direction: 'recvonly' })
   383→      senders.audio = at.sender
   384→      senders.video = vt.sender
   385→    }
   386→    this.sendersMap.set(peerId, senders)
   387→
   388→    pc.onicecandidate = (e) => {
   389→      if (e.candidate) this.socket?.emit('rtc:ice', { to: peerId, candidate: e.candidate.toJSON() })
   390→    }
   391→
   392→    pc.ontrack = (e) => {
   393→      const stream = e.streams[0]
   394→      if (!stream) return
   395→      putStream(peerId, stream)
   396→      trackAudio(peerId, stream)
   397→      useCallStore.getState().bumpPeerStream(peerId, true)
   398→    }
   399→
   400→    pc.onnegotiationneeded = async () => {
   401→      try {
   402→        this.makingOffer.set(peerId, true)
   403→        const offer = await pc.createOffer()
   404→        await pc.setLocalDescription({ type: 'offer', sdp: mungeOpus(offer.sdp ?? '') })
   405→        const desc = pc.localDescription
   406→        if (desc) this.socket?.emit('rtc:offer', { to: peerId, sdp: desc.sdp })
   407→      } catch {
   408→        /* ignore */
   409→      } finally {
   410→        this.makingOffer.set(peerId, false)
   411→      }
   412→    }
   413→
   414→    pc.onconnectionstatechange = () => {
   415→      if (pc.connectionState === 'failed') {
   416→        try {
   417→          pc.restartIce()
   418→        } catch {
   419→          /* ignore */
   420→        }
   421→      }
   422→    }
   423→
   424→    // Prime encoding params on the fresh senders
   425→    this.capAudioSender(pc)
   426→    const videoSender = senders.video
   427→    if (videoSender) {
   428→      try {
   429→        const params = videoSender.getParameters()
   430→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   431→        params.encodings[0].maxBitrate = this.resolveProfile().maxBitrate
   432→        void videoSender.setParameters(params).catch(() => undefined)
   433→      } catch {
   434→        /* ignore */
   435→      }
   436→    }
   437→  }
   438→
   439→  private dropPeer(peerId: string) {
   440→    const pc = this.pcs.get(peerId)
   441→    if (pc) {
   442→      pc.close()
   443→    }
   444→    this.pcs.delete(peerId)
   445→    this.sendersMap.delete(peerId)
   446→    this.pendingIce.delete(peerId)
   447→    this.links.delete(peerId)
   448→    trackAudio(peerId, null)
   449→    putStream(peerId, null)
   450→    useCallStore.getState().bumpPeerStream(peerId, false)
   451→    useCallStore.getState().removePeer(peerId)
   452→  }
   453→
   454→  private reconcilePeers(
   455→    peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>,
   456→  ) {
   457→    const store = useCallStore.getState()
   458→    const seen = new Set<string>()
   459→    for (const p of peers) {
   460→      if (p.id === this.myId) continue
   461→      seen.add(p.id)
   462→      this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   463→    }
   464→    for (const id of Object.keys(store.peers)) {
   465→      if (!seen.has(id)) this.dropPeer(id)
   466→    }
   467→    // After a reconnect, old connections may be stale — rebuild them
   468→    for (const [peerId, pc] of this.pcs) {
   469→      if (pc.connectionState === 'failed' || pc.connectionState === 'closed') {
   470→        this.dropPeer(peerId)
   471→        const p = peers.find((x) => x.id === peerId)
   472→        if (p) this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   473→      }
   474→    }
   475→  }
   476→
   477→  private async flushIce(peerId: string) {
   478→    const pc = this.pcs.get(peerId)
   479→    const queued = this.pendingIce.get(peerId) ?? []
   480→    if (!pc || queued.length === 0) return
   481→    this.pendingIce.set(peerId, [])
   482→    for (const c of queued) {
   483→      try {
   484→        await pc.addIceCandidate(c)
   485→      } catch {
   486→        /* ignore */
   487→      }
   488→    }
   489→  }

   491→  /* ---------------- controls ---------------- */
   492→
   493→  toggleMic(): boolean {
   494→    const next = !useCallStore.getState().self.micOn
   495→    if (this.micTrack) this.micTrack.enabled = next
   496→    useCallStore.getState().setSelf({ micOn: next })
   497→    this.socket?.emit('state:update', { micOn: next })
   498→    return next
   499→  }
   500→
   501→  async toggleCam(): Promise<boolean> {
   502→    const next = !useCallStore.getState().self.camOn
   503→    useCallStore.getState().setSelf({ camOn: next })
   504→    this.socket?.emit('state:update', { camOn: next })
   505→
   506→    if (next && !this.cameraTrack && this.mediaMode !== 'none') {
   507→      // try to acquire a camera mid-call (video track added -> renegotiation fires)
   508→      try {
   509→        const profile = this.resolveProfile()
   510→        const stream = await navigator.mediaDevices.getUserMedia({ video: this.videoConstraints(profile) })
   511→        const track = stream.getVideoTracks()[0]
   512→        this.cameraTrack = track
   513→        track.enabled = true
   514→        if (this.localStream) this.localStream.addTrack(track)
   515→        else this.localStream = stream
   516→        this.publishLocalTrack(track, this.localStream)
   517→        putStream('self', this.localStream)
   518→        await this.setVideoSendersActive(true)
   519→      } catch {
   520→        useCallStore.getState().setSelf({ camOn: false })
   521→        this.socket?.emit('state:update', { camOn: false })
   522→        return false
   523→      }
   524→      return true
   525→    }
   526→
   527→    if (this.cameraTrack) this.cameraTrack.enabled = next
   528→    await this.setVideoSendersActive(next && !this.screenTrack)
   529→    return next
   530→  }

   532→  private publishLocalTrack(track: MediaStreamTrack, owner: MediaStream) {
   533→    for (const [peerId, pc] of this.pcs) {
   534→      const senders = this.sendersMap.get(peerId) ?? {}
   535→      try {
   536→        if (track.kind === 'video' && senders.video && senders.video.track?.kind === 'video') {
   537→          // Swap the outgoing source — no renegotiation, seamless for viewers
   538→          void senders.video.replaceTrack(track).catch(() => {
   539→            try {
   540→              const s = pc.addTrack(track, owner)
   541→              senders.video = s
   542→              this.sendersMap.set(peerId, senders)
   543→            } catch {
   544→              /* ignore */
   545→            }
   546→          })
   547→        } else {
   548→          const s = pc.addTrack(track, owner)
   549→          if (track.kind === 'audio') senders.audio = s
   550→          else senders.video = s
   551→          this.sendersMap.set(peerId, senders)
   552→        }
   553→      } catch {
   554→        /* ignore */
   555→      }
   556→    }
   557→  }
   558→
   559→  async toggleShare(): Promise<boolean> {
   560→    const sharing = !useCallStore.getState().self.sharing
   561→    if (sharing) {
   562→      try {
   563→        const display = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: false })
   564→        const track = display.getVideoTracks()[0]
   565→        this.screenTrack = track
   566→        track.addEventListener('ended', () => {
   567→          void this.stopShare()
   568→        })
   569→        this.publishLocalTrack(track, display)
   570→        // mirror the shared screen into the local preview tile
   571→        putStream('self', display)
   572→        useCallStore.getState().bumpPeerStream('self', true)
   573→        useCallStore.getState().setSelf({ sharing: true })
   574→        this.socket?.emit('state:update', { sharing: true })
   575→        useCallStore.getState().setSpotlight('self')
   576→        return true
   577→      } catch {
   578→        return false
   579→      }
   580→    } else {
   581→      await this.stopShare()
   582→      return false
   583→    }
   584→  }
   585→
   586→  private async stopShare() {
   587→    this.screenTrack?.stop()
   588→    this.screenTrack = null
   589→    if (this.cameraTrack && this.cameraTrack.readyState === 'live') {
   590→      // restore the camera as the outgoing video source and the local preview
   591→      this.publishLocalTrack(this.cameraTrack, this.localStream ?? new MediaStream([this.cameraTrack]))
   592→      if (this.localStream) putStream('self', this.localStream)
   593→      await this.setVideoSendersActive(useCallStore.getState().self.camOn)
   594→    } else {
   595→      // no camera available: stop sending video entirely
   596→      await this.setVideoSendersActive(false)
   597→    }
   598→    useCallStore.getState().setSelf({ sharing: false })
   599→    this.socket?.emit('state:update', { sharing: false })
   600→    const spotlight = useCallStore.getState().spotlight
   601→    if (spotlight === 'self') useCallStore.getState().setSpotlight(null)
   602→  }

   604→  private async setVideoSendersActive(active: boolean) {
   605→    for (const [, senders] of this.sendersMap) {
   606→      const sender = senders.video
   607→      if (!sender) continue
   608→      try {
   609→        const params = sender.getParameters()
   610→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   611→        params.encodings[0].active = active
   612→        await sender.setParameters(params)
   613→      } catch {
   614→        /* browser may not support active toggle */
   615→      }
   616→    }
   617→  }

   619→  private capAudioSender(pc: RTCPeerConnection) {
   620→    const sender = pc.getSenders().find((s) => s.track?.kind === 'audio')
   621→    if (!sender) return
   622→    try {
   623→      const params = sender.getParameters()
   624→      if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   625→      params.encodings[0].maxBitrate = AUDIO_SENDER_BITRATE
   626→      void sender.setParameters(params).catch(() => undefined)
   627→    } catch {
   628→      /* ignore */
   629→    }
   630→  }

   634→  private resolveProfile(): QualityProfile {
   635→    const preset = useCallStore.getState().preset
   636→    const key = preset === 'auto' ? this.adaptiveProfile : preset
   637→    return QUALITY_PROFILES[key]
   638→  }
   639→
   640→  private effectiveLabel(): string {
   641→    const preset = useCallStore.getState().preset
   642→    const profile = this.resolveProfile()
   643→    const short = profile.label.split('·')[1]?.trim() ?? ''
   644→    return preset === 'auto' ? `Auto · ${short}` : profile.label
   645→  }

   647→  async setPreset(p: QualityPreset) {
   648→    this.preset = p
   649→    useCallStore.getState().setPreset(p)
   650→    if (p !== 'auto') this.adaptiveProfile = p
   651→    useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   652→    await this.applyProfile()
   653→  }
   654→
   655→  private async applyProfile() {
   656→    const profile = this.resolveProfile()
   657→    // 1) Capture resolution / fps (no renegotiation needed)
   658→    if (this.cameraTrack && !this.screenTrack && this.cameraTrack.readyState === 'live') {
   659→      try {
   660→        await this.cameraTrack.applyConstraints(this.videoConstraints(profile))
   661→      } catch {
   662→        /* some cameras ignore constraints */
   663→      }
   664→    }
   665→    // 2) Bitrate caps per sender, respecting per-link adaptive caps
   666→    for (const [peerId, senders] of this.sendersMap) {
   667→      const sender = senders.video
   668→      if (!sender) continue
   669→      const link = this.links.get(peerId)
   670→      const cap = link?.lastCap ?? null
   671→      const target = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   672→      try {
   673→        const params = sender.getParameters()
   674→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   675→        params.encodings[0].maxBitrate = target
   676→        await sender.setParameters(params)
   677→      } catch {
   678→        /* ignore */
   679→      }
   680→    }
   681→  }

   683→  /* ---------------- stats / adaptation ---------------- */

   685→  private async pollStats() {
   686→    if (this.pcs.size === 0) return
   687→    let overall: NetworkLevel = 3
   688→    let totalKbps = 0
   689→    const nextPrev = new Map<string, number>()
   690→
   691→    for (const [peerId, pc] of this.pcs) {
   692→      if (pc.connectionState !== 'connected') {
   693→        overall = Math.min(overall, 1) as NetworkLevel
   694→        continue
   695→      }
   696→      let rtt: number | null = null
   697→      let loss: number | null = null
   698→      let bytesDelta = 0
   699→      try {
   700→        const stats = await pc.getStats()
   701→        stats.forEach((report) => {
   702→          const r = report as RTCStats & {
   703→            currentRoundTripTime?: number
   704→            fractionLost?: number
   705→            type: string
   706→            state?: string
   707→            nominated?: boolean
   708→            bytesSent?: number
   709→            kind?: string
   710→          }
   711→          if (r.type === 'candidate-pair' && r.state === 'succeeded' && typeof r.currentRoundTripTime === 'number') {
   712→            rtt = rtt === null ? r.currentRoundTripTime : Math.min(rtt, r.currentRoundTripTime)
   713→          }
   714→          if (r.type === 'remote-inbound-rtp' && typeof r.fractionLost === 'number') {
   715→            loss = loss === null ? r.fractionLost : Math.max(loss, r.fractionLost)
   716→          }
   717→          if (r.type === 'outbound-rtp' && r.kind === 'video' && typeof r.bytesSent === 'number') {
   718→            const prevB = this.prevBytes.get(r.id) ?? r.bytesSent
   719→            bytesDelta += Math.max(0, r.bytesSent - prevB)
   720→            nextPrev.set(r.id, r.bytesSent)
   721→          }
   722→        })
   723→      } catch {
   724→        continue
   725→      }
   726→
   727→      const level = classifyLink(rtt, loss)
   728→      const prevLevel = useCallStore.getState().peers[peerId]?.level
   729→      if (prevLevel !== level) useCallStore.getState().patchPeer(peerId, { level })
   730→      overall = Math.min(overall, level) as NetworkLevel
   731→      totalKbps += (bytesDelta * 8) / (STATS_INTERVAL_MS * 1000)
   732→
   733→      // Adaptive per-link video cap
   734→      const link = this.links.get(peerId)
   735→      const cap = LINK_CAPS[level]
   736→      if (link && link.lastCap !== cap) {
   737→        link.lastCap = cap
   738→        const sender = this.sendersMap.get(peerId)?.video
   739→        if (sender) {
   740→          try {
   741→            const params = sender.getParameters()
   742→            if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   743→            const profile = this.resolveProfile()
   744→            params.encodings[0].maxBitrate = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   745→            await sender.setParameters(params)
   746→          } catch {
   747→            /* ignore */
   748→          }
   749→        }
   750→      }
   751→    }
   752→
   753→    this.prevBytes = nextPrev
   754→
   755→    const prevOverall = useCallStore.getState().overallLevel
   756→    if (prevOverall !== overall) {
   757→      useCallStore.getState().setOverallLevel(overall)
   758→      useCallStore.getState().patchPeer(this.myId, { level: overall })
   759→    }
   760→    useCallStore.setState({ bitrateKbps: Math.round(totalKbps) })
   761→
   762→    // Auto quality profile with hysteresis
   763→    if (this.preset === 'auto') {
   764→      if (overall <= 1) {
   765→        this.badCount++
   766→        this.goodCount = 0
   767→      } else if (overall === 3) {
   768→        this.goodCount++
   769→        this.badCount = 0
   770→      } else {
   771→        this.badCount = 0
   772→        this.goodCount = 0
   773→      }
   774→      const order: Array<Exclude<QualityPreset, 'auto'>> = ['saver', 'balanced', 'sharp']
   775→      const idx = order.indexOf(this.adaptiveProfile)
   776→      if (this.badCount >= 2 && idx > 0) {
   777→        this.adaptiveProfile = order[idx - 1]
   778→        this.badCount = 0
   779→        this.goodCount = 0
   780→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   781→        void this.applyProfile()
   782→      } else if (this.goodCount >= 8 && idx < order.length - 1) {
   783→        this.adaptiveProfile = order[idx + 1]
   784→        this.badCount = 0
   785→        this.goodCount = 0
   786→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   787→        void this.applyProfile()
   788→      }
   789→    }
   790→  }

   800→  private rejoin() {
   801→    if (!this.roomId || !this.socket) return
   802→    this.socket.emit(
   803→      'room:join',
   804→      { roomId: this.roomId, name: useCallStore.getState().self.name },
   805→      (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   806→        // The server hands out a fresh socket id after a reconnect — adopt it,
   807→        // otherwise our own messages and presence updates stop matching.
   808→        if (res?.ok) {
   809→          if (res.self?.id && res.self.id !== this.myId) {
   810→            this.myId = res.self.id
   811→            useCallStore.getState().setSelf({ id: res.self.id })
   812→          }
   813→          if (res.peers) this.reconcilePeers(res.peers)
   814→        }
   815→      },
   816→    )
   817→  }

   819→  async leave(silent = false) {
   820→    if (this.statsTimer !== null) {
   821→      window.clearInterval(this.statsTimer)
   822→      this.statsTimer = null
   823→    }
   824→    this.socket?.emit('room:leave')
   825→    for (const peerId of Array.from(this.pcs.keys())) this.dropPeer(peerId)
   826→    this.screenTrack?.stop()
   827→    this.localStream?.getTracks().forEach((t) => {
   828→      try {
   829→        t.stop()
   830→      } catch {
   831→        /* ignore */
   832→      }
   833→    })
   834→    const selfStreamTest = (this.localStream as (MediaStream & { __stop?: () => void }) | null)?.__stop
   835→    if (selfStreamTest) selfStreamTest()
   836→    this.localStream = null
   837→    this.cameraTrack = null
   838→    this.micTrack = null
   839→    clearAllAudio()
   840→    putStream('self', null)
   841→    this.pcs.clear()
   842→    this.pendingIce.clear()
   843→    this.links.clear()
   844→    this.prevBytes.clear()
   845→    this.socket?.disconnect()
   846→    this.socket = null
   847→    this.myId = ''
   848→    this.started = false
   849→    if (!silent) {
   850→      useCallStore.getState().reset()
   851→    }
   852→  }
