    21→/**
    22→ * Pico call engine — peer-to-peer WebRTC mesh.
    23→ *
    24→ * Design goals:
    25→ *  - Media never touches a server: every audio/video packet travels directly
    26→ *    between participants (mesh topology). The signaling socket only carries
    27→ *    SDP/ICE/chat text.
    28→ *  - Renegotiation-free quality control: resolution/fps changes ride on
    29→ *    applyConstraints(), bitrate changes on setParameters(), screen share on
    30→ *    replaceTrack(). No expensive offer/answer churn.
    31→ *  - Opus DTX + 24 kbps cap: silence costs (almost) nothing on the wire.
    32→ *  - Per-link adaptive bitrate: getStats() every 3s classifies each
    33→ *    connection (RTT + packet loss) and caps outgoing video per link, so one
    34→ *    weak participant never drags the whole call down.
    35→ *  - Perfect-negotiation pattern (polite/impolite) for glare-free offers.
    36→ */
    37→
    38→const ICE_SERVERS: RTCIceServer[] = [
    39→  { urls: ['stun:stun.l.google.com:19302', 'stun:stun1.l.google.com:19302'] },
    40→  { urls: ['stun:stun.cloudflare.com:3478'] },
    41→]
    42→
    43→/**
    44→ * ICE servers with TURN support, fetched from /api/ice. TURN is what makes
    45→ * calls work between strict NATs (mobile carriers, offices, hotel wifi) —
    46→ * without it a meaningful slice of real-world pairs can never connect.
    47→ * Cached for an hour; a failed fetch falls back to STUN-only, never blocks
    48→ * the call for more than the 4s timeout.
    49→ */
    50→const ICE_CACHE_TTL_MS = 3_600_000
    51→let iceCache: { servers: RTCIceServer[]; at: number } | null = null
    52→
    53→async function fetchIceServers(): Promise<RTCIceServer[]> {
    54→  if (iceCache && Date.now() - iceCache.at < ICE_CACHE_TTL_MS) return iceCache.servers
    55→  try {
    56→    const ctrl = new AbortController()
    57→    const timer = window.setTimeout(() => ctrl.abort(), 4000)
    58→    const res = await fetch('/api/ice', { signal: ctrl.signal, cache: 'no-store' })
    59→    window.clearTimeout(timer)
    60→    if (res.ok) {
    61→      const data = (await res.json()) as { iceServers?: RTCIceServer[] }
    62→      if (Array.isArray(data.iceServers) && data.iceServers.length > 0) {
    63→        iceCache = { servers: data.iceServers, at: Date.now() }
    64→        return iceCache.servers
    65→      }
    66→    }
    67→  } catch {
    68→    /* unreachable or slow — STUN-only still works on friendly networks */
    69→  }
    70→  iceCache = { servers: ICE_SERVERS, at: Date.now() }
    71→  return iceCache.servers
    72→}

   113→/**
   114→ * Opus tuning per m-section. Speech sections get DTX + a 24 kbps ceiling so
   115→ * silence costs (almost) nothing; music sections get stereo and a high
   116→ * ceiling. Sections are recognized by their msid (local sends) or by mid
   117→ * (recvonly answers, where the msid lives in the remote offer).
   118→ */
   119→function mungeOpus(sdp: string, musicIds: Set<string>, musicMids: Set<string>): string {
   120→  const lines = sdp.split('\r\n')
   121→  const mIdx: number[] = []
   122→  lines.forEach((l, i) => {
   123→    if (l.startsWith('m=')) mIdx.push(i)
   124→  })
   125→  mIdx.push(lines.length)
   126→
   127→  const sectionMusic = new Array<boolean>(lines.length).fill(false)
   128→  for (let s = 0; s < mIdx.length - 1; s++) {
   129→    const from = mIdx[s]
   130→    const to = mIdx[s + 1]
   131→    let music = false
   132→    for (let i = from; i < to; i++) {
   133→      const msid = lines[i].match(/^a=msid:(\S+)/)
   134→      if (msid && musicIds.has(msid[1])) {
   135→        music = true
   136→        break
   137→      }
   138→      const mid = lines[i].match(/^a=mid:(.+)$/)
   139→      if (mid && musicMids.has(mid[1].trim())) {
   140→        music = true
   141→        break
   142→      }
   143→    }
   144→    for (let i = from; i < to; i++) sectionMusic[i] = music
   145→  }
   146→
   147→  const opusPTs = new Set<string>()
   148→  for (const line of lines) {
   149→    const m = line.match(/^a=rtpmap:(\d+) opus\/48000/i)
   150→    if (m) opusPTs.add(m[1])
   151→  }
   152→
   153→  const out = lines.map((line, i) => {
   154→    const m = line.match(/^a=fmtp:(\d+)(.*)$/)
   155→    if (!m || !opusPTs.has(m[1])) return line
   156→    let params = m[2] ?? ''
   157→    if (sectionMusic[i]) {
   158→      params = params.replace(/;?(usedtx|maxaveragebitrate|stereo)=[^;]*/gi, '')
   159→      params += ';stereo=1;maxaveragebitrate=128000'
   160→    } else {
   161→      params = params.replace(/;?(stereo|maxaveragebitrate)=[^;]*/gi, '')
   162→      if (!/usedtx/i.test(params)) params += ';usedtx=1'
   163→      if (!/maxaveragebitrate/i.test(params)) params += ';maxaveragebitrate=24000'
   164→    }
   165→    return `a=fmtp:${m[1]}${params}`
   166→  })
   167→  return out.join('\r\n')
   168→}

   170→function classifyLink(rttSec: number | null, lossFrac: number | null): NetworkLevel {
   171→  const rtt = rttSec ?? 0.4
   172→  const loss = lossFrac ?? 0.05
   173→  if (rtt > 0.5 || loss > 0.08) return 0
   174→  if (rtt > 0.3 || loss > 0.05) return 1
   175→  if (rtt > 0.15 || loss > 0.02) return 2
   176→  return 3
   177→}

   183→export class CallEngine {
   184→  private socket: Socket | null = null
   185→  private roomId = ''
   186→  private myId = ''
   187→  private localStream: MediaStream | null = null
   188→  private cameraTrack: MediaStreamTrack | null = null
   189→  private micTrack: MediaStreamTrack | null = null
   190→  private screenTrack: MediaStreamTrack | null = null
   191→  private pcs = new Map<string, RTCPeerConnection>()
   192→  private sendersMap = new Map<string, { audio?: RTCRtpSender; video?: RTCRtpSender; music?: RTCRtpSender }>()
   193→  private pendingIce = new Map<string, RTCIceCandidateInit[]>()
   194→  private links = new Map<string, LinkInfo>()
   195→  private statsTimer: number | null = null
   196→  private makingOffer = new Map<string, boolean>()
   197→  private ignoreOffer = new Map<string, boolean>()
   198→  /** Per-peer signaling op queue — SRD/SLD calls must never interleave. */
   199→  private sigQueue = new Map<string, Promise<void>>()
   200→  /** When our last offer went out per peer — drives the renegotiation watchdog. */
   201→  private lastOfferAt = new Map<string, number>()
   202→  private preset: QualityPreset = 'balanced'
   203→  private adaptiveProfile: Exclude<QualityPreset, 'auto'> = 'balanced'
   204→  private badCount = 0
   205→  private goodCount = 0
   206→  private started = false
   207→  private testMode = false
   208→  private mediaMode: MediaMode = 'full'
   209→  private prevBytes = new Map<string, number>()
   210→  private passwordHash = ''
   211→  private iceServers: RTCIceServer[] = ICE_SERVERS
   212→
   213→  /* listen together */
   214→  private musicSource: MusicSource | null = null
   215→  private musicHandleStop: (() => void) | null = null
   216→  private musicUnsub: (() => void) | null = null
   217→  private musicMetaTimer: number | null = null
   218→  private musicStreamId = ''
   219→  private musicOrigin: 'file' | 'test' | 'tab' = 'file'
   220→  private lastMusicBitrate = 0
   221→  private remoteMusicStreamIds = new Set<string>()
   222→  private musicMids = new Set<string>()
   223→  private lastRemoteSdp = ''
   224→
   225→  async start(opts: StartOptions): Promise<{ mode: MediaMode }> {
   226→    if (this.started) await this.leave(true)
   227→    this.started = true
   228→    this.roomId = opts.roomId
   229→    this.preset = opts.preset
   230→    this.adaptiveProfile = opts.preset === 'auto' ? 'balanced' : opts.preset
   231→    this.testMode = !!opts.testMode
   232→    this.passwordHash = opts.passwordHash ?? ''
   233→
   234→    const store = useCallStore.getState()
   235→    store.setSelf({ name: opts.name })
   236→    store.setPreset(opts.preset)
   237→    store.setEffectiveLabel(this.effectiveLabel())
   238→
   239→    // ---- ICE (TURN/STUN) — resolved before any peer connection is built ----
   240→    this.iceServers = await fetchIceServers()
   241→
   242→    // ---- Signaling ----
   243→    // Polling-first is deliberate: the initial handshake rides plain HTTP,
   244→    // which survives every proxy, VPN and captive portal on the planet, then
   245→    // socket.io transparently upgrades to websocket. Websocket-first dead-ends
   246→    // behind proxies that block or hang upgrades — there is no downgrade path.
   247→    const socket = io(SIGNALING_URL, {
   248→      path: SOCKET_PATH,
   249→      transports: ['polling', 'websocket'],
   250→      upgrade: true,
   251→      reconnection: true,
   252→      reconnectionAttempts: Infinity,
   253→      reconnectionDelay: 1000,
   254→      reconnectionDelayMax: 4000,
   255→      timeout: 10000,
   256→    })
   257→    this.socket = socket
   258→    // Dev/debug handle (harmless in prod, invaluable for live diagnostics)
   259→    if (typeof window !== 'undefined') {
   260→      ;(window as unknown as { __pico?: unknown }).__pico = { engine: this, socket }
   261→    }
   262→    socket.on('connect', () => {
   263→      useCallStore.getState().setSignalingConnected(true)
   264→      if (this.myId) this.rejoin()
   265→    })
   266→    socket.on('disconnect', () => useCallStore.getState().setSignalingConnected(false))
   267→    socket.on('connect_error', () => useCallStore.getState().setSignalingConnected(false))
   268→
   269→    // ---- Local media ----
   270→    this.localStream = opts.initialStream ?? null
   271→    if (this.localStream) {
   272→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   273→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   274→      this.mediaMode = this.cameraTrack ? 'full' : this.micTrack ? 'audio' : 'none'
   275→      // Register the handed-over preview stream so tiles can attach to it
   276→      putStream('self', this.localStream)
   277→    }
   278→    if (!this.localStream || (!this.cameraTrack && !this.micTrack)) {
   279→      this.mediaMode = await this.acquireMedia()
   280→    }
   281→    this.applyInitialTrackState()
   282→    trackAudio('self', this.localStream)
   283→
   284→    // ---- Room ----
   285→    // The ack races a timeout so a dead signaling server can never hang the UI.
   286→    // A rejected join (full room, wrong passcode, locked) aborts the start.
   287→    const joinResult = await new Promise<{ ok: boolean; reason?: string; hostId?: string; locked?: boolean }>((resolve) => {
   288→      let settled = false
   289→      const finish = (r: { ok: boolean; reason?: string }) => {
   290→        if (!settled) {
   291→          settled = true
   292→          resolve(r)
   293→        }
   294→      }
   295→      const guard = window.setTimeout(
   296→        () => finish({ ok: false, reason: 'Could not reach the call relay. Check your connection and try again.' }),
   297→        12000,
   298→      )
   299→      socket.emit(
   300→        'room:join',
   301→        { roomId: this.roomId, name: opts.name, passwordHash: this.passwordHash },
   302→        (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; hostId?: string; locked?: boolean; hasPassword?: boolean; music?: { peerId: string; title: string; playing: boolean; position: number; duration: number; streamId?: string } | null; reason?: string }) => {
   303→          window.clearTimeout(guard)
   304→          if (!res?.ok) {
   305→            finish({ ok: false, reason: res?.reason ?? 'Could not join the room' })
   306→            return
   307→          }
   308→          this.myId = res?.self?.id ?? socket.id!
   309→          useCallStore.getState().setSelf({ id: this.myId })
   310→          useCallStore.getState().setHostId(res.hostId ?? null)
   311→          useCallStore.getState().setLocked(!!res.locked)
   312→          useCallStore.getState().setPasscodeProtected(!!res.hasPassword)
   313→          if (res.music?.peerId) {
   314→            // Late joiner: the music track is already flowing — learn its
   315→            // stream id now so the incoming ontrack is routed to the player.
   316→            this.remoteMusicStreamIds.clear()
   317→            if (res.music.streamId) this.remoteMusicStreamIds.add(res.music.streamId)
   318→            useCallStore.getState().setMusic({
   319→              peerId: res.music.peerId,
   320→              title: res.music.title,
   321→              playing: res.music.playing,
   322→              position: res.music.position,
   323→              duration: res.music.duration,
   324→            })
   325→          }
   326→          if (res?.peers) this.reconcilePeers(res.peers)
   327→          finish({ ok: true })
   328→        },
   329→      )
   330→    })
   331→
   332→    if (!joinResult.ok) {
   333→      await this.cleanupAfterFailedJoin()
   334→      throw new Error(joinResult.reason ?? 'Could not join the room')
   335→    }
   336→
   337→    // ---- Events ----
   338→    socket.on('peer:joined', ({ peer }: { peer: { id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean } }) => {
   339→      this.ensurePeer(peer.id, peer.name, peer.micOn, peer.camOn, peer.sharing)
   340→      useCallStore.getState().addChat({
   341→        id: 'sys-' + Math.random(),
   342→        from: 'system',
   343→        name: 'System',
   344→        text: `${peer.name} joined`,
   345→        ts: Date.now(),
   346→        system: true,
   347→      })
   348→    })
   349→    socket.on('peer:left', ({ peerId }: { peerId: string }) => this.dropPeer(peerId))
   350→    socket.on('room:peers', ({ peers }: { peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   351→      // Only meaningful after a reconnect — reconcile full presence
   352→      if (this.myId) this.reconcilePeers(peers)
   353→    })
   354→    socket.on('peer:state', ({ peerId, micOn, camOn, sharing }: { peerId: string; micOn: boolean; camOn: boolean; sharing: boolean }) => {
   355→      const store = useCallStore.getState()
   356→      store.patchPeer(peerId, { micOn, camOn, sharing })
   357→      // Follow whoever is presenting, so a remote screen never hides in a small tile
   358→      if (sharing && store.spotlight !== peerId) store.setSpotlight(peerId)
   359→      else if (!sharing && store.spotlight === peerId) store.setSpotlight(null)
   360→    })
   361→    socket.on('chat:message', (msg: { id: string; from: string; name: string; text: string; ts: number }) => {
   362→      useCallStore.getState().addChat(msg)
   363→    })
   364→
   365→    /* ---- security & host controls ---- */
   366→    socket.on('host:changed', ({ hostId }: { hostId: string }) => {
   367→      useCallStore.getState().setHostId(hostId)
   368→      if (hostId === this.myId) {
   369→        toast('You are now the host', { duration: 2400 })
   370→        useCallStore.getState().addChat({
   371→          id: 'sys-' + Math.random(),
   372→          from: 'system',
   373→          name: 'System',
   374→          text: 'You are now the host',
   375→          ts: Date.now(),
   376→          system: true,
   377→        })
   378→      }
   379→    })
   380→    socket.on('host:force-mute', () => {
   381→      if (this.micTrack) this.micTrack.enabled = false
   382→      useCallStore.getState().setSelf({ micOn: false })
   383→      this.socket?.emit('state:update', { micOn: false })
   384→      toast('The host muted your microphone', { duration: 2800 })
   385→    })
   386→    socket.on('room:locked', ({ locked }: { locked: boolean }) => {
   387→      useCallStore.getState().setLocked(locked)
   388→      useCallStore.getState().addChat({
   389→        id: 'sys-' + Math.random(),
   390→        from: 'system',
   391→        name: 'System',
   392→        text: locked ? 'The host locked this call' : 'The host unlocked this call',
   393→        ts: Date.now(),
   394→        system: true,
   395→      })
   396→    })
   397→    socket.on('room:kicked', () => {
   398→      useCallStore.getState().setKicked(true)
   399→    })
   400→
   401→    /* ---- listen together ---- */
   402→    socket.on('music:state', (data: { peerId: string; action: 'start' | 'meta' | 'stop'; streamId?: string; title?: string; playing?: boolean; position?: number; duration?: number }) => {
   403→      const store = useCallStore.getState()
   404→      if (data.action === 'stop' || !data.peerId) {
   405→        this.remoteMusicStreamIds.clear()
   406→        detachRemoteMusic()
   407→        store.setMusic({ ...INITIAL_MUSIC })
   408→        return
   409→      }
   410→      if (data.action === 'start' && data.streamId) {
   411→        this.remoteMusicStreamIds.add(data.streamId)
   412→        store.setMusic({
   413→          peerId: data.peerId,
   414→          title: data.title ?? 'Music',
   415→          playing: !!data.playing,
   416→          position: data.position ?? 0,
   417→          duration: data.duration ?? 0,
   418→        })
   419→      } else if (data.action === 'meta') {
   420→        store.setMusic({
   421→          title: data.title ?? store.music.title,
   422→          playing: !!data.playing,
   423→          position: data.position ?? 0,
   424→          duration: data.duration ?? store.music.duration,
   425→        })
   426→      }
   427→    })
   428→    socket.on('music:busy', () => {
   429→      toast('Someone is already sharing music', { duration: 2400 })
   430→    })
   431→
   432→    socket.on('rtc:offer', ({ from, sdp }: { from: string; sdp: string }) => {
   433→      const pc = this.pcs.get(from)
   434→      if (!pc) return
   435→      const polite = this.myId < from
   436→      this.enqueueSig(from, async () => {
   437→        try {
   438→          // Re-evaluate glare inside the queue — the state may have moved while
   439→          // this op waited for the previous one.
   440→          const offerCollision = this.makingOffer.get(from) || pc.signalingState !== 'stable'
   441→          this.ignoreOffer.set(from, !polite && offerCollision)
   442→          if (this.ignoreOffer.get(from)) return
   443→          await pc.setRemoteDescription({ type: 'offer', sdp: this.munge(sdp) })
   444→          await this.flushIce(from)
   445→          const answer = await pc.createAnswer()
   446→          await pc.setLocalDescription({ type: 'answer', sdp: this.munge(answer.sdp ?? '', pc.remoteDescription?.sdp ?? '') })
   447→          const desc = pc.localDescription
   448→          if (desc) this.socket?.emit('rtc:answer', { to: from, sdp: desc.sdp })
   449→        } catch (err) {
   450→          // A peer that rebuilt its connection mid-session can offer a
   451→          // different m-line layout than ours; rebuild our side so the next
   452→          // offer lands on a matching PC instead of failing forever.
   453→          if (err instanceof Error && /m-line/i.test(err.message) && this.pcs.get(from) === pc) {
   454→            const info = useCallStore.getState().peers[from]
   455→            this.dropPeer(from)
   456→            this.ensurePeer(from, info?.name ?? 'Guest', info?.micOn ?? true, info?.camOn ?? true, info?.sharing ?? false)
   457→          }
   458→        }
   459→      })
   460→    })
   461→
   462→    socket.on('rtc:answer', ({ from, sdp }: { from: string; sdp: string }) => {
   463→      const pc = this.pcs.get(from)
   464→      if (!pc) return
   465→      this.enqueueSig(from, async () => {
   466→        try {
   467→          if (pc.signalingState !== 'have-local-offer') return
   468→          await pc.setRemoteDescription({ type: 'answer', sdp: this.munge(sdp, pc.localDescription?.sdp ?? '') })
   469→          await this.flushIce(from)
   470→        } catch {
   471→          /* the renegotiation watchdog re-offers if the pair stays stuck */
   472→        }
   473→      })
   474→    })
   475→
   476→    socket.on('rtc:ice', ({ from, candidate }: { from: string; candidate: RTCIceCandidateInit }) => {
   477→      const pc = this.pcs.get(from)
   478→      if (!pc) return
   479→      if (pc.remoteDescription) {
   480→        pc.addIceCandidate(candidate).catch(() => {
   481→          if (!this.ignoreOffer.get(from)) {
   482→            this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   483→          }
   484→        })
   485→      } else {
   486→        this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   487→      }
   488→    })
   489→
   490→    // ---- Stats loop ----
   491→    this.statsTimer = window.setInterval(() => void this.pollStats(), STATS_INTERVAL_MS)
   492→
   493→    return { mode: this.mediaMode }
   494→  }

   496→  /* ---------------- media ---------------- */
   497→
   498→  private async acquireMedia(): Promise<MediaMode> {
   499→    if (this.testMode) {
   500→      this.localStream = await createTestStream()
   501→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   502→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   503→      putStream('self', this.localStream)
   504→      return 'full'
   505→    }
   506→    const audio: MediaTrackConstraints = {
   507→      echoCancellation: true,
   508→      noiseSuppression: true,
   509→      autoGainControl: true,
   510→      channelCount: 1,
   511→    }
   512→    const video = this.videoConstraints(this.resolveProfile())
   513→    // Full quality first
   514→    try {
   515→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio, video })
   516→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   517→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   518→      this.attachLocalStream()
   519→      return 'full'
   520→    } catch {
   521→      /* fall through */
   522→    }
   523→    // Audio only
   524→    try {
   525→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio })
   526→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   527→      this.cameraTrack = null
   528→      this.attachLocalStream()
   529→      return 'audio'
   530→    } catch {
   531→      /* fall through */
   532→    }
   533→    // Listener mode
   534→    this.localStream = null
   535→    return 'none'
   536→  }

   541→      height: { ideal: profile.height },
   542→      frameRate: { ideal: profile.frameRate, max: profile.frameRate },
   543→      facingMode: 'user',
   544→    }
   545→  }

   559→  /* ---------------- peers ---------------- */
   560→
   561→  private ensurePeer(
   562→    peerId: string,
   563→    name: string,
   564→    micOn = true,
   565→    camOn = true,
   566→    sharing = false,
   567→  ) {
   568→    const store = useCallStore.getState()
   569→    if (!store.peers[peerId]) {
   570→      store.upsertPeer({ id: peerId, name, micOn, camOn, sharing, level: 3, hasStream: false, streamVersion: 0 })
   571→    }
   572→    if (this.pcs.has(peerId)) return
   573→
   574→    const pc = new RTCPeerConnection({
   575→      iceServers: this.iceServers,
   576→      iceCandidatePoolSize: 4,
   577→      bundlePolicy: 'max-bundle',
   578→      rtcpMuxPolicy: 'require',
   579→    })
   580→    this.pcs.set(peerId, pc)
   581→    this.links.set(peerId, { lastCap: null })
   582→    this.makingOffer.set(peerId, false)
   583→    this.ignoreOffer.set(peerId, false)
   584→
   585→    // Local media into this connection
   586→    const senders: { audio?: RTCRtpSender; video?: RTCRtpSender; music?: RTCRtpSender } = {}
   587→    if (this.localStream) {
   588→      for (const track of this.localStream.getTracks()) {
   589→        try {
   590→          const sender = pc.addTrack(track, this.localStream)
   591→          if (track.kind === 'audio') senders.audio = sender
   592→          else senders.video = sender
   593→        } catch {
   594→          /* ignore */
   595→        }
   596→      }
   597→      // A rebuilt connection (reconnect, ICE restart) must carry the CURRENT
   598→      // outgoing video — if the screen is being shared, that is the screen
   599→      // track, not the camera that localStream still holds.
   600→      if (this.screenTrack && this.screenTrack.readyState === 'live' && senders.video) {
   601→        void senders.video.replaceTrack(this.screenTrack).catch(() => undefined)
   602→      }
   603→      // Keep camera-off cameras off on fresh connections too
   604→      if (!useCallStore.getState().self.camOn && !this.screenTrack) {
   605→        void this.setVideoSendersActive(false)
   606→      }
   607→    } else {
   608→      const at = pc.addTransceiver('audio', { direction: 'recvonly' })
   609→      const vt = pc.addTransceiver('video', { direction: 'recvonly' })
   610→      senders.audio = at.sender
   611→      senders.video = vt.sender
   612→    }
   613→    this.sendersMap.set(peerId, senders)
   614→
   615→    // Reserve the music m-line when this peer is the active listen-together
   616→    // sharer: that PC's layout is [audio, video, music], and a connection
   617→    // rebuilt mid-session must match it or every offer fails m-line order
   618→    // checks on the polite side (silent negotiation deadlock).
   619→    const musicPeerId = useCallStore.getState().music.peerId
   620→    if (musicPeerId && musicPeerId === peerId && !senders.music) {
   621→      try {
   622→        const t = pc.addTransceiver('audio', { direction: 'recvonly' })
   623→        senders.music = t.sender
   624→      } catch {
   625→        /* ignore */
   626→      }
   627→    }
   628→
   629→    pc.onicecandidate = (e) => {
   630→      if (e.candidate) this.socket?.emit('rtc:ice', { to: peerId, candidate: e.candidate.toJSON() })
   631→    }
   632→
   633→    pc.ontrack = (e) => {
   634→      const stream = e.streams[0]
   635→      if (!stream) return
   636→      // Listen-together streams ride their own MediaStream — recognize them by
   637→      // the stream id the sharer announced before the track ever lands.
   638→      if (this.remoteMusicStreamIds.has(stream.id)) {
   639→        void attachRemoteMusic(stream)
   640→        return
   641→      }
   642→      putStream(peerId, stream)
   643→      trackAudio(peerId, stream)
   644→      useCallStore.getState().bumpPeerStream(peerId, true)
   645→    }
   646→
   647→    pc.onnegotiationneeded = () => {
   648→      this.enqueueSig(peerId, () => this.makeOffer(peerId, pc))
   649→    }
   650→
   651→    pc.onconnectionstatechange = () => {
   652→      if (pc.connectionState === 'failed') {
   653→        try {
   654→          pc.restartIce()
   655→        } catch {
   656→          /* ignore */
   657→        }
   658→      }
   659→    }
   660→
   661→    // Prime encoding params on the fresh senders
   662→    this.capAudioSender(pc)
   663→    const videoSender = senders.video
   664→    if (videoSender) {
   665→      try {
   666→        const params = videoSender.getParameters()
   667→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   668→        params.encodings[0].maxBitrate = this.resolveProfile().maxBitrate
   669→        void videoSender.setParameters(params).catch(() => undefined)
   670→      } catch {
   671→        /* ignore */
   672→      }
   673→    }
   674→    // Someone joining mid-song gets the music track immediately
   675→    if (this.musicSource) this.ensureMusicSender(peerId, pc)
   676→  }
   677→
   678→  private dropPeer(peerId: string) {
   679→    const pc = this.pcs.get(peerId)
   680→    if (pc) {
   681→      pc.close()
   682→    }
   683→    this.pcs.delete(peerId)
   684→    this.sendersMap.delete(peerId)
   685→    this.pendingIce.delete(peerId)
   686→    this.links.delete(peerId)
   687→    this.makingOffer.delete(peerId)
   688→    this.ignoreOffer.delete(peerId)
   689→    this.sigQueue.delete(peerId)
   690→    this.lastOfferAt.delete(peerId)
   691→    trackAudio(peerId, null)
   692→    putStream(peerId, null)
   693→    useCallStore.getState().bumpPeerStream(peerId, false)
   694→    useCallStore.getState().removePeer(peerId)
   695→    // If the music sharer left, the session is over
   696→    if (useCallStore.getState().music.peerId === peerId) {
   697→      detachRemoteMusic()
   698→      this.remoteMusicStreamIds.clear()
   699→      useCallStore.getState().setMusic({ ...INITIAL_MUSIC })
   700→    }
   701→  }

   703→  private reconcilePeers(
   704→    peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>,
   705→  ) {
   706→    const store = useCallStore.getState()
   707→    const seen = new Set<string>()
   708→    for (const p of peers) {
   709→      if (p.id === this.myId) continue
   710→      seen.add(p.id)
   711→      this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   712→    }
   713→    for (const id of Object.keys(store.peers)) {
   714→      if (!seen.has(id)) this.dropPeer(id)
   715→    }
   716→    // Follow whoever is presenting — also for late joiners and reconnects,
   717→    // not just on live peer:state events.
   718→    if (!store.spotlight) {
   719→      const presenter = peers.find((p) => p.sharing && p.id !== this.myId)
   720→      if (presenter) useCallStore.getState().setSpotlight(presenter.id)
   721→    }
   722→    // After a reconnect, old connections may be stale — rebuild them
   723→    for (const [peerId, pc] of this.pcs) {
   724→      if (pc.connectionState === 'failed' || pc.connectionState === 'closed') {
   725→        this.dropPeer(peerId)
   726→        const p = peers.find((x) => x.id === peerId)
   727→        if (p) this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   728→      }
   729→    }
   730→  }

   732→  private async flushIce(peerId: string) {
   733→    const pc = this.pcs.get(peerId)
   734→    const queued = this.pendingIce.get(peerId) ?? []
   735→    if (!pc || queued.length === 0) return
   736→    this.pendingIce.set(peerId, [])
   737→    for (const c of queued) {
   738→      try {
   739→        await pc.addIceCandidate(c)
   740→      } catch {
   741→        /* ignore */
   742→      }
   743→    }
   744→  }
   745→
   746→  /**
   747→   * Serialize signaling ops per peer. Perfect negotiation only works when
   748→   * setRemoteDescription / setLocalDescription never interleave — during
   749→   * simultaneous reconnects both peers offer at once, and an in-flight
   750→   * setLocalDescription makes the other side's implicit rollback reject.
   751→   * Without this queue that rejection was silently swallowed and the pair
   752→   * sat in have-local-offer forever.
   753→   */
   754→  private enqueueSig(peerId: string, op: () => Promise<void>): Promise<void> {
   755→    const prev = this.sigQueue.get(peerId) ?? Promise.resolve()
   756→    const next = prev.then(op, op)
   757→    this.sigQueue.set(peerId, next)
   758→    next.catch(() => undefined) // branch: keep the chain usable, log nothing
   759→    return next
   760→  }
   761→
   762→  private async makeOffer(peerId: string, pc: RTCPeerConnection) {
   763→    this.makingOffer.set(peerId, true)
   764→    try {
   765→      const offer = await pc.createOffer()
   766→      await pc.setLocalDescription({ type: 'offer', sdp: this.munge(offer.sdp ?? '') })
   767→      const desc = pc.localDescription
   768→      if (desc) {
   769→        this.lastOfferAt.set(peerId, Date.now())
   770→        this.socket?.emit('rtc:offer', { to: peerId, sdp: desc.sdp })
   771→      }
   772→    } finally {
   773→      this.makingOffer.set(peerId, false)
   774→    }
   775→  }

   779→  private musicIds(): Set<string> {
   780→    const ids = new Set<string>()

   786→  /** mids of sections that carry a music msid in a reference (offer) SDP. */
   787→  private midsWithMusicSdp(referenceSdp: string): Set<string> {
   788→    const ids = this.musicIds()
   789→    const mids = new Set<string>()
   790→    if (!referenceSdp || ids.size === 0) return mids
   791→    let mid = ''
   792→    for (const line of referenceSdp.split('\r\n')) {
   793→      if (line.startsWith('m=')) mid = ''
   794→      const m1 = line.match(/^a=mid:(.+)$/)
   795→      if (m1) mid = m1[1].trim()
   796→      const m2 = line.match(/^a=msid:(\S+)/)
   797→      if (m2 && mid && ids.has(m2[1])) mids.add(mid)
   798→    }
   799→    return mids
   800→  }
   801→
   802→  private munge(sdp: string, referenceSdp = ''): string {
   803→    const ids = this.musicIds()
   804→    if (ids.size === 0) return mungeOpus(sdp, ids, new Set())
   805→    return mungeOpus(sdp, ids, this.midsWithMusicSdp(referenceSdp))
   806→  }

   996→  async toggleCam(): Promise<boolean> {
   997→    const next = !useCallStore.getState().self.camOn
   998→    useCallStore.getState().setSelf({ camOn: next })
   999→    this.socket?.emit('state:update', { camOn: next })
  1000→
  1001→    if (next && !this.cameraTrack && this.mediaMode !== 'none') {
  1002→      // try to acquire a camera mid-call (video track added -> renegotiation fires)
  1003→      try {
  1004→        const profile = this.resolveProfile()

  1006→        const track = stream.getVideoTracks()[0]
  1007→        this.cameraTrack = track

  1010→        else this.localStream = stream
  1011→        this.publishLocalTrack(track, this.localStream)
  1012→        putStream('self', this.localStream)
  1013→        await this.setVideoSendersActive(true)
  1014→      } catch {
  1015→        useCallStore.getState().setSelf({ camOn: false })
  1016→        this.socket?.emit('state:update', { camOn: false })
  1017→        return false
  1018→      }
  1019→      return true
  1020→    }
  1021→
  1022→    if (this.cameraTrack) this.cameraTrack.enabled = next
  1023→    await this.setVideoSendersActive(next && !this.screenTrack)
  1024→    return next
  1025→  }

  1027→  private publishLocalTrack(track: MediaStreamTrack, owner: MediaStream) {
  1028→    for (const [peerId, pc] of this.pcs) {
  1029→      const senders = this.sendersMap.get(peerId) ?? {}
  1030→      try {
  1031→        if (track.kind === 'video' && senders.video && senders.video.track?.kind === 'video') {
  1032→          // Swap the outgoing source — no renegotiation, seamless for viewers
  1033→          void senders.video.replaceTrack(track).catch(() => {
  1034→            try {
  1035→              const s = pc.addTrack(track, owner)
  1036→              senders.video = s
  1037→              this.sendersMap.set(peerId, senders)
  1038→            } catch {
  1039→              /* ignore */
  1040→            }
  1041→          })
  1042→        } else {
  1043→          const s = pc.addTrack(track, owner)
  1044→          if (track.kind === 'audio') senders.audio = s
  1045→          else senders.video = s
  1046→          this.sendersMap.set(peerId, senders)
  1047→        }
  1048→      } catch {
  1049→        /* ignore */
  1050→      }
  1051→    }
  1052→  }

  1054→  async toggleShare(): Promise<'started' | 'stopped' | 'cancelled'> {
  1055→    const sharing = useCallStore.getState().self.sharing
  1056→    if (!sharing) {
  1057→      try {
  1058→        const display = this.testMode
  1059→          ? await createTestScreenStream()
  1060→          : await navigator.mediaDevices.getDisplayMedia({ video: true, audio: true })
  1061→        const track = display.getVideoTracks()[0]
  1062→        if (!track) return 'cancelled'
  1063→        this.screenTrack = track
  1064→        track.addEventListener('ended', () => {
  1065→          void this.stopShare()
  1066→        })
  1067→        this.publishLocalTrack(track, display)
  1068→        // mirror the shared screen into the local preview tile
  1069→        putStream('self', display)
  1070→        useCallStore.getState().bumpPeerStream('self', true)
  1071→        useCallStore.getState().setSelf({ sharing: true })
  1072→        this.socket?.emit('state:update', { sharing: true })
  1073→        useCallStore.getState().setSpotlight('self')
  1074→        // Tab / system audio, when the browser hands it over, rides the
  1075→        // music channel (dedicated Opus stereo sender) — not the mic.
  1076→        const audioTrack = display.getAudioTracks()[0]
  1077→        if (audioTrack && !this.musicSource) {
  1078→          this.startMusicShare(
  1079→            {
  1080→              stream: new MediaStream([audioTrack]),
  1081→              title: 'Shared tab audio',
  1082→              duration: 0,
  1083→              getPosition: () => 0,
  1084→              isPlaying: () => true,
  1085→            },
  1086→            'tab',
  1087→          )
  1088→        }
  1089→        return 'started'
  1090→      } catch {
  1091→        return 'cancelled'
  1092→      }
  1093→    } else {
  1094→      await this.stopShare()
  1095→      return 'stopped'
  1096→    }
  1097→  }

  1099→  private async stopShare() {
  1100→    this.screenTrack?.stop()
  1101→    this.screenTrack = null
  1102→    if (this.musicOrigin === 'tab' && this.musicSource) {
  1103→      this.stopMusicShare()
  1104→    }
  1105→    if (this.cameraTrack && this.cameraTrack.readyState === 'live') {
  1106→      // restore the camera as the outgoing video source and the local preview
  1107→      this.publishLocalTrack(this.cameraTrack, this.localStream ?? new MediaStream([this.cameraTrack]))

  1109→      await this.setVideoSendersActive(useCallStore.getState().self.camOn)

  1111→      // no camera available: stop sending video entirely
  1112→      await this.setVideoSendersActive(false)

  1115→    this.socket?.emit('state:update', { sharing: false })

  1117→    if (spotlight === 'self') useCallStore.getState().setSpotlight(null)
  1118→  }

  1171→  private async applyProfile() {
  1172→    const profile = this.resolveProfile()
  1173→    // 1) Capture resolution / fps (no renegotiation needed)
  1174→    if (this.cameraTrack && !this.screenTrack && this.cameraTrack.readyState === 'live') {
  1175→      try {
  1176→        await this.cameraTrack.applyConstraints(this.videoConstraints(profile))
  1177→      } catch {
  1178→        /* some cameras ignore constraints */
  1179→      }
  1180→    }
  1181→    // 2) Bitrate caps per sender, respecting per-link adaptive caps
  1182→    for (const [peerId, senders] of this.sendersMap) {
  1183→      const sender = senders.video
  1184→      if (!sender) continue
  1185→      const link = this.links.get(peerId)
  1186→      const cap = link?.lastCap ?? null
  1187→      const target = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
  1188→      try {
  1189→        const params = sender.getParameters()
  1190→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
  1191→        params.encodings[0].maxBitrate = target
  1192→        await sender.setParameters(params)
  1193→      } catch {
  1194→        /* ignore */
  1195→      }
  1196→    }
  1197→  }

  1201→  private async pollStats() {
  1202→    if (this.pcs.size === 0) return
  1203→
  1204→    // Signaling watchdog: an offer left hanging for 9s+ (a rare race between
  1205→    // simultaneous reconnects) self-heals with a fresh offer.
  1206→    for (const [peerId, pc] of this.pcs) {
  1207→      if (pc.signalingState !== 'have-local-offer') continue
  1208→      if (Date.now() - (this.lastOfferAt.get(peerId) ?? 0) < 9000) continue
  1209→      if (!this.socket?.connected) continue
  1210→      this.lastOfferAt.set(peerId, Date.now())
  1211→      this.enqueueSig(peerId, () => this.makeOffer(peerId, pc))
  1212→    }
  1213→
  1214→    let overall: NetworkLevel = 3
  1215→    let totalKbps = 0
  1216→    const nextPrev = new Map<string, number>()
  1217→
  1218→    for (const [peerId, pc] of this.pcs) {
  1219→      if (pc.connectionState !== 'connected') {
  1220→        overall = Math.min(overall, 1) as NetworkLevel
  1221→        continue
  1222→      }
  1223→      let rtt: number | null = null
  1224→      let loss: number | null = null
  1225→      let bytesDelta = 0
  1226→      try {
  1227→        const stats = await pc.getStats()
  1228→        stats.forEach((report) => {
  1229→          const r = report as RTCStats & {
  1230→            currentRoundTripTime?: number
  1231→            fractionLost?: number
  1232→            type: string
  1233→            state?: string
  1234→            nominated?: boolean
  1235→            bytesSent?: number
  1236→            kind?: string
  1237→          }
  1238→          if (r.type === 'candidate-pair' && r.state === 'succeeded' && typeof r.currentRoundTripTime === 'number') {
  1239→            rtt = rtt === null ? r.currentRoundTripTime : Math.min(rtt, r.currentRoundTripTime)
  1240→          }
  1241→          if (r.type === 'remote-inbound-rtp' && typeof r.fractionLost === 'number') {
  1242→            loss = loss === null ? r.fractionLost : Math.max(loss, r.fractionLost)
  1243→          }
  1244→          if (r.type === 'outbound-rtp' && typeof r.bytesSent === 'number' && (r.kind === 'video' || r.kind === 'audio')) {
  1245→            const prevB = this.prevBytes.get(r.id) ?? r.bytesSent
  1246→            bytesDelta += Math.max(0, r.bytesSent - prevB)
  1247→            nextPrev.set(r.id, r.bytesSent)
  1248→          }
  1249→        })
  1250→      } catch {
  1251→        continue
  1252→      }
  1253→
  1254→      const level = classifyLink(rtt, loss)
  1255→      const prevLevel = useCallStore.getState().peers[peerId]?.level
  1256→      if (prevLevel !== level) useCallStore.getState().patchPeer(peerId, { level })
  1257→      overall = Math.min(overall, level) as NetworkLevel
  1258→      totalKbps += (bytesDelta * 8) / (STATS_INTERVAL_MS * 1000)
  1259→
  1260→      // Adaptive per-link video cap
  1261→      const link = this.links.get(peerId)
  1262→      const cap = LINK_CAPS[level]
  1263→      if (link && link.lastCap !== cap) {
  1264→        link.lastCap = cap
  1265→        const sender = this.sendersMap.get(peerId)?.video
  1266→        if (sender) {
  1267→          try {
  1268→            const params = sender.getParameters()
  1269→            if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
  1270→            const profile = this.resolveProfile()
  1271→            params.encodings[0].maxBitrate = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
  1272→            await sender.setParameters(params)
  1273→          } catch {
  1274→            /* ignore */
  1275→          }
  1276→        }
  1277→      }
  1278→    }
  1279→
  1280→    this.prevBytes = nextPrev
  1281→
  1282→    const prevOverall = useCallStore.getState().overallLevel
  1283→    if (prevOverall !== overall) {
  1284→      useCallStore.getState().setOverallLevel(overall)
  1285→      useCallStore.getState().patchPeer(this.myId, { level: overall })
  1286→    }
  1287→    useCallStore.setState({ bitrateKbps: Math.round(totalKbps) })
  1288→
  1289→    // Music quality follows the network instead of fighting it
  1290→    if (this.musicSource) this.applyMusicBitrate(overall)
  1291→
  1292→    // Auto quality profile with hysteresis
  1293→    if (this.preset === 'auto') {
  1294→      if (overall <= 1) {
  1295→        this.badCount++
  1296→        this.goodCount = 0
  1297→      } else if (overall === 3) {
  1298→        this.goodCount++
  1299→        this.badCount = 0
  1300→      } else {
  1301→        this.badCount = 0
  1302→        this.goodCount = 0
  1303→      }
  1304→      const order: Array<Exclude<QualityPreset, 'auto'>> = ['saver', 'balanced', 'sharp']
  1305→      const idx = order.indexOf(this.adaptiveProfile)
  1306→      if (this.badCount >= 2 && idx > 0) {
  1307→        this.adaptiveProfile = order[idx - 1]
  1308→        this.badCount = 0
  1309→        this.goodCount = 0
  1310→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
  1311→        void this.applyProfile()
  1312→      } else if (this.goodCount >= 8 && idx < order.length - 1) {
  1313→        this.adaptiveProfile = order[idx + 1]
  1314→        this.badCount = 0
  1315→        this.goodCount = 0
  1316→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
  1317→        void this.applyProfile()
  1318→      }
  1319→    }
  1320→  }

  1330→  private rejoin() {
  1331→    if (!this.roomId || !this.socket) return
  1332→    this.socket.emit(
  1333→      'room:join',
  1334→      { roomId: this.roomId, name: useCallStore.getState().self.name, passwordHash: this.passwordHash },
  1335→      (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; hostId?: string; locked?: boolean; hasPassword?: boolean; music?: { peerId: string; title: string; playing: boolean; position: number; duration: number; streamId?: string } | null }) => {
  1336→        // The server hands out a fresh socket id after a reconnect — adopt it,
  1337→        // otherwise our own messages and presence updates stop matching.
  1338→        if (res?.ok) {
  1339→          if (res.self?.id && res.self.id !== this.myId) {
  1340→            this.myId = res.self.id
  1341→            useCallStore.getState().setSelf({ id: res.self.id })
  1342→          }
  1343→          if (res.hostId) useCallStore.getState().setHostId(res.hostId)
  1344→          useCallStore.getState().setLocked(!!res.locked)
  1345→          useCallStore.getState().setPasscodeProtected(!!res.hasPassword)
  1346→          // Restore the listen-together session (if it survived the reconnect)
  1347→          // so the incoming music track is routed to the player, not a tile.
  1348→          if (res.music?.peerId && res.music.peerId !== this.myId) {
  1349→            this.remoteMusicStreamIds.clear()
  1350→            if (res.music.streamId) this.remoteMusicStreamIds.add(res.music.streamId)
  1351→            useCallStore.getState().setMusic({
  1352→              peerId: res.music.peerId,
  1353→              title: res.music.title,
  1354→              playing: res.music.playing,
  1355→              position: res.music.position,
  1356→              duration: res.music.duration,
  1357→            })
  1358→          }
  1359→          if (res.peers) this.reconcilePeers(res.peers)
  1360→          // Re-announce our live state: the server rebuilt our peer record
  1361→          // with defaults, so everyone else must re-learn mic/cam/sharing —
  1362→          // otherwise a mid-share reconnect drops the remote spotlight.
  1363→          this.socket?.emit('state:update', {
  1364→            micOn: useCallStore.getState().self.micOn,
  1365→            camOn: useCallStore.getState().self.camOn,
  1366→            sharing: !!(this.screenTrack && this.screenTrack.readyState === 'live'),
  1367→          })
  1368→          // Same for an active listen-together session: the server dropped it
  1369→          // when our old socket left, so re-announce to restore playback for
  1370→          // everyone (and our own late-join mapping on the next ontrack).
  1371→          if (this.musicSource && this.musicStreamId) {
  1372→            this.socket?.emit('music:state', {
  1373→              action: 'start',
  1374→              streamId: this.musicStreamId,
  1375→              title: this.musicSource.title,
  1376→              playing: this.musicSource.isPlaying(),
  1377→              position: this.musicSource.getPosition(),
  1378→              duration: this.musicSource.duration,
  1379→            })
  1380→          }
  1381→        }
  1382→      },
  1383→    )
  1384→  }

  1386→  async leave(silent = false) {
  1387→    if (this.statsTimer !== null) {
  1388→      window.clearInterval(this.statsTimer)
  1389→      this.statsTimer = null
  1390→    }
  1391→    this.stopMusicShare()
  1392→    detachRemoteMusic()
  1393→    this.socket?.emit('room:leave')
  1394→    for (const peerId of Array.from(this.pcs.keys())) this.dropPeer(peerId)
  1395→    this.screenTrack?.stop()
  1396→    this.localStream?.getTracks().forEach((t) => {
  1397→      try {
  1398→        t.stop()
  1399→      } catch {
  1400→        /* ignore */
  1401→      }
  1402→    })
  1403→    const selfStreamTest = (this.localStream as (MediaStream & { __stop?: () => void }) | null)?.__stop
  1404→    if (selfStreamTest) selfStreamTest()
  1405→    this.localStream = null
  1406→    this.cameraTrack = null
  1407→    this.micTrack = null
  1408→    clearAllAudio()
  1409→    putStream('self', null)
  1410→    this.pcs.clear()
  1411→    this.pendingIce.clear()
  1412→    this.links.clear()
  1413→    this.prevBytes.clear()
  1414→    this.socket?.disconnect()
  1415→    this.socket = null
  1416→    this.myId = ''
  1417→    this.started = false
  1418→    if (!silent) {
  1419→      useCallStore.getState().reset()
  1420→    }
  1421→  }
