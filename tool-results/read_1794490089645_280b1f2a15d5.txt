    14→    14→/**
    15→    15→ * Pico call engine — peer-to-peer WebRTC mesh.
    16→    16→ *
    17→    17→ * Design goals:
    18→    18→ *  - Media never touches a server: every audio/video packet travels directly
    19→    19→ *    between participants (mesh topology). The signaling socket only carries
    20→    20→ *    SDP/ICE/chat text.
    21→    21→ *  - Renegotiation-free quality control: resolution/fps changes ride on
    22→    22→ *    applyConstraints(), bitrate changes on setParameters(), screen share on
    23→    23→ *    replaceTrack(). No expensive offer/answer churn.
    24→    24→ *  - Opus DTX + 24 kbps cap: silence costs (almost) nothing on the wire.
    25→    25→ *  - Per-link adaptive bitrate: getStats() every 3s classifies each
    26→    26→ *    connection (RTT + packet loss) and caps outgoing video per link, so one
    27→    27→ *    weak participant never drags the whole call down.
    28→    28→ *  - Perfect-negotiation pattern (polite/impolite) for glare-free offers.
    29→    29→ */

    58→    58→/** Add Opus DTX + low bitrate cap to SDP — near-zero cost for silence. */
    59→    59→function mungeOpus(sdp: string): string {
    60→    60→  const lines = sdp.split('\r\n')
    61→    61→  const opusPTs = new Set<string>()
    62→    62→  for (const line of lines) {
    63→    63→    const m = line.match(/^a=rtpmap:(\d+) opus\/48000/i)
    64→    64→    if (m) opusPTs.add(m[1])
    65→    65→  }
    66→    66→  const out = lines.map((line) => {
    67→    67→    const m = line.match(/^a=fmtp:(\d+)(.*)$/)
    68→    68→    if (m && opusPTs.has(m[1])) {
    69→    69→      let params = m[2] ?? ''
    70→    70→      if (!/usedtx/i.test(params)) params += ';usedtx=1'
    71→    71→      if (!/maxaveragebitrate/i.test(params)) params += ';maxaveragebitrate=24000'
    72→    72→      return `a=fmtp:${m[1]}${params}`
    73→    73→    }
    74→    74→    return line
    75→    75→  })
    76→    76→  return out.join('\r\n')
    77→    77→}
    78→    78→
    79→    79→function classifyLink(rttSec: number | null, lossFrac: number | null): NetworkLevel {
    80→    80→  const rtt = rttSec ?? 0.4
    81→    81→  const loss = lossFrac ?? 0.05
    82→    82→  if (rtt > 0.5 || loss > 0.08) return 0
    83→    83→  if (rtt > 0.3 || loss > 0.05) return 1
    84→    84→  if (rtt > 0.15 || loss > 0.02) return 2
    85→    85→  return 3
    86→    86→}

   102→   102→  private pendingIce = new Map<string, RTCIceCandidateInit[]>()
   103→   103→  private links = new Map<string, LinkInfo>()
   104→   104→  private statsTimer: number | null = null
   105→   105→  private makingOffer = new Map<string, boolean>()
   106→   106→  private ignoreOffer = new Map<string, boolean>()
   107→   107→  private preset: QualityPreset = 'balanced'
   108→   108→  private adaptiveProfile: Exclude<QualityPreset, 'auto'> = 'balanced'
   109→   109→  private badCount = 0
   110→   110→  private goodCount = 0
   111→   111→  private started = false

   116→   116→  async start(opts: StartOptions): Promise<{ mode: MediaMode }> {
   117→   117→    if (this.started) await this.leave(true)
   118→   118→    this.started = true
   119→   119→    this.roomId = opts.roomId
   120→   120→    this.preset = opts.preset
   121→   121→    this.adaptiveProfile = opts.preset === 'auto' ? 'balanced' : opts.preset
   122→   122→    this.testMode = !!opts.testMode
   123→   123→
   124→   124→    const store = useCallStore.getState()
   125→   125→    store.setSelf({ name: opts.name })
   126→   126→    store.setPreset(opts.preset)
   127→   127→    store.setEffectiveLabel(this.effectiveLabel())
   128→   128→
   129→   129→    // ---- Signaling ----
   130→   130→    const socket = io('/?XTransformPort=3003', {
   131→   131→      path: '/',
   132→   132→      transports: ['websocket', 'polling'],
   133→   133→      reconnection: true,
   134→   134→      reconnectionAttempts: Infinity,
   135→   135→      reconnectionDelay: 1000,
   136→   136→      reconnectionDelayMax: 4000,
   137→   137→      timeout: 8000,
   138→   138→    })
   139→   139→    this.socket = socket
   140→   140→    // Dev/debug handle (harmless in prod, invaluable for live diagnostics)
   141→   141→    if (typeof window !== 'undefined') {
   142→   142→      ;(window as unknown as { __pico?: unknown }).__pico = { engine: this, socket }
   143→   143→    }
   144→   144→    socket.on('connect', () => {
   145→   145→      useCallStore.getState().setSignalingConnected(true)
   146→   146→      if (this.myId) this.rejoin()
   147→   147→    })
   148→   148→    socket.on('disconnect', () => useCallStore.getState().setSignalingConnected(false))
   149→   149→    socket.on('connect_error', () => useCallStore.getState().setSignalingConnected(false))
   150→   150→
   151→   151→    // ---- Local media ----
   152→   152→    this.localStream = opts.initialStream ?? null
   153→   153→    if (this.localStream) {
   154→   154→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   155→   155→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   156→   156→      this.mediaMode = this.cameraTrack ? 'full' : this.micTrack ? 'audio' : 'none'
   157→   157→      // Register the handed-over preview stream so tiles can attach to it
   158→   158→      putStream('self', this.localStream)
   159→   159→    }
   160→   160→    if (!this.localStream || (!this.cameraTrack && !this.micTrack)) {
   161→   161→      this.mediaMode = await this.acquireMedia()
   162→   162→    }
   163→   163→    this.applyInitialTrackState()
   164→   164→
   165→   165→    // ---- Room ----
   166→   166→    await new Promise<void>((resolve) => {
   167→   167→      socket.emit(
   168→   168→        'room:join',
   169→   169→        { roomId: this.roomId, name: opts.name },
   170→   170→        (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; reason?: string }) => {
   171→   171→          if (!res?.ok) {
   172→   172→            useCallStore.getState().addChat({
   173→   173→              id: 'sys-' + Date.now(),
   174→   174→              from: 'system',
   175→   175→              name: 'System',
   176→   176→              text: res?.reason ?? 'Could not join the room',
   177→   177→              ts: Date.now(),
   178→   178→              system: true,
   179→   179→            })
   180→   180→          }
   181→   181→          this.myId = res?.self?.id ?? socket.id!
   182→   182→          useCallStore.getState().setSelf({ id: this.myId })
   183→   183→          if (res?.peers) this.reconcilePeers(res.peers)
   184→   184→          resolve()
   185→   185→        },
   186→   186→      )
   187→   187→    })
   188→   188→
   189→   189→    // ---- Events ----
   190→   190→    socket.on('peer:joined', ({ peer }: { peer: { id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean } }) => {
   191→   191→      this.ensurePeer(peer.id, peer.name, peer.micOn, peer.camOn, peer.sharing)
   192→   192→      useCallStore.getState().addChat({
   193→   193→        id: 'sys-' + Math.random(),
   194→   194→        from: 'system',
   195→   195→        name: 'System',
   196→   196→        text: `${peer.name} joined`,
   197→   197→        ts: Date.now(),
   198→   198→        system: true,
   199→   199→      })
   200→   200→    })
   201→   201→    socket.on('peer:left', ({ peerId }: { peerId: string }) => this.dropPeer(peerId))
   202→   202→    socket.on('room:peers', ({ peers }: { peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   203→   203→      // Only meaningful after a reconnect — reconcile full presence
   204→   204→      if (this.myId) this.reconcilePeers(peers)
   205→   205→    })
   206→   206→    socket.on('peer:state', ({ peerId, micOn, camOn, sharing }: { peerId: string; micOn: boolean; camOn: boolean; sharing: boolean }) => {
   207→   207→      useCallStore.getState().patchPeer(peerId, { micOn, camOn, sharing })
   208→   208→    })
   209→   209→    socket.on('chat:message', (msg: { id: string; from: string; name: string; text: string; ts: number }) => {
   210→   210→      useCallStore.getState().addChat(msg)
   211→   211→    })
   212→   212→
   213→   213→    socket.on('rtc:offer', async ({ from, sdp }: { from: string; sdp: string }) => {
   214→   214→      try {
   215→   215→        const pc = this.pcs.get(from)
   216→   216→        if (!pc) return
   217→   217→        const polite = this.myId < from
   218→   218→        const offerCollision = this.makingOffer.get(from) || pc.signalingState !== 'stable'
   219→   219→        this.ignoreOffer.set(from, !polite && offerCollision)
   220→   220→        if (this.ignoreOffer.get(from)) return
   221→   221→        await pc.setRemoteDescription({ type: 'offer', sdp: mungeOpus(sdp) })
   222→   222→        await this.flushIce(from)
   223→   223→        const answer = await pc.createAnswer()
   224→   224→        answer.sdp = mungeOpus(answer.sdp ?? '')
   225→   225→        await pc.setLocalDescription(answer)
   226→   226→        this.socket?.emit('rtc:answer', { to: from, sdp: pc.localDescription!.sdp })
   227→   227→      } catch {
   228→   228→        /* negotiation raced — next round will fix it */
   229→   229→      }
   230→   230→    })
   231→   231→
   232→   232→    socket.on('rtc:answer', async ({ from, sdp }: { from: string; sdp: string }) => {
   233→   233→      try {
   234→   234→        const pc = this.pcs.get(from)
   235→   235→        if (!pc || pc.signalingState !== 'have-local-offer') return
   236→   236→        await pc.setRemoteDescription({ type: 'answer', sdp: mungeOpus(sdp) })
   237→   237→        await this.flushIce(from)
   238→   238→      } catch {
   239→   239→        /* ignore */
   240→   240→      }
   241→   241→    })
   242→   242→
   243→   243→    socket.on('rtc:ice', ({ from, candidate }: { from: string; candidate: RTCIceCandidateInit }) => {
   244→   244→      const pc = this.pcs.get(from)
   245→   245→      if (!pc) return
   246→   246→      if (pc.remoteDescription) {
   247→   247→        pc.addIceCandidate(candidate).catch(() => {
   248→   248→          if (!this.ignoreOffer.get(from)) {
   249→   249→            this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   250→   250→          }
   251→   251→        })
   252→   252→      } else {
   253→   253→        this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   254→   254→      }
   255→   255→    })
   256→   256→
   257→   257→    // ---- Stats loop ----
   258→   258→    this.statsTimer = window.setInterval(() => void this.pollStats(), STATS_INTERVAL_MS)
   259→   259→
   260→   260→    return { mode: this.mediaMode }
   261→   261→  }
   262→   262→
   263→   263→  /* ---------------- media ---------------- */
   264→   264→
   265→   265→  private async acquireMedia(): Promise<MediaMode> {
   266→   266→    if (this.testMode) {
   267→   267→      this.localStream = await createTestStream()
   268→   268→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   269→   269→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   270→   270→      putStream('self', this.localStream)
   271→   271→      return 'full'
   272→   272→    }
   273→   273→    const audio: MediaTrackConstraints = {
   274→   274→      echoCancellation: true,
   275→   275→      noiseSuppression: true,
   276→   276→      autoGainControl: true,
   277→   277→      channelCount: 1,
   278→   278→    }
   279→   279→    const video = this.videoConstraints(this.resolveProfile())
   280→   280→    // Full quality first
   281→   281→    try {
   282→   282→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio, video })
   283→   283→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   284→   284→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   285→   285→      this.attachLocalStream()
   286→   286→      return 'full'
   287→   287→    } catch {
   288→   288→      /* fall through */
   289→   289→    }
   290→   290→    // Audio only
   291→   291→    try {
   292→   292→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio })
   293→   293→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   294→   294→      this.cameraTrack = null
   295→   295→      this.attachLocalStream()
   296→   296→      return 'audio'
   297→   297→    } catch {
   298→   298→      /* fall through */
   299→   299→    }
   300→   300→    // Listener mode
   301→   301→    this.localStream = null
   302→   302→    return 'none'
   303→   303→  }
   304→   304→
   305→   305→  private videoConstraints(profile: QualityProfile): MediaTrackConstraints {
   306→   306→    return {
   307→   307→      width: { ideal: profile.width },
   308→   308→      height: { ideal: profile.height },
   309→   309→      frameRate: { ideal: profile.frameRate, max: profile.frameRate },
   310→   310→      facingMode: 'user',
   311→   311→    }
   312→   312→  }
   313→   313→
   314→   314→  private attachLocalStream() {
   315→   315→    if (!this.localStream) return
   316→   316→    putStream('self', this.localStream)
   317→   317→  }

   319→   319→  private applyInitialTrackState() {
   320→   320→    const s = useCallStore.getState().self
   321→   321→    if (this.cameraTrack) this.cameraTrack.enabled = s.camOn
   322→   322→    if (this.micTrack) this.micTrack.enabled = s.micOn
   323→   323→    if (!s.camOn) void this.setVideoSendersActive(false)
   324→   324→  }

   326→   326→  /* ---------------- peers ---------------- */
   327→   327→
   328→   328→  private ensurePeer(
   329→   329→    peerId: string,
   330→   330→    name: string,
   331→   331→    micOn = true,
   332→   332→    camOn = true,
   333→   333→    sharing = false,
   334→   334→  ) {
   335→   335→    const store = useCallStore.getState()
   336→   336→    if (!store.peers[peerId]) {
   337→   337→      store.upsertPeer({ id: peerId, name, micOn, camOn, sharing, level: 3, hasStream: false, streamVersion: 0 })
   338→   338→    }
   339→   339→    if (this.pcs.has(peerId)) return
   340→   340→
   341→   341→    const pc = new RTCPeerConnection({
   342→   342→      iceServers: ICE_SERVERS,
   343→   343→      iceCandidatePoolSize: 4,
   344→   344→      bundlePolicy: 'max-bundle',
   345→   345→      rtcpMuxPolicy: 'require',
   346→   346→    })
   347→   347→    this.pcs.set(peerId, pc)
   348→   348→    this.links.set(peerId, { lastCap: null })
   349→   349→    this.makingOffer.set(peerId, false)
   350→   350→    this.ignoreOffer.set(peerId, false)
   351→   351→
   352→   352→    // Local media into this connection
   353→   353→    const senders: { audio?: RTCRtpSender; video?: RTCRtpSender } = {}
   354→   354→    if (this.localStream) {
   355→   355→      for (const track of this.localStream.getTracks()) {
   356→   356→        try {
   357→   357→          const sender = pc.addTrack(track, this.localStream)
   358→   358→          if (track.kind === 'audio') senders.audio = sender
   359→   359→          else senders.video = sender
   360→   360→        } catch {
   361→   361→          /* ignore */
   362→   362→        }
   363→   363→      }
   364→   364→    } else {
   365→   365→      const at = pc.addTransceiver('audio', { direction: 'recvonly' })
   366→   366→      const vt = pc.addTransceiver('video', { direction: 'recvonly' })
   367→   367→      senders.audio = at.sender
   368→   368→      senders.video = vt.sender
   369→   369→    }
   370→   370→    this.sendersMap.set(peerId, senders)
   371→   371→
   372→   372→    pc.onicecandidate = (e) => {
   373→   373→      if (e.candidate) this.socket?.emit('rtc:ice', { to: peerId, candidate: e.candidate.toJSON() })
   374→   374→    }
   375→   375→
   376→   376→    pc.ontrack = (e) => {
   377→   377→      const stream = e.streams[0]
   378→   378→      if (!stream) return
   379→   379→      putStream(peerId, stream)
   380→   380→      useCallStore.getState().bumpPeerStream(peerId, true)
   381→   381→    }
   382→   382→
   383→   383→    pc.onnegotiationneeded = async () => {
   384→   384→      try {
   385→   385→        this.makingOffer.set(peerId, true)
   386→   386→        const offer = await pc.createOffer()
   387→   387→        await pc.setLocalDescription({ type: 'offer', sdp: mungeOpus(offer.sdp ?? '') })
   388→   388→        const desc = pc.localDescription
   389→   389→        if (desc) this.socket?.emit('rtc:offer', { to: peerId, sdp: desc.sdp })
   390→   390→      } catch {
   391→   391→        /* ignore */
   392→   392→      } finally {
   393→   393→        this.makingOffer.set(peerId, false)
   394→   394→      }
   395→   395→    }
   396→   396→
   397→   397→    pc.onconnectionstatechange = () => {
   398→   398→      if (pc.connectionState === 'failed') {
   399→   399→        try {
   400→   400→          pc.restartIce()
   401→   401→        } catch {
   402→   402→          /* ignore */
   403→   403→        }
   404→   404→      }
   405→   405→    }
   406→   406→
   407→   407→    // Prime encoding params on the fresh senders
   408→   408→    this.capAudioSender(pc)
   409→   409→    const videoSender = senders.video
   410→   410→    if (videoSender) {
   411→   411→      try {
   412→   412→        const params = videoSender.getParameters()
   413→   413→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   414→   414→        params.encodings[0].maxBitrate = this.resolveProfile().maxBitrate
   415→   415→        void videoSender.setParameters(params).catch(() => undefined)
   416→   416→      } catch {
   417→   417→        /* ignore */
   418→   418→      }
   419→   419→    }
   420→   420→  }
   421→   421→
   422→   422→  private dropPeer(peerId: string) {
   423→   423→    const pc = this.pcs.get(peerId)
   424→   424→    if (pc) {
   425→   425→      pc.close()
   426→   426→    }
   427→   427→    this.pcs.delete(peerId)
   428→   428→    this.sendersMap.delete(peerId)
   429→   429→    this.pendingIce.delete(peerId)
   430→   430→    this.links.delete(peerId)
   431→   431→    putStream(peerId, null)
   432→   432→    useCallStore.getState().bumpPeerStream(peerId, false)
   433→   433→    useCallStore.getState().removePeer(peerId)
   434→   434→  }
   435→   435→
   436→   436→  private reconcilePeers(
   437→   437→    peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>,
   438→   438→  ) {
   439→   439→    const store = useCallStore.getState()
   440→   440→    const seen = new Set<string>()
   441→   441→    for (const p of peers) {
   442→   442→      if (p.id === this.myId) continue
   443→   443→      seen.add(p.id)
   444→   444→      this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   445→   445→    }
   446→   446→    for (const id of Object.keys(store.peers)) {
   447→   447→      if (!seen.has(id)) this.dropPeer(id)
   448→   448→    }
   449→   449→    // After a reconnect, old connections may be stale — rebuild them
   450→   450→    for (const [peerId, pc] of this.pcs) {
   451→   451→      if (pc.connectionState === 'failed' || pc.connectionState === 'closed') {
   452→   452→        this.dropPeer(peerId)
   453→   453→        const p = peers.find((x) => x.id === peerId)
   454→   454→        if (p) this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   455→   455→      }
   456→   456→    }

   483→   483→  async toggleCam(): Promise<boolean> {
   484→   484→    const next = !useCallStore.getState().self.camOn
   485→   485→    useCallStore.getState().setSelf({ camOn: next })
   486→   486→    this.socket?.emit('state:update', { camOn: next })
   487→   487→
   488→   488→    if (next && !this.cameraTrack && this.mediaMode !== 'none') {
   489→   489→      // try to acquire a camera mid-call (video track added -> renegotiation fires)
   490→   490→      try {
   491→   491→        const profile = this.resolveProfile()
   492→   492→        const stream = await navigator.mediaDevices.getUserMedia({ video: this.videoConstraints(profile) })
   493→   493→        const track = stream.getVideoTracks()[0]
   494→   494→        this.cameraTrack = track
   495→   495→        track.enabled = true
   496→   496→        if (this.localStream) this.localStream.addTrack(track)
   497→   497→        else this.localStream = stream
   498→   498→        this.publishLocalTrack(track, this.localStream)
   499→   499→        putStream('self', this.localStream)
   500→   500→        await this.setVideoSendersActive(true)
   501→   501→      } catch {
   502→   502→        useCallStore.getState().setSelf({ camOn: false })
   503→   503→        this.socket?.emit('state:update', { camOn: false })
   504→   504→        return false
   505→   505→      }
   506→   506→      return true
   507→   507→    }
   508→   508→
   509→   509→    if (this.cameraTrack) this.cameraTrack.enabled = next
   510→   510→    await this.setVideoSendersActive(next && !this.screenTrack)
   511→   511→    return next
   512→   512→  }

   514→   514→  private publishLocalTrack(track: MediaStreamTrack, owner: MediaStream) {
   515→   515→    for (const [peerId, pc] of this.pcs) {
   516→   516→      const senders = this.sendersMap.get(peerId) ?? {}
   517→   517→      try {
   518→   518→        if (track.kind === 'video' && senders.video && senders.video.track?.kind === 'video') {
   519→   519→          // Swap the outgoing source — no renegotiation, seamless for viewers
   520→   520→          void senders.video.replaceTrack(track).catch(() => {
   521→   521→            try {
   522→   522→              const s = pc.addTrack(track, owner)
   523→   523→              senders.video = s
   524→   524→              this.sendersMap.set(peerId, senders)
   525→   525→            } catch {
   526→   526→              /* ignore */
   527→   527→            }
   528→   528→          })
   529→   529→        } else {
   530→   530→          const s = pc.addTrack(track, owner)
   531→   531→          if (track.kind === 'audio') senders.audio = s
   532→   532→          else senders.video = s
   533→   533→          this.sendersMap.set(peerId, senders)
   534→   534→        }
   535→   535→      } catch {
   536→   536→        /* ignore */
   537→   537→      }
   538→   538→    }
   539→   539→  }

   541→   541→  async toggleShare(): Promise<boolean> {
   542→   542→    const sharing = !useCallStore.getState().self.sharing
   543→   543→    if (sharing) {
   544→   544→      try {
   545→   545→        const display = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: false })
   546→   546→        const track = display.getVideoTracks()[0]
   547→   547→        this.screenTrack = track
   548→   548→        track.addEventListener('ended', () => {
   549→   549→          void this.stopShare()
   550→   550→        })

   552→   552→        // mirror the shared screen into the local preview tile
   553→   553→        putStream('self', display)
   554→   554→        useCallStore.getState().bumpPeerStream('self', true)
   555→   555→        useCallStore.getState().setSelf({ sharing: true })
   556→   556→        this.socket?.emit('state:update', { sharing: true })
   557→   557→        useCallStore.getState().setSpotlight('self')
   558→   558→        return true
   559→   559→      } catch {
   560→   560→        return false
   561→   561→      }
   562→   562→    } else {
   563→   563→      await this.stopShare()
   564→   564→      return false
   565→   565→    }
   566→   566→  }

   568→   568→  private async stopShare() {
   569→   569→    this.screenTrack?.stop()
   570→   570→    this.screenTrack = null
   571→   571→    if (this.cameraTrack && this.cameraTrack.readyState === 'live') {
   572→   572→      // restore the camera as the outgoing video source and the local preview
   573→   573→      this.publishLocalTrack(this.cameraTrack, this.localStream ?? new MediaStream([this.cameraTrack]))
   574→   574→      if (this.localStream) putStream('self', this.localStream)
   575→   575→      await this.setVideoSendersActive(useCallStore.getState().self.camOn)
   576→   576→    } else {
   577→   577→      // no camera available: stop sending video entirely
   578→   578→      await this.setVideoSendersActive(false)
   579→   579→    }
   580→   580→    useCallStore.getState().setSelf({ sharing: false })
   581→   581→    this.socket?.emit('state:update', { sharing: false })
   582→   582→    const spotlight = useCallStore.getState().spotlight
   583→   583→    if (spotlight === 'self') useCallStore.getState().setSpotlight(null)
   584→   584→  }

   637→   637→  private async applyProfile() {
   638→   638→    const profile = this.resolveProfile()
   639→   639→    // 1) Capture resolution / fps (no renegotiation needed)
   640→   640→    if (this.cameraTrack && !this.screenTrack && this.cameraTrack.readyState === 'live') {
   641→   641→      try {
   642→   642→        await this.cameraTrack.applyConstraints(this.videoConstraints(profile))
   643→   643→      } catch {
   644→   644→        /* some cameras ignore constraints */
   645→   645→      }
   646→   646→    }
   647→   647→    // 2) Bitrate caps per sender, respecting per-link adaptive caps
   648→   648→    for (const [peerId, senders] of this.sendersMap) {
   649→   649→      const sender = senders.video
   650→   650→      if (!sender) continue
   651→   651→      const link = this.links.get(peerId)
   652→   652→      const cap = link?.lastCap ?? null
   653→   653→      const target = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   654→   654→      try {
   655→   655→        const params = sender.getParameters()
   656→   656→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   657→   657→        params.encodings[0].maxBitrate = target
   658→   658→        await sender.setParameters(params)
   659→   659→      } catch {
   660→   660→        /* ignore */
   661→   661→      }
   662→   662→    }
   663→   663→  }

   665→   665→  /* ---------------- stats / adaptation ---------------- */
   666→   666→
   667→   667→  private async pollStats() {
   668→   668→    if (this.pcs.size === 0) return
   669→   669→    let overall: NetworkLevel = 3
   670→   670→    let totalKbps = 0
   671→   671→
   672→   672→    for (const [peerId, pc] of this.pcs) {
   673→   673→      if (pc.connectionState !== 'connected') {
   674→   674→        overall = Math.min(overall, 1) as NetworkLevel
   675→   675→        continue
   676→   676→      }
   677→   677→      let rtt: number | null = null
   678→   678→      let loss: number | null = null
   679→   679→      let bytesDelta = 0
   680→   680→      try {
   681→   681→        const stats = await pc.getStats()
   682→   682→        const nextPrev = new Map<string, number>()
   683→   683→        stats.forEach((report) => {
   684→   684→          const r = report as RTCStats & {
   685→   685→            currentRoundTripTime?: number
   686→   686→            fractionLost?: number
   687→   687→            type: string
   688→   688→            state?: string
   689→   689→            nominated?: boolean
   690→   690→            bytesSent?: number
   691→   691→            kind?: string
   692→   692→          }
   693→   693→          if (r.type === 'candidate-pair' && r.state === 'succeeded' && typeof r.currentRoundTripTime === 'number') {
   694→   694→            rtt = rtt === null ? r.currentRoundTripTime : Math.min(rtt, r.currentRoundTripTime)
   695→   695→          }
   696→   696→          if (r.type === 'remote-inbound-rtp' && typeof r.fractionLost === 'number') {
   697→   697→            loss = loss === null ? r.fractionLost : Math.max(loss, r.fractionLost)
   698→   698→          }
   699→   699→          if (r.type === 'outbound-rtp' && r.kind === 'video' && typeof r.bytesSent === 'number') {
   700→   700→            const prevB = this.prevBytes.get(r.id) ?? r.bytesSent
   701→   701→            bytesDelta += Math.max(0, r.bytesSent - prevB)
   702→   702→            nextPrev.set(r.id, r.bytesSent)
   703→   703→          }
   704→   704→        })
   705→   705→        this.prevBytes = nextPrev
   706→   706→      } catch {
   707→   707→        continue
   708→   708→      }
   709→   709→
   710→   710→      const level = classifyLink(rtt, loss)
   711→   711→      const prevLevel = useCallStore.getState().peers[peerId]?.level
   712→   712→      if (prevLevel !== level) useCallStore.getState().patchPeer(peerId, { level })
   713→   713→      overall = Math.min(overall, level) as NetworkLevel
   714→   714→      totalKbps += (bytesDelta * 8) / (STATS_INTERVAL_MS * 1000)
   715→   715→
   716→   716→      // Adaptive per-link video cap
   717→   717→      const link = this.links.get(peerId)
   718→   718→      const cap = LINK_CAPS[level]
   719→   719→      if (link && link.lastCap !== cap) {
   720→   720→        link.lastCap = cap
   721→   721→        const sender = this.sendersMap.get(peerId)?.video
   722→   722→        if (sender) {
   723→   723→          try {
   724→   724→            const params = sender.getParameters()
   725→   725→            if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   726→   726→            const profile = this.resolveProfile()
   727→   727→            params.encodings[0].maxBitrate = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   728→   728→            await sender.setParameters(params)
   729→   729→          } catch {
   730→   730→            /* ignore */
   731→   731→          }
   732→   732→        }
   733→   733→      }
   734→   734→    }
   735→   735→
   736→   736→    const prevOverall = useCallStore.getState().overallLevel
   737→   737→    if (prevOverall !== overall) {
   738→   738→      useCallStore.getState().setOverallLevel(overall)
   739→   739→      useCallStore.getState().patchPeer(this.myId, { level: overall })
   740→   740→    }
   741→   741→    useCallStore.setState({ bitrateKbps: Math.round(totalKbps) })
   742→   742→
   743→   743→    // Auto quality profile with hysteresis
   744→   744→    if (this.preset === 'auto') {
   745→   745→      if (overall <= 1) {
   746→   746→        this.badCount++
   747→   747→        this.goodCount = 0
   748→   748→      } else if (overall === 3) {
   749→   749→        this.goodCount++
   750→   750→        this.badCount = 0
   751→   751→      } else {
   752→   752→        this.badCount = 0
   753→   753→        this.goodCount = 0
   754→   754→      }
   755→   755→      const order: Array<Exclude<QualityPreset, 'auto'>> = ['saver', 'balanced', 'sharp']
   756→   756→      const idx = order.indexOf(this.adaptiveProfile)
   757→   757→      if (this.badCount >= 2 && idx > 0) {
   758→   758→        this.adaptiveProfile = order[idx - 1]
   759→   759→        this.badCount = 0
   760→   760→        this.goodCount = 0
   761→   761→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   762→   762→        void this.applyProfile()
   763→   763→      } else if (this.goodCount >= 8 && idx < order.length - 1) {
   764→   764→        this.adaptiveProfile = order[idx + 1]
   765→   765→        this.badCount = 0
   766→   766→        this.goodCount = 0
   767→   767→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   768→   768→        void this.applyProfile()
   769→   769→      }
   770→   770→    }
   771→   771→  }

   792→   792→  async leave(silent = false) {
   793→   793→    if (this.statsTimer !== null) {
   794→   794→      window.clearInterval(this.statsTimer)
   795→   795→      this.statsTimer = null
   796→   796→    }
   797→   797→    this.socket?.emit('room:leave')
   798→   798→    for (const peerId of Array.from(this.pcs.keys())) this.dropPeer(peerId)
   799→   799→    this.screenTrack?.stop()
   800→   800→    this.localStream?.getTracks().forEach((t) => {
   801→   801→      try {
   802→   802→        t.stop()
   803→   803→      } catch {
   804→   804→        /* ignore */
   805→   805→      }
   806→   806→    })
   807→   807→    const selfStreamTest = (this.localStream as (MediaStream & { __stop?: () => void }) | null)?.__stop
   808→   808→    if (selfStreamTest) selfStreamTest()
   809→   809→    this.localStream = null
   810→   810→    this.cameraTrack = null
   811→   811→    this.micTrack = null
   812→   812→    putStream('self', null)
   813→   813→    this.pcs.clear()
   814→   814→    this.pendingIce.clear()
   815→   815→    this.links.clear()
   816→   816→    this.prevBytes.clear()
   817→   817→    this.socket?.disconnect()
   818→   818→    this.socket = null
   819→   819→    this.myId = ''
   820→   820→    this.started = false
   821→   821→    if (!silent) {
   822→   822→      useCallStore.getState().reset()
   823→   823→    }
   824→   824→  }
   825→   825→}
