    70→export function ChatPanel({ onClose }: { onClose: () => void }) {
    71→  const chat = useCallStore((s) => s.chat)
    72→  const selfId = useCallStore((s) => s.self.id)
    73→  const [draft, setDraft] = useState('')
    74→  const listRef = useRef<HTMLDivElement>(null)
    75→
    76→  useEffect(() => {
    77→    const el = listRef.current
    78→    if (el) el.scrollTop = el.scrollHeight
    79→  }, [chat.length])
    80→
    81→  const send = () => {
    82→    const text = draft.trim()
    83→    if (!text) return
    84→    getEngine().sendChat(text)
    85→    setDraft('')
    86→  }
    87→
    88→  return (
    89→    <Sheet title="Chat" onClose={onClose}>
    90→      <div ref={listRef} className="ios-scroll flex-1 space-y-2.5 overflow-y-auto px-4 py-4">
    91→        {chat.length === 0 && (
    92→          <p className="mt-10 text-center text-[13px] font-medium text-[#a9a9b0]">Messages appear here</p>

    95→          m.system ? (
    96→            <p key={m.id} className="text-center text-[11px] font-semibold uppercase tracking-[0.06em] text-[#a9a9b0]">
    97→              {m.text}
    98→            </p>
    99→          ) : m.from === selfId ? (
   100→            <div key={m.id} className="flex justify-end">
   101→              <div className="max-w-[80%] rounded-[18px] rounded-br-[6px] bg-[#0d0d0f] px-3.5 py-2">
   102→                <p className="whitespace-pre-wrap break-words text-[13px] leading-snug text-white">{m.text}</p>
   103→                <p className="mt-0.5 text-right text-[10px] font-medium text-white/55">
   104→                  {new Date(m.ts).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
   105→                </p>
   106→              </div>
   107→            </div>
   108→          ) : (
   109→            <div key={m.id} className="flex justify-start">
   110→              <div className="max-w-[80%] rounded-[18px] rounded-bl-[6px] bg-[#f5f5f7] px-3.5 py-2">
   111→                <p className="text-[11px] font-bold text-[#7c7c82]">{m.name}</p>

   123→      <footer className="safe-bottom border-t border-black/[0.05] p-3">
   124→        <div className="flex items-center gap-2">
   125→          <input
   126→            value={draft}
   127→            onChange={(e) => setDraft(e.target.value)}
   128→            onKeyDown={(e) => {
   129→              if (e.key === 'Enter' && !e.shiftKey) {

   144→          >
   145→            <ArrowUp className="h-4 w-4" strokeWidth={2.6} />
   146→          </button>

   157→export function ParticipantsPanel({ onClose, roomCode }: { onClose: () => void; roomCode: string }) {
   158→  const self = useCallStore((s) => s.self)
   159→  const peers = useCallStore((s) => s.peers)
   160→  const hostId = useCallStore((s) => s.hostId)
   161→  const locked = useCallStore((s) => s.locked)
   162→  const music = useCallStore((s) => s.music)
   163→  const [copied, setCopied] = useState(false)
   164→  const [removed, setRemoved] = useState<string | null>(null)
   165→
   166→  const isHost = !!self.id && hostId === self.id
   167→
   168→  const inviteLink = typeof window !== 'undefined' ? `${window.location.origin}/?room=${roomCode}` : ''
   169→
   170→  const copy = async () => {
   171→    try {
   172→      await navigator.clipboard.writeText(inviteLink)
   173→    } catch {
   174→      const ta = document.createElement('textarea')
   175→      ta.value = inviteLink
   176→      document.body.appendChild(ta)
   177→      ta.select()
   178→      document.execCommand('copy')
   179→      ta.remove()
   180→    }
   181→    setCopied(true)
   182→    setTimeout(() => setCopied(false), 1600)
   183→  }
   184→
   185→  const everyone = [
   186→    {
   187→      id: self.id || 'self',
   188→      name: self.name || 'You',
   189→      micOn: self.micOn,
   190→      camOn: self.camOn,
   191→      sharing: self.sharing,
   192→      isSelf: true,
   193→    },
   194→    ...Object.values(peers).map((p) => ({
   195→      id: p.id,
   196→      name: p.name,
   197→      micOn: p.micOn,
   198→      camOn: p.camOn,
   199→      sharing: p.sharing,
   200→      isSelf: false,
   201→    })),
   202→  ]
   203→
   204→  const remove = (id: string, name: string) => {
   205→    getEngine().hostRemove(id)
   206→    setRemoved(name)
   207→    setTimeout(() => setRemoved(null), 2200)
   208→  }
   209→
   210→  return (
   211→    <Sheet title={`People · ${everyone.length}`} onClose={onClose}>
   212→      <div className="border-b border-black/[0.05] p-3">
   213→        <button onClick={copy} className="ios-pill-soft h-10 w-full text-[13px]">
   214→          {copied ? <Check className="h-4 w-4 text-[#30d158]" strokeWidth={2.6} /> : <Link2 className="h-4 w-4 text-[#7c7c82]" strokeWidth={2.2} />}
   215→          {copied ? 'Link copied' : 'Copy invite link'}
   216→        </button>
   217→        <p className="mt-2 text-center font-mono text-[11px] font-medium tracking-wide text-[#a9a9b0]">
   218→          {locked ? 'Locked · nobody new can join' : roomCode}
   219→        </p>
   220→      </div>
   221→
   222→      {removed && (
   223→        <p className="px-4 pt-3 text-[12px] font-semibold text-[#ff3b30]" role="status">
   224→          {removed} was removed from the call
   225→        </p>
   226→      )}
   227→
   228→      <ul className="ios-scroll flex-1 overflow-y-auto p-2">
   229→        {everyone.map((p) => (
   230→          <li key={p.id} className="flex items-center gap-3 rounded-2xl px-3 py-2.5">
   231→            <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-[#e8e8ed] text-[12px] font-bold text-[#7c7c82]">
   232→              {p.name.slice(0, 2).toUpperCase()}
   233→            </div>
   234→            <div className="min-w-0 flex-1">
   235→              <p className="flex items-center gap-1.5 truncate text-[15px] font-semibold text-[#0d0d0f]">
   236→                {p.name}
   237→                {p.isSelf ? ' (You)' : ''}
   238→                {hostId === p.id && <Crown className="h-3.5 w-3.5 shrink-0 text-[#f5a623]" aria-label="Host" strokeWidth={2.4} />}
   239→              </p>
   240→              <p className="text-[12px] font-medium text-[#a9a9b0]">
   241→                {music.peerId === p.id
   242→                  ? 'Playing music for everyone'
   243→                  : p.sharing
   244→                    ? 'Sharing screen'
   245→                    : p.camOn
   246→                      ? 'Camera on'
   247→                      : 'Camera off'}
   248→              </p>
   249→            </div>
   250→            <div className="flex items-center gap-1.5 text-[#7c7c82]">
   251→              {!p.micOn && <MicOff className="h-4 w-4 text-[#ff3b30]" aria-label="Muted" strokeWidth={2.4} />}
   252→              {!p.camOn && !p.sharing && <CameraOff className="h-4 w-4" aria-label="Camera off" strokeWidth={2.4} />}
   253→              {p.sharing && <MonitorUp className="h-4 w-4 text-[#30d158]" aria-label="Sharing" strokeWidth={2.4} />}
   254→            </div>
   255→            {isHost && !p.isSelf && (
   256→              <div className="flex shrink-0 items-center gap-1">
   257→                {p.micOn && (
   258→                  <button
   259→                    onClick={() => getEngine().hostMute(p.id)}
   260→                    aria-label={`Mute ${p.name}`}
   261→                    title={`Mute ${p.name}`}
   262→                    className="press flex h-8 w-8 items-center justify-center rounded-full bg-[#f5f5f7] text-[#3c3c43] hover:bg-[#ebebf0] active:scale-90"
   263→                  >
   264→                    <MicOff className="h-3.5 w-3.5" strokeWidth={2.4} />
   265→                  </button>
   266→                )}
   267→                <button
   268→                  onClick={() => remove(p.id, p.name)}
   269→                  aria-label={`Remove ${p.name}`}
   270→                  title={`Remove ${p.name}`}
   271→                  className="press flex h-8 w-8 items-center justify-center rounded-full bg-[#ff3b30]/10 text-[#ff3b30] hover:bg-[#ff3b30]/15 active:scale-90"
   272→                >
   273→                  <UserMinus className="h-3.5 w-3.5" strokeWidth={2.4} />
   274→                </button>
   275→              </div>
   276→            )}
   277→          </li>
   278→        ))}
   279→      </ul>
   280→
   281→      {isHost && (
   282→        <div className="border-t border-black/[0.05] p-3">
   283→          <div className="flex gap-2">
   284→            <button
   285→              onClick={() => getEngine().hostMuteAll()}
   286→              className="ios-pill-soft h-10 flex-1 text-[13px]"
   287→            >
   288→              <MicOff className="h-4 w-4 text-[#7c7c82]" strokeWidth={2.2} />
   289→              Mute everyone
   290→            </button>
   291→            <button
   292→              onClick={() => getEngine().hostLock(!locked)}
   293→              className="ios-pill-soft h-10 flex-1 text-[13px]"
   294→            >
   295→              {locked ? <LockOpen className="h-4 w-4 text-[#30d158]" strokeWidth={2.2} /> : <Lock className="h-4 w-4 text-[#7c7c82]" strokeWidth={2.2} />}
   296→              {locked ? 'Unlock room' : 'Lock room'}
   297→            </button>
   298→          </div>
   299→          <p className="mt-2 text-center text-[11px] font-medium text-[#a9a9b0]">
   300→            You are the host — you can mute or remove anyone
   301→          </p>
   302→        </div>
   303→      )}
   304→
   305→      <footer className="border-t border-black/[0.05] px-4 py-3 text-center text-[12px] font-medium text-[#a9a9b0]">
   306→        <Users className="mr-1 inline h-3 w-3" /> Peer-to-peer · up to 8 people
   307→      </footer>
   308→    </Sheet>
   309→  )

   316→const fmtTime = (t: number) => {
   317→  const s = Math.max(0, Math.floor(t))
   318→  const m = Math.floor(s / 60)
   319→  return `${m}:${String(s % 60).padStart(2, '0')}`

   322→export function MusicPanel({ onClose }: { onClose: () => void }) {
   323→  const music = useCallStore((s) => s.music)
   324→  const selfId = useCallStore((s) => s.self.id)
   325→  const peers = useCallStore((s) => s.peers)
   326→  const fileRef = useRef<HTMLInputElement>(null)
   327→  const [position, setPosition] = useState(0)
   328→  const [monitorVol, setMonitorVol] = useState(1)
   329→  const [remoteVol, setRemoteVol] = useState(1)
   330→  const [starting, setStarting] = useState(false)
   331→  const testMode = typeof window !== 'undefined' && window.location.search.includes('test=1')
   332→
   333→  const isSelf = music.peerId !== null && music.peerId === selfId
   334→  const sharerName =
   335→    music.peerId === null ? '' : music.peerId === selfId ? 'You' : peers[music.peerId]?.name ?? 'Someone'
   336→
   337→  // Live position for the sharer — read straight from the active source
   338→  useEffect(() => {
   339→    if (!isSelf) return
   340→    const t = window.setInterval(() => {
   341→      const h = getEngine().activeMusic
   342→      if (h) setPosition(h.getPosition())
   343→    }, 500)
   344→    return () => window.clearInterval(t)
   345→  }, [isSelf])
   346→
   347→  const beginShare = (handle: MusicHandle, origin: 'file' | 'test') => {
   348→    handle.onMeta(() => getEngine().broadcastMusicMeta())
   349→    getEngine().startMusicShare(handle, origin)
   350→    void handle.play().then(() => getEngine().broadcastMusicMeta())
   351→  }
   352→
   353→  const onFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   354→    const file = e.target.files?.[0]
   355→    if (!file) return
   356→    try {
   357→      const handle = createFileMusicHandle(file)
   358→      beginShare(handle, 'file')
   359→    } catch {
   360→      /* file unreadable */
   361→    }
   362→    e.target.value = ''
   363→  }
   364→
   365→  const startTest = () => {
   366→    setStarting(true)
   367→    try {
   368→      beginShare(createTestMusicHandle(), 'test')
   369→    } finally {
   370→      setStarting(false)
   371→    }
   372→  }
   373→
   374→  const stop = () => {
   375→    // One call tears down senders, store state and the element-backed handle
   376→    getEngine().stopMusicShare()
   377→  }
   378→
   379→  const togglePlay = () => {
   380→    const h = getEngine().activeMusic
   381→    if (!h) return
   382→    if (h.isPlaying()) h.pause?.()
   383→    else void h.play?.()
   384→    getEngine().broadcastMusicMeta()
   385→  }
   386→
   387→  const active = getEngine().activeMusic
   388→  const duration = active?.duration || 0
   389→  const canControl = !!active && (!!active.play || originKind(active) !== 'tab')
   390→
   391→  return (
   392→    <Sheet title="Listen together" onClose={onClose}>
   393→      <div className="ios-scroll flex-1 overflow-y-auto p-4">
   394→        {music.peerId === null ? (
   395→          <>
   396→            <div className="mb-4 flex items-start gap-3 rounded-[18px] bg-[#f5f5f7] px-4 py-3.5">
   397→              <Music2 className="mt-0.5 h-5 w-5 shrink-0 text-[#0d0d0f]" strokeWidth={2.2} />
   398→              <p className="text-[13px] leading-relaxed text-[#3c3c43]">
   399→                Play music from this device and everyone in the call hears the same thing, in sync.
   400→                It rides its own stereo audio channel — nobody&apos;s mic or video is touched.
   401→              </p>
   402→            </div>
   403→
   404→            <input ref={fileRef} type="file" accept="audio/*" onChange={onFile} className="hidden" />
   405→            <button onClick={() => fileRef.current?.click()} className="ios-pill h-12 w-full">
   406→              <FileAudio className="h-4 w-4" strokeWidth={2.2} />
   407→              Choose a music file
   408→            </button>
   409→            <p className="mt-2.5 text-center text-[12px] leading-relaxed text-[#a9a9b0]">
   410→              MP3, M4A, WAV, OGG — stays on your device.
   411→              <br />
   412→              Uses about 96 kbps per listener. Silence costs nothing.
   413→            </p>
   414→
   415→            {testMode && (
   416→              <button onClick={startTest} disabled={starting} className="ios-pill-soft mt-4 h-11 w-full text-[13px]">
   417→                <Music2 className="h-4 w-4 text-[#7c7c82]" strokeWidth={2.2} />
   418→                {starting ? 'Starting…' : 'Play demo tune (test mode)'}
   419→              </button>
   420→            )}
   421→          </>
   422→        ) : isSelf ? (
   423→          <>
   424→            <div className="rounded-[18px] bg-[#0d0d0f] px-4 py-4 text-white">
   425→              <div className="flex items-center gap-3">
   426→                <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-full bg-white/10">
   427→                  <Music2 className="h-5 w-5 text-[#30d158]" strokeWidth={2.2} />
   428→                </div>
   429→                <div className="min-w-0 flex-1">
   430→                  <p className="truncate text-[15px] font-bold">{music.title}</p>
   431→                  <p className="text-[12px] font-medium text-white/60">
   432→                    {music.playing ? 'Playing for everyone' : 'Paused'}
   433→                  </p>
   434→                </div>
   435→                <span className="flex items-end gap-[2.5px]" aria-hidden>
   436→                  {[0, 1, 2, 3].map((i) => (
   437→                    <span
   438→                      key={i}
   439→                      className="eq-bar eq-bar-dark"
   440→                      style={{ animationDelay: `${i * 140}ms`, animationPlayState: music.playing ? 'running' : 'paused' }}
   441→                    />
   442→                  ))}
   443→                </span>
   444→              </div>
   445→
   446→              {duration > 0 && canControl && (
   447→                <div className="mt-4">
   448→                  <input
   449→                    type="range"
   450→                    min={0}
   451→                    max={Math.floor(duration)}
   452→                    value={Math.min(position, duration)}
   453→                    onChange={(e) => {
   454→                      const t = Number(e.target.value)
   455→                      setPosition(t)
   456→                      active?.seek?.(t)
   457→                    }}
   458→                    onMouseUp={() => getEngine().broadcastMusicMeta()}
   459→                    onTouchEnd={() => getEngine().broadcastMusicMeta()}
   460→                    aria-label="Seek"
   461→                    className="ios-range"
   462→                  />
   463→                  <div className="mt-1 flex justify-between font-mono text-[10px] tabular-nums text-white/55">
   464→                    <span>{fmtTime(position)}</span>
   465→                    <span>{fmtTime(duration)}</span>
   466→                  </div>
   467→                </div>
   468→              )}
   469→
   470→              <div className="mt-4 flex items-center justify-between">
   471→                {canControl ? (
   472→                  <button
   473→                    onClick={togglePlay}
   474→                    aria-label={music.playing ? 'Pause music' : 'Play music'}
   475→                    className="press flex h-11 w-11 items-center justify-center rounded-full bg-white text-[#0d0d0f] active:scale-90"
   476→                  >
   477→                    {music.playing ? <Pause className="h-5 w-5" strokeWidth={2.4} /> : <Play className="h-5 w-5 translate-x-[1px]" strokeWidth={2.4} />}
   478→                  </button>
   479→                ) : (
   480→                  <span className="text-[12px] font-semibold text-white/60">Sharing this tab&apos;s sound</span>
   481→                )}
   482→                <button
   483→                  onClick={stop}
   484→                  className="flex h-11 items-center gap-2 rounded-full bg-[#ff3b30] px-5 text-[14px] font-bold text-white transition-all duration-[130ms] ease-[cubic-bezier(0.34,1.35,0.64,1)] hover:bg-[#eb3329] active:scale-95"
   485→                >
   486→                  <Square className="h-3.5 w-3.5" strokeWidth={2.6} />
   487→                  Stop
   488→                </button>
   489→              </div>
   490→
   491→              {canControl && (
   492→                <div className="mt-4 flex items-center gap-2.5 border-t border-white/10 pt-3.5">
   493→                  <Volume2 className="h-4 w-4 shrink-0 text-white/60" strokeWidth={2.2} />
   494→                  <input
   495→                    type="range"
   496→                    min={0}
   497→                    max={100}
   498→                    value={Math.round(monitorVol * 100)}
   499→                    onChange={(e) => {
   500→                      const v = Number(e.target.value) / 100
   501→                      setMonitorVol(v)
   502→                      active?.setMonitorVolume?.(v)
   503→                    }}
   504→                    aria-label="Volume for you"
