    14→/**
    15→ * Pico call engine — peer-to-peer WebRTC mesh.
    16→ *
    17→ * Design goals:
    18→ *  - Media never touches a server: every audio/video packet travels directly
    19→ *    between participants (mesh topology). The signaling socket only carries
    20→ *    SDP/ICE/chat text.
    21→ *  - Renegotiation-free quality control: resolution/fps changes ride on
    22→ *    applyConstraints(), bitrate changes on setParameters(), screen share on
    23→ *    replaceTrack(). No expensive offer/answer churn.
    24→ *  - Opus DTX + 24 kbps cap: silence costs (almost) nothing on the wire.
    25→ *  - Per-link adaptive bitrate: getStats() every 3s classifies each
    26→ *    connection (RTT + packet loss) and caps outgoing video per link, so one
    27→ *    weak participant never drags the whole call down.
    28→ *  - Perfect-negotiation pattern (polite/impolite) for glare-free offers.
    29→ */
    30→
    31→const ICE_SERVERS: RTCIceServer[] = [
    32→  { urls: ['stun:stun.l.google.com:19302', 'stun:stun1.l.google.com:19302'] },
    33→  { urls: ['stun:stun.cloudflare.com:3478'] },
    34→]
    35→
    36→const AUDIO_SENDER_BITRATE = 24_000
    37→const STATS_INTERVAL_MS = 3000
    38→
    39→export type MediaMode = 'full' | 'audio' | 'none'

    58→/** Add Opus DTX + low bitrate cap to SDP — near-zero cost for silence. */
    59→function mungeOpus(sdp: string): string {
    60→  const lines = sdp.split('\r\n')
    61→  const opusPTs = new Set<string>()
    62→  for (const line of lines) {
    63→    const m = line.match(/^a=rtpmap:(\d+) opus\/48000/i)
    64→    if (m) opusPTs.add(m[1])
    65→  }
    66→  const out = lines.map((line) => {
    67→    const m = line.match(/^a=fmtp:(\d+)(.*)$/)
    68→    if (m && opusPTs.has(m[1])) {
    69→      let params = m[2] ?? ''
    70→      if (!/usedtx/i.test(params)) params += ';usedtx=1'
    71→      if (!/maxaveragebitrate/i.test(params)) params += ';maxaveragebitrate=24000'
    72→      return `a=fmtp:${m[1]}${params}`
    73→    }
    74→    return line
    75→  })
    76→  return out.join('\r\n')
    77→}
    78→
    79→function classifyLink(rttSec: number | null, lossFrac: number | null): NetworkLevel {
    80→  const rtt = rttSec ?? 0.4
    81→  const loss = lossFrac ?? 0.05
    82→  if (rtt > 0.5 || loss > 0.08) return 0
    83→  if (rtt > 0.3 || loss > 0.05) return 1
    84→  if (rtt > 0.15 || loss > 0.02) return 2
    85→  return 3
    86→}

    92→export class CallEngine {
    93→  private socket: Socket | null = null
    94→  private roomId = ''
    95→  private myId = ''
    96→  private localStream: MediaStream | null = null
    97→  private cameraTrack: MediaStreamTrack | null = null
    98→  private micTrack: MediaStreamTrack | null = null
    99→  private screenTrack: MediaStreamTrack | null = null
   100→  private pcs = new Map<string, RTCPeerConnection>()
   101→  private sendersMap = new Map<string, { audio?: RTCRtpSender; video?: RTCRtpSender }>()
   102→  private pendingIce = new Map<string, RTCIceCandidateInit[]>()
   103→  private links = new Map<string, LinkInfo>()
   104→  private statsTimer: number | null = null
   105→  private makingOffer = new Map<string, boolean>()
   106→  private ignoreOffer = new Map<string, boolean>()
   107→  private preset: QualityPreset = 'balanced'
   108→  private adaptiveProfile: Exclude<QualityPreset, 'auto'> = 'balanced'
   109→  private badCount = 0
   110→  private goodCount = 0
   111→  private started = false
   112→  private testMode = false
   113→  private mediaMode: MediaMode = 'full'
   114→  private prevBytes = new Map<string, number>()
   115→
   116→  async start(opts: StartOptions): Promise<{ mode: MediaMode }> {
   117→    if (this.started) await this.leave(true)
   118→    this.started = true
   119→    this.roomId = opts.roomId
   120→    this.preset = opts.preset
   121→    this.adaptiveProfile = opts.preset === 'auto' ? 'balanced' : opts.preset
   122→    this.testMode = !!opts.testMode
   123→
   124→    const store = useCallStore.getState()
   125→    store.setSelf({ name: opts.name })
   126→    store.setPreset(opts.preset)
   127→    store.setEffectiveLabel(this.effectiveLabel())
   128→
   129→    // ---- Signaling ----
   130→    const socket = io('/?XTransformPort=3003', {
   131→      path: '/',
   132→      transports: ['websocket', 'polling'],
   133→      reconnection: true,
   134→      reconnectionAttempts: Infinity,
   135→      reconnectionDelay: 1000,
   136→      reconnectionDelayMax: 4000,
   137→      timeout: 8000,
   138→    })
   139→    this.socket = socket
   140→    // Dev/debug handle (harmless in prod, invaluable for live diagnostics)
   141→    if (typeof window !== 'undefined') {
   142→      ;(window as unknown as { __pico?: unknown }).__pico = { engine: this, socket }
   143→    }
   144→    socket.on('connect', () => {
   145→      useCallStore.getState().setSignalingConnected(true)
   146→      if (this.myId) this.rejoin()
   147→    })
   148→    socket.on('disconnect', () => useCallStore.getState().setSignalingConnected(false))
   149→    socket.on('connect_error', () => useCallStore.getState().setSignalingConnected(false))
   150→
   151→    // ---- Local media ----
   152→    this.localStream = opts.initialStream ?? null
   153→    if (this.localStream) {
   154→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   155→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   156→      this.mediaMode = this.cameraTrack ? 'full' : this.micTrack ? 'audio' : 'none'
   157→      // Register the handed-over preview stream so tiles can attach to it
   158→      putStream('self', this.localStream)
   159→    }
   160→    if (!this.localStream || (!this.cameraTrack && !this.micTrack)) {
   161→      this.mediaMode = await this.acquireMedia()
   162→    }
   163→    this.applyInitialTrackState()
   164→
   165→    // ---- Room ----
   166→    await new Promise<void>((resolve) => {
   167→      socket.emit(
   168→        'room:join',
   169→        { roomId: this.roomId, name: opts.name },
   170→        (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; reason?: string }) => {
   171→          if (!res?.ok) {
   172→            useCallStore.getState().addChat({
   173→              id: 'sys-' + Date.now(),
   174→              from: 'system',
   175→              name: 'System',
   176→              text: res?.reason ?? 'Could not join the room',
   177→              ts: Date.now(),
   178→              system: true,
   179→            })
   180→          }
   181→          this.myId = res?.self?.id ?? socket.id!
   182→          useCallStore.getState().setSelf({ id: this.myId })
   183→          if (res?.peers) this.reconcilePeers(res.peers)
   184→          resolve()
   185→        },
   186→      )
   187→    })
   188→
   189→    // ---- Events ----
   190→    socket.on('peer:joined', ({ peer }: { peer: { id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean } }) => {
   191→      this.ensurePeer(peer.id, peer.name, peer.micOn, peer.camOn, peer.sharing)
   192→      useCallStore.getState().addChat({
   193→        id: 'sys-' + Math.random(),
   194→        from: 'system',
   195→        name: 'System',
   196→        text: `${peer.name} joined`,
   197→        ts: Date.now(),
   198→        system: true,
   199→      })
   200→    })
   201→    socket.on('peer:left', ({ peerId }: { peerId: string }) => this.dropPeer(peerId))
   202→    socket.on('room:peers', ({ peers }: { peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   203→      // Only meaningful after a reconnect — reconcile full presence
   204→      if (this.myId) this.reconcilePeers(peers)
   205→    })
   206→    socket.on('peer:state', ({ peerId, micOn, camOn, sharing }: { peerId: string; micOn: boolean; camOn: boolean; sharing: boolean }) => {
   207→      useCallStore.getState().patchPeer(peerId, { micOn, camOn, sharing })
   208→    })
   209→    socket.on('chat:message', (msg: { id: string; from: string; name: string; text: string; ts: number }) => {
   210→      useCallStore.getState().addChat(msg)
   211→    })
   212→
   213→    socket.on('rtc:offer', async ({ from, sdp }: { from: string; sdp: string }) => {
   214→      try {
   215→        const pc = this.pcs.get(from)
   216→        if (!pc) return
   217→        const polite = this.myId < from
   218→        const offerCollision = this.makingOffer.get(from) || pc.signalingState !== 'stable'
   219→        this.ignoreOffer.set(from, !polite && offerCollision)
   220→        if (this.ignoreOffer.get(from)) return
   221→        await pc.setRemoteDescription({ type: 'offer', sdp: mungeOpus(sdp) })
   222→        await this.flushIce(from)
   223→        const answer = await pc.createAnswer()
   224→        answer.sdp = mungeOpus(answer.sdp ?? '')
   225→        await pc.setLocalDescription(answer)
   226→        this.socket?.emit('rtc:answer', { to: from, sdp: pc.localDescription!.sdp })
   227→      } catch {
   228→        /* negotiation raced — next round will fix it */
   229→      }
   230→    })
   231→
   232→    socket.on('rtc:answer', async ({ from, sdp }: { from: string; sdp: string }) => {
   233→      try {
   234→        const pc = this.pcs.get(from)
   235→        if (!pc || pc.signalingState !== 'have-local-offer') return
   236→        await pc.setRemoteDescription({ type: 'answer', sdp: mungeOpus(sdp) })
   237→        await this.flushIce(from)
   238→      } catch {
   239→        /* ignore */
   240→      }
   241→    })
   242→
   243→    socket.on('rtc:ice', ({ from, candidate }: { from: string; candidate: RTCIceCandidateInit }) => {
   244→      const pc = this.pcs.get(from)
   245→      if (!pc) return
   246→      if (pc.remoteDescription) {
   247→        pc.addIceCandidate(candidate).catch(() => {
   248→          if (!this.ignoreOffer.get(from)) {
   249→            this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   250→          }
   251→        })
   252→      } else {
   253→        this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   254→      }
   255→    })
   256→
   257→    // ---- Stats loop ----
   258→    this.statsTimer = window.setInterval(() => void this.pollStats(), STATS_INTERVAL_MS)
   259→
   260→    return { mode: this.mediaMode }
   261→  }
   262→
   263→  /* ---------------- media ---------------- */
   264→
   265→  private async acquireMedia(): Promise<MediaMode> {
   266→    if (this.testMode) {
   267→      this.localStream = await createTestStream()
   268→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   269→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   270→      putStream('self', this.localStream)
   271→      return 'full'
   272→    }
   273→    const audio: MediaTrackConstraints = {
   274→      echoCancellation: true,
   275→      noiseSuppression: true,
   276→      autoGainControl: true,
   277→      channelCount: 1,
   278→    }
   279→    const video = this.videoConstraints(this.resolveProfile())
   280→    // Full quality first
   281→    try {
   282→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio, video })
   283→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   284→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   285→      this.attachLocalStream()
   286→      return 'full'
   287→    } catch {
   288→      /* fall through */
   289→    }
   290→    // Audio only
   291→    try {
   292→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio })
   293→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   294→      this.cameraTrack = null
   295→      this.attachLocalStream()
   296→      return 'audio'
   297→    } catch {
   298→      /* fall through */
   299→    }
   300→    // Listener mode
   301→    this.localStream = null
   302→    return 'none'
   303→  }
   304→
   305→  private videoConstraints(profile: QualityProfile): MediaTrackConstraints {
   306→    return {
   307→      width: { ideal: profile.width },
   308→      height: { ideal: profile.height },
   309→      frameRate: { ideal: profile.frameRate, max: profile.frameRate },
   310→      facingMode: 'user',
   311→    }
   312→  }
   313→
   314→  private attachLocalStream() {
   315→    if (!this.localStream) return
   316→    putStream('self', this.localStream)
   317→  }
   318→
   319→  private applyInitialTrackState() {
   320→    const s = useCallStore.getState().self
   321→    if (this.cameraTrack) this.cameraTrack.enabled = s.camOn
   322→    if (this.micTrack) this.micTrack.enabled = s.micOn
   323→    if (!s.camOn) void this.setVideoSendersActive(false)
   324→  }
   325→
   326→  /* ---------------- peers ---------------- */
   327→
   328→  private ensurePeer(
   329→    peerId: string,
   330→    name: string,
   331→    micOn = true,
   332→    camOn = true,
   333→    sharing = false,
   334→  ) {
   335→    const store = useCallStore.getState()
   336→    if (!store.peers[peerId]) {
   337→      store.upsertPeer({ id: peerId, name, micOn, camOn, sharing, level: 3, hasStream: false, streamVersion: 0 })
   338→    }
   339→    if (this.pcs.has(peerId)) return
   340→
   341→    const pc = new RTCPeerConnection({
   342→      iceServers: ICE_SERVERS,
   343→      iceCandidatePoolSize: 4,
   344→      bundlePolicy: 'max-bundle',
   345→      rtcpMuxPolicy: 'require',
   346→    })
   347→    this.pcs.set(peerId, pc)
   348→    this.links.set(peerId, { lastCap: null })
   349→    this.makingOffer.set(peerId, false)
   350→    this.ignoreOffer.set(peerId, false)
   351→
   352→    // Local media into this connection
   353→    const senders: { audio?: RTCRtpSender; video?: RTCRtpSender } = {}
   354→    if (this.localStream) {
   355→      for (const track of this.localStream.getTracks()) {
   356→        try {
   357→          const sender = pc.addTrack(track, this.localStream)
   358→          if (track.kind === 'audio') senders.audio = sender
   359→          else senders.video = sender
   360→        } catch {
   361→          /* ignore */
   362→        }
   363→      }
   364→    } else {
   365→      const at = pc.addTransceiver('audio', { direction: 'recvonly' })
   366→      const vt = pc.addTransceiver('video', { direction: 'recvonly' })
   367→      senders.audio = at.sender
   368→      senders.video = vt.sender
   369→    }
   370→    this.sendersMap.set(peerId, senders)
   371→
   372→    pc.onicecandidate = (e) => {
   373→      if (e.candidate) this.socket?.emit('rtc:ice', { to: peerId, candidate: e.candidate.toJSON() })
   374→    }
   375→
   376→    pc.ontrack = (e) => {
   377→      const stream = e.streams[0]
   378→      if (!stream) return
   379→      putStream(peerId, stream)
   380→      useCallStore.getState().bumpPeerStream(peerId, true)
   381→    }
   382→
   383→    pc.onnegotiationneeded = async () => {
   384→      try {
   385→        this.makingOffer.set(peerId, true)
   386→        const offer = await pc.createOffer()
   387→        await pc.setLocalDescription({ type: 'offer', sdp: mungeOpus(offer.sdp ?? '') })
   388→        const desc = pc.localDescription
   389→        if (desc) this.socket?.emit('rtc:offer', { to: peerId, sdp: desc.sdp })
   390→      } catch {
   391→        /* ignore */
   392→      } finally {
   393→        this.makingOffer.set(peerId, false)
   394→      }
   395→    }
   396→
   397→    pc.onconnectionstatechange = () => {
   398→      if (pc.connectionState === 'failed') {
   399→        try {
   400→          pc.restartIce()
   401→        } catch {
   402→          /* ignore */
   403→        }
   404→      }
   405→    }
   406→
   407→    // Prime encoding params on the fresh senders
   408→    this.capAudioSender(pc)
   409→    const videoSender = senders.video
   410→    if (videoSender) {
   411→      try {
   412→        const params = videoSender.getParameters()
   413→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   414→        params.encodings[0].maxBitrate = this.resolveProfile().maxBitrate
   415→        void videoSender.setParameters(params).catch(() => undefined)
   416→      } catch {
   417→        /* ignore */
   418→      }
   419→    }
   420→  }
   421→
   422→  private dropPeer(peerId: string) {
   423→    const pc = this.pcs.get(peerId)
   424→    if (pc) {
   425→      pc.close()
   426→    }
   427→    this.pcs.delete(peerId)
   428→    this.sendersMap.delete(peerId)
   429→    this.pendingIce.delete(peerId)
   430→    this.links.delete(peerId)
   431→    putStream(peerId, null)
   432→    useCallStore.getState().bumpPeerStream(peerId, false)
   433→    useCallStore.getState().removePeer(peerId)
   434→  }
   435→
   436→  private reconcilePeers(
   437→    peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>,
   438→  ) {
   439→    const store = useCallStore.getState()
   440→    const seen = new Set<string>()
   441→    for (const p of peers) {
   442→      if (p.id === this.myId) continue
   443→      seen.add(p.id)
   444→      this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   445→    }
   446→    for (const id of Object.keys(store.peers)) {
   447→      if (!seen.has(id)) this.dropPeer(id)
   448→    }
   449→    // After a reconnect, old connections may be stale — rebuild them
   450→    for (const [peerId, pc] of this.pcs) {
   451→      if (pc.connectionState === 'failed' || pc.connectionState === 'closed') {
   452→        this.dropPeer(peerId)
   453→        const p = peers.find((x) => x.id === peerId)
   454→        if (p) this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   455→      }
   456→    }
   457→  }
   458→
   459→  private async flushIce(peerId: string) {
   460→    const pc = this.pcs.get(peerId)
   461→    const queued = this.pendingIce.get(peerId) ?? []
   462→    if (!pc || queued.length === 0) return
   463→    this.pendingIce.set(peerId, [])
   464→    for (const c of queued) {
   465→      try {
   466→        await pc.addIceCandidate(c)
   467→      } catch {
   468→        /* ignore */
   469→      }
   470→    }
   471→  }

   473→  /* ---------------- controls ---------------- */
   474→
   475→  toggleMic(): boolean {
   476→    const next = !useCallStore.getState().self.micOn
   477→    if (this.micTrack) this.micTrack.enabled = next
   478→    useCallStore.getState().setSelf({ micOn: next })
   479→    this.socket?.emit('state:update', { micOn: next })
   480→    return next
   481→  }
   482→
   483→  async toggleCam(): Promise<boolean> {
   484→    const next = !useCallStore.getState().self.camOn
   485→    useCallStore.getState().setSelf({ camOn: next })
   486→    this.socket?.emit('state:update', { camOn: next })
   487→
   488→    if (next && !this.cameraTrack && this.mediaMode !== 'none') {
   489→      // try to acquire a camera mid-call (video track added -> renegotiation fires)
   490→      try {
   491→        const profile = this.resolveProfile()
   492→        const stream = await navigator.mediaDevices.getUserMedia({ video: this.videoConstraints(profile) })
   493→        const track = stream.getVideoTracks()[0]
   494→        this.cameraTrack = track
   495→        track.enabled = true
   496→        if (this.localStream) this.localStream.addTrack(track)
   497→        else this.localStream = stream
   498→        this.publishLocalTrack(track, this.localStream)
   499→        putStream('self', this.localStream)
   500→        await this.setVideoSendersActive(true)
   501→      } catch {
   502→        useCallStore.getState().setSelf({ camOn: false })
   503→        this.socket?.emit('state:update', { camOn: false })
   504→        return false
   505→      }
   506→      return true
   507→    }
   508→
   509→    if (this.cameraTrack) this.cameraTrack.enabled = next
   510→    await this.setVideoSendersActive(next && !this.screenTrack)
   511→    return next
   512→  }
   513→
   514→  private publishLocalTrack(track: MediaStreamTrack, owner: MediaStream) {
   515→    for (const [peerId, pc] of this.pcs) {
   516→      const senders = this.sendersMap.get(peerId) ?? {}
   517→      try {
   518→        if (track.kind === 'video' && senders.video && senders.video.track?.kind === 'video') {
   519→          // Swap the outgoing source — no renegotiation, seamless for viewers
   520→          void senders.video.replaceTrack(track).catch(() => {
   521→            try {
   522→              const s = pc.addTrack(track, owner)
   523→              senders.video = s
   524→              this.sendersMap.set(peerId, senders)
   525→            } catch {
   526→              /* ignore */
   527→            }
   528→          })
   529→        } else {
   530→          const s = pc.addTrack(track, owner)
   531→          if (track.kind === 'audio') senders.audio = s
   532→          else senders.video = s
   533→          this.sendersMap.set(peerId, senders)
   534→        }
   535→      } catch {
   536→        /* ignore */
   537→      }
   538→    }
   539→  }
   540→
   541→  async toggleShare(): Promise<boolean> {
   542→    const sharing = !useCallStore.getState().self.sharing
   543→    if (sharing) {
   544→      try {
   545→        const display = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: false })
   546→        const track = display.getVideoTracks()[0]
   547→        this.screenTrack = track
   548→        track.addEventListener('ended', () => {
   549→          void this.stopShare()
   550→        })
   551→        this.publishLocalTrack(track, display)
   552→        // mirror the shared screen into the local preview tile
   553→        putStream('self', display)
   554→        useCallStore.getState().bumpPeerStream('self', true)
   555→        useCallStore.getState().setSelf({ sharing: true })
   556→        this.socket?.emit('state:update', { sharing: true })
   557→        useCallStore.getState().setSpotlight('self')
   558→        return true
   559→      } catch {
   560→        return false
   561→      }
   562→    } else {
   563→      await this.stopShare()
   564→      return false
   565→    }
   566→  }
   567→
   568→  private async stopShare() {
   569→    this.screenTrack?.stop()
   570→    this.screenTrack = null
   571→    if (this.cameraTrack && this.cameraTrack.readyState === 'live') {
   572→      // restore the camera as the outgoing video source and the local preview
   573→      this.publishLocalTrack(this.cameraTrack, this.localStream ?? new MediaStream([this.cameraTrack]))
   574→      if (this.localStream) putStream('self', this.localStream)
   575→      await this.setVideoSendersActive(useCallStore.getState().self.camOn)
   576→    } else {
   577→      // no camera available: stop sending video entirely
   578→      await this.setVideoSendersActive(false)
   579→    }
   580→    useCallStore.getState().setSelf({ sharing: false })
   581→    this.socket?.emit('state:update', { sharing: false })
   582→    const spotlight = useCallStore.getState().spotlight
   583→    if (spotlight === 'self') useCallStore.getState().setSpotlight(null)
   584→  }

   586→  private async setVideoSendersActive(active: boolean) {
   587→    for (const [, senders] of this.sendersMap) {
   588→      const sender = senders.video
   589→      if (!sender) continue
   590→      try {
   591→        const params = sender.getParameters()
   592→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   593→        params.encodings[0].active = active
   594→        await sender.setParameters(params)
   595→      } catch {
   596→        /* browser may not support active toggle */
   597→      }
   598→    }
   599→  }

   601→  private capAudioSender(pc: RTCPeerConnection) {
   602→    const sender = pc.getSenders().find((s) => s.track?.kind === 'audio')
   603→    if (!sender) return
   604→    try {
   605→      const params = sender.getParameters()
   606→      if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   607→      params.encodings[0].maxBitrate = AUDIO_SENDER_BITRATE
   608→      void sender.setParameters(params).catch(() => undefined)
   609→    } catch {
   610→      /* ignore */
   611→    }
   612→  }

   616→  private resolveProfile(): QualityProfile {
   617→    const preset = useCallStore.getState().preset
   618→    const key = preset === 'auto' ? this.adaptiveProfile : preset
   619→    return QUALITY_PROFILES[key]
   620→  }

   622→  private effectiveLabel(): string {
   623→    const preset = useCallStore.getState().preset
   624→    const profile = this.resolveProfile()
   625→    const short = profile.label.split('·')[1]?.trim() ?? ''
   626→    return preset === 'auto' ? `Auto · ${short}` : profile.label
   627→  }

   629→  async setPreset(p: QualityPreset) {
   630→    this.preset = p
   631→    useCallStore.getState().setPreset(p)
   632→    if (p !== 'auto') this.adaptiveProfile = p
   633→    useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   634→    await this.applyProfile()
   635→  }
   636→
   637→  private async applyProfile() {
   638→    const profile = this.resolveProfile()
   639→    // 1) Capture resolution / fps (no renegotiation needed)
   640→    if (this.cameraTrack && !this.screenTrack && this.cameraTrack.readyState === 'live') {
   641→      try {
   642→        await this.cameraTrack.applyConstraints(this.videoConstraints(profile))
   643→      } catch {
   644→        /* some cameras ignore constraints */
   645→      }
   646→    }
   647→    // 2) Bitrate caps per sender, respecting per-link adaptive caps
   648→    for (const [peerId, senders] of this.sendersMap) {
   649→      const sender = senders.video
   650→      if (!sender) continue
   651→      const link = this.links.get(peerId)
   652→      const cap = link?.lastCap ?? null
   653→      const target = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   654→      try {
   655→        const params = sender.getParameters()
   656→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   657→        params.encodings[0].maxBitrate = target
   658→        await sender.setParameters(params)
   659→      } catch {
   660→        /* ignore */
   661→      }
   662→    }
   663→  }

   665→  /* ---------------- stats / adaptation ---------------- */
   666→
   667→  private async pollStats() {
   668→    if (this.pcs.size === 0) return
   669→    let overall: NetworkLevel = 3
   670→    let totalKbps = 0
   671→
   672→    for (const [peerId, pc] of this.pcs) {
   673→      if (pc.connectionState !== 'connected') {
   674→        overall = Math.min(overall, 1) as NetworkLevel
   675→        continue
   676→      }
   677→      let rtt: number | null = null
   678→      let loss: number | null = null
   679→      let bytesDelta = 0
   680→      try {
   681→        const stats = await pc.getStats()
   682→        const nextPrev = new Map<string, number>()
   683→        stats.forEach((report) => {
   684→          const r = report as RTCStats & {
   685→            currentRoundTripTime?: number
   686→            fractionLost?: number
   687→            type: string
   688→            state?: string
   689→            nominated?: boolean
   690→            bytesSent?: number
   691→            kind?: string
   692→          }
   693→          if (r.type === 'candidate-pair' && r.state === 'succeeded' && typeof r.currentRoundTripTime === 'number') {
   694→            rtt = rtt === null ? r.currentRoundTripTime : Math.min(rtt, r.currentRoundTripTime)
   695→          }
   696→          if (r.type === 'remote-inbound-rtp' && typeof r.fractionLost === 'number') {
   697→            loss = loss === null ? r.fractionLost : Math.max(loss, r.fractionLost)
   698→          }
   699→          if (r.type === 'outbound-rtp' && r.kind === 'video' && typeof r.bytesSent === 'number') {
   700→            const prevB = this.prevBytes.get(r.id) ?? r.bytesSent
   701→            bytesDelta += Math.max(0, r.bytesSent - prevB)
   702→            nextPrev.set(r.id, r.bytesSent)
   703→          }
   704→        })
   705→        this.prevBytes = nextPrev
   706→      } catch {
   707→        continue
   708→      }
   709→
   710→      const level = classifyLink(rtt, loss)
   711→      const prevLevel = useCallStore.getState().peers[peerId]?.level
   712→      if (prevLevel !== level) useCallStore.getState().patchPeer(peerId, { level })
   713→      overall = Math.min(overall, level) as NetworkLevel
   714→      totalKbps += (bytesDelta * 8) / (STATS_INTERVAL_MS * 1000)
   715→
   716→      // Adaptive per-link video cap
   717→      const link = this.links.get(peerId)
   718→      const cap = LINK_CAPS[level]
   719→      if (link && link.lastCap !== cap) {
   720→        link.lastCap = cap
   721→        const sender = this.sendersMap.get(peerId)?.video
   722→        if (sender) {
   723→          try {
   724→            const params = sender.getParameters()
   725→            if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   726→            const profile = this.resolveProfile()
   727→            params.encodings[0].maxBitrate = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   728→            await sender.setParameters(params)
   729→          } catch {
   730→            /* ignore */
   731→          }
   732→        }
   733→      }
   734→    }
   735→
   736→    const prevOverall = useCallStore.getState().overallLevel
   737→    if (prevOverall !== overall) {
   738→      useCallStore.getState().setOverallLevel(overall)
   739→      useCallStore.getState().patchPeer(this.myId, { level: overall })
   740→    }
   741→    useCallStore.setState({ bitrateKbps: Math.round(totalKbps) })
   742→
   743→    // Auto quality profile with hysteresis
   744→    if (this.preset === 'auto') {
   745→      if (overall <= 1) {
   746→        this.badCount++
   747→        this.goodCount = 0
   748→      } else if (overall === 3) {
   749→        this.goodCount++
   750→        this.badCount = 0
   751→      } else {
   752→        this.badCount = 0
   753→        this.goodCount = 0
   754→      }
   755→      const order: Array<Exclude<QualityPreset, 'auto'>> = ['saver', 'balanced', 'sharp']
   756→      const idx = order.indexOf(this.adaptiveProfile)
   757→      if (this.badCount >= 2 && idx > 0) {
   758→        this.adaptiveProfile = order[idx - 1]
   759→        this.badCount = 0
   760→        this.goodCount = 0
   761→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   762→        void this.applyProfile()
   763→      } else if (this.goodCount >= 8 && idx < order.length - 1) {
   764→        this.adaptiveProfile = order[idx + 1]
   765→        this.badCount = 0
   766→        this.goodCount = 0
   767→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   768→        void this.applyProfile()
   769→      }
   770→    }
   771→  }

   781→  private rejoin() {
   782→    if (!this.roomId || !this.socket) return
   783→    this.socket.emit(
   784→      'room:join',
   785→      { roomId: this.roomId, name: useCallStore.getState().self.name },
   786→      (res: { ok: boolean; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {

   788→      },
   789→    )
   790→  }

   792→  async leave(silent = false) {
   793→    if (this.statsTimer !== null) {
   794→      window.clearInterval(this.statsTimer)
   795→      this.statsTimer = null
   796→    }
   797→    this.socket?.emit('room:leave')
   798→    for (const peerId of Array.from(this.pcs.keys())) this.dropPeer(peerId)
   799→    this.screenTrack?.stop()
   800→    this.localStream?.getTracks().forEach((t) => {
   801→      try {
   802→        t.stop()
   803→      } catch {
   804→        /* ignore */
   805→      }
   806→    })
   807→    const selfStreamTest = (this.localStream as (MediaStream & { __stop?: () => void }) | null)?.__stop
   808→    if (selfStreamTest) selfStreamTest()
   809→    this.localStream = null
   810→    this.cameraTrack = null
   811→    this.micTrack = null
   812→    putStream('self', null)
   813→    this.pcs.clear()
   814→    this.pendingIce.clear()
   815→    this.links.clear()
   816→    this.prevBytes.clear()
   817→    this.socket?.disconnect()
   818→    this.socket = null
   819→    this.myId = ''
   820→    this.started = false
   821→    if (!silent) {
   822→      useCallStore.getState().reset()
   823→    }
   824→  }
