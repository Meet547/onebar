    21→/**
    22→ * Pico call engine — peer-to-peer WebRTC mesh.
    23→ *
    24→ * Design goals:
    25→ *  - Media never touches a server: every audio/video packet travels directly
    26→ *    between participants (mesh topology). The signaling socket only carries
    27→ *    SDP/ICE/chat text.
    28→ *  - Renegotiation-free quality control: resolution/fps changes ride on
    29→ *    applyConstraints(), bitrate changes on setParameters(), screen share on
    30→ *    replaceTrack(). No expensive offer/answer churn.
    31→ *  - Opus DTX + 24 kbps cap: silence costs (almost) nothing on the wire.
    32→ *  - Per-link adaptive bitrate: getStats() every 3s classifies each
    33→ *    connection (RTT + packet loss) and caps outgoing video per link, so one
    34→ *    weak participant never drags the whole call down.
    35→ *  - Perfect-negotiation pattern (polite/impolite) for glare-free offers.
    36→ */
    37→
    38→const ICE_SERVERS: RTCIceServer[] = [
    39→  { urls: ['stun:stun.l.google.com:19302', 'stun:stun1.l.google.com:19302'] },
    40→  { urls: ['stun:stun.cloudflare.com:3478'] },
    41→]
    42→
    43→/**
    44→ * ICE servers with TURN support, fetched from /api/ice. TURN is what makes
    45→ * calls work between strict NATs (mobile carriers, offices, hotel wifi) —
    46→ * without it a meaningful slice of real-world pairs can never connect.
    47→ * Cached for an hour; a failed fetch falls back to STUN-only, never blocks
    48→ * the call for more than the 4s timeout.
    49→ */
    50→const ICE_CACHE_TTL_MS = 3_600_000
    51→let iceCache: { servers: RTCIceServer[]; at: number } | null = null
    52→
    53→async function fetchIceServers(): Promise<RTCIceServer[]> {
    54→  if (iceCache && Date.now() - iceCache.at < ICE_CACHE_TTL_MS) return iceCache.servers
    55→  try {
    56→    const ctrl = new AbortController()
    57→    const timer = window.setTimeout(() => ctrl.abort(), 4000)
    58→    const res = await fetch('/api/ice', { signal: ctrl.signal, cache: 'no-store' })
    59→    window.clearTimeout(timer)
    60→    if (res.ok) {
    61→      const data = (await res.json()) as { iceServers?: RTCIceServer[] }
    62→      if (Array.isArray(data.iceServers) && data.iceServers.length > 0) {
    63→        iceCache = { servers: data.iceServers, at: Date.now() }
    64→        return iceCache.servers
    65→      }
    66→    }
    67→  } catch {
    68→    /* unreachable or slow — STUN-only still works on friendly networks */
    69→  }
    70→  iceCache = { servers: ICE_SERVERS, at: Date.now() }
    71→  return iceCache.servers
    72→}

   113→/**
   114→ * Opus tuning per m-section. Speech sections get DTX + a 24 kbps ceiling so
   115→ * silence costs (almost) nothing; music sections get stereo and a high
   116→ * ceiling. Sections are recognized by their msid (local sends) or by mid
   117→ * (recvonly answers, where the msid lives in the remote offer).
   118→ */
   119→function mungeOpus(sdp: string, musicIds: Set<string>, musicMids: Set<string>): string {
   120→  const lines = sdp.split('\r\n')
   121→  const mIdx: number[] = []
   122→  lines.forEach((l, i) => {
   123→    if (l.startsWith('m=')) mIdx.push(i)
   124→  })
   125→  mIdx.push(lines.length)
   126→
   127→  const sectionMusic = new Array<boolean>(lines.length).fill(false)
   128→  for (let s = 0; s < mIdx.length - 1; s++) {
   129→    const from = mIdx[s]
   130→    const to = mIdx[s + 1]
   131→    let music = false
   132→    for (let i = from; i < to; i++) {
   133→      const msid = lines[i].match(/^a=msid:(\S+)/)
   134→      if (msid && musicIds.has(msid[1])) {
   135→        music = true
   136→        break
   137→      }
   138→      const mid = lines[i].match(/^a=mid:(.+)$/)
   139→      if (mid && musicMids.has(mid[1].trim())) {
   140→        music = true
   141→        break
   142→      }
   143→    }
   144→    for (let i = from; i < to; i++) sectionMusic[i] = music
   145→  }
   146→
   147→  const opusPTs = new Set<string>()
   148→  for (const line of lines) {
   149→    const m = line.match(/^a=rtpmap:(\d+) opus\/48000/i)
   150→    if (m) opusPTs.add(m[1])
   151→  }
   152→
   153→  const out = lines.map((line, i) => {
   154→    const m = line.match(/^a=fmtp:(\d+)(.*)$/)
   155→    if (!m || !opusPTs.has(m[1])) return line
   156→    let params = m[2] ?? ''
   157→    if (sectionMusic[i]) {
   158→      params = params.replace(/;?(usedtx|maxaveragebitrate|stereo)=[^;]*/gi, '')
   159→      params += ';stereo=1;maxaveragebitrate=128000'
   160→    } else {
   161→      params = params.replace(/;?(stereo|maxaveragebitrate)=[^;]*/gi, '')
   162→      if (!/usedtx/i.test(params)) params += ';usedtx=1'
   163→      if (!/maxaveragebitrate/i.test(params)) params += ';maxaveragebitrate=24000'
   164→    }
   165→    return `a=fmtp:${m[1]}${params}`
   166→  })
   167→  return out.join('\r\n')
   168→}
   169→
   170→function classifyLink(rttSec: number | null, lossFrac: number | null): NetworkLevel {
   171→  const rtt = rttSec ?? 0.4
   172→  const loss = lossFrac ?? 0.05
   173→  if (rtt > 0.5 || loss > 0.08) return 0
   174→  if (rtt > 0.3 || loss > 0.05) return 1
   175→  if (rtt > 0.15 || loss > 0.02) return 2
   176→  return 3
   177→}

   188→  private cameraTrack: MediaStreamTrack | null = null

   192→  private sendersMap = new Map<string, { audio?: RTCRtpSender; video?: RTCRtpSender; music?: RTCRtpSender }>()
   193→  private pendingIce = new Map<string, RTCIceCandidateInit[]>()
   194→  private links = new Map<string, LinkInfo>()
   195→  private statsTimer: number | null = null
   196→  private makingOffer = new Map<string, boolean>()
   197→  private ignoreOffer = new Map<string, boolean>()
   198→  /** Per-peer signaling op queue — SRD/SLD calls must never interleave. */
   199→  private sigQueue = new Map<string, Promise<void>>()
   200→  /** When our last offer went out per peer — drives the renegotiation watchdog. */
   201→  private lastOfferAt = new Map<string, number>()
   202→  private preset: QualityPreset = 'balanced'
   203→  private adaptiveProfile: Exclude<QualityPreset, 'auto'> = 'balanced'
   204→  private badCount = 0
   205→  private goodCount = 0
   206→  private started = false
   207→  private testMode = false
   208→  private mediaMode: MediaMode = 'full'
   209→  private prevBytes = new Map<string, number>()
   210→  private passwordHash = ''

   212→
   213→  /* listen together */
   214→  private musicSource: MusicSource | null = null
   215→  private musicHandleStop: (() => void) | null = null
   216→  private musicUnsub: (() => void) | null = null
   217→  private musicMetaTimer: number | null = null
   218→  private musicStreamId = ''
   219→  private musicOrigin: 'file' | 'test' | 'tab' = 'file'

   222→  private musicMids = new Set<string>()

   225→  async start(opts: StartOptions): Promise<{ mode: MediaMode }> {
   226→    if (this.started) await this.leave(true)
   227→    this.started = true
   228→    this.roomId = opts.roomId
   229→    this.preset = opts.preset
   230→    this.adaptiveProfile = opts.preset === 'auto' ? 'balanced' : opts.preset
   231→    this.testMode = !!opts.testMode
   232→    this.passwordHash = opts.passwordHash ?? ''
   233→
   234→    const store = useCallStore.getState()
   235→    store.setSelf({ name: opts.name })
   236→    store.setPreset(opts.preset)
   237→    store.setEffectiveLabel(this.effectiveLabel())
   238→
   239→    // ---- ICE (TURN/STUN) — resolved before any peer connection is built ----
   240→    this.iceServers = await fetchIceServers()
   241→
   242→    // ---- Signaling ----
   243→    const socket = io(SIGNALING_URL, {
   244→      path: SOCKET_PATH,
   245→      transports: ['websocket', 'polling'],
   246→      reconnection: true,
   247→      reconnectionAttempts: Infinity,
   248→      reconnectionDelay: 1000,
   249→      reconnectionDelayMax: 4000,
   250→      timeout: 8000,
   251→    })
   252→    this.socket = socket
   253→    // Dev/debug handle (harmless in prod, invaluable for live diagnostics)
   254→    if (typeof window !== 'undefined') {
   255→      ;(window as unknown as { __pico?: unknown }).__pico = { engine: this, socket }
   256→    }
   257→    socket.on('connect', () => {
   258→      useCallStore.getState().setSignalingConnected(true)
   259→      if (this.myId) this.rejoin()
   260→    })
   261→    socket.on('disconnect', () => useCallStore.getState().setSignalingConnected(false))
   262→    socket.on('connect_error', () => useCallStore.getState().setSignalingConnected(false))
   263→
   264→    // ---- Local media ----
   265→    this.localStream = opts.initialStream ?? null
   266→    if (this.localStream) {
   267→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   268→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   269→      this.mediaMode = this.cameraTrack ? 'full' : this.micTrack ? 'audio' : 'none'
   270→      // Register the handed-over preview stream so tiles can attach to it
   271→      putStream('self', this.localStream)
   272→    }
   273→    if (!this.localStream || (!this.cameraTrack && !this.micTrack)) {
   274→      this.mediaMode = await this.acquireMedia()
   275→    }
   276→    this.applyInitialTrackState()
   277→    trackAudio('self', this.localStream)
   278→
   279→    // ---- Room ----
   280→    // The ack races a timeout so a dead signaling server can never hang the UI.
   281→    // A rejected join (full room, wrong passcode, locked) aborts the start.
   282→    const joinResult = await new Promise<{ ok: boolean; reason?: string; hostId?: string; locked?: boolean }>((resolve) => {
   283→      let settled = false
   284→      const finish = (r: { ok: boolean; reason?: string }) => {
   285→        if (!settled) {
   286→          settled = true
   287→          resolve(r)
   288→        }
   289→      }
   290→      const guard = window.setTimeout(() => finish({ ok: false, reason: 'Could not reach the call relay. Check your connection.' }), 12000)
   291→      socket.emit(
   292→        'room:join',
   293→        { roomId: this.roomId, name: opts.name, passwordHash: this.passwordHash },
   294→        (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; hostId?: string; locked?: boolean; hasPassword?: boolean; music?: { peerId: string; title: string; playing: boolean; position: number; duration: number; streamId?: string } | null; reason?: string }) => {
   295→          window.clearTimeout(guard)
   296→          if (!res?.ok) {
   297→            finish({ ok: false, reason: res?.reason ?? 'Could not join the room' })
   298→            return
   299→          }
   300→          this.myId = res?.self?.id ?? socket.id!
   301→          useCallStore.getState().setSelf({ id: this.myId })
   302→          useCallStore.getState().setHostId(res.hostId ?? null)
   303→          useCallStore.getState().setLocked(!!res.locked)
   304→          useCallStore.getState().setPasscodeProtected(!!res.hasPassword)
   305→          if (res.music?.peerId) {
   306→            // Late joiner: the music track is already flowing — learn its
   307→            // stream id now so the incoming ontrack is routed to the player.
   308→            this.remoteMusicStreamIds.clear()
   309→            if (res.music.streamId) this.remoteMusicStreamIds.add(res.music.streamId)
   310→            useCallStore.getState().setMusic({
   311→              peerId: res.music.peerId,
   312→              title: res.music.title,
   313→              playing: res.music.playing,
   314→              position: res.music.position,
   315→              duration: res.music.duration,
   316→            })
   317→          }
   318→          if (res?.peers) this.reconcilePeers(res.peers)
   319→          finish({ ok: true })
   320→        },
   321→      )
   322→    })
   323→
   324→    if (!joinResult.ok) {
   325→      await this.cleanupAfterFailedJoin()
   326→      throw new Error(joinResult.reason ?? 'Could not join the room')
   327→    }
   328→
   329→    // ---- Events ----
   330→    socket.on('peer:joined', ({ peer }: { peer: { id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean } }) => {
   331→      this.ensurePeer(peer.id, peer.name, peer.micOn, peer.camOn, peer.sharing)
   332→      useCallStore.getState().addChat({
   333→        id: 'sys-' + Math.random(),
   334→        from: 'system',
   335→        name: 'System',
   336→        text: `${peer.name} joined`,
   337→        ts: Date.now(),
   338→        system: true,
   339→      })
   340→    })
   341→    socket.on('peer:left', ({ peerId }: { peerId: string }) => this.dropPeer(peerId))
   342→    socket.on('room:peers', ({ peers }: { peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   343→      // Only meaningful after a reconnect — reconcile full presence
   344→      if (this.myId) this.reconcilePeers(peers)
   345→    })
   346→    socket.on('peer:state', ({ peerId, micOn, camOn, sharing }: { peerId: string; micOn: boolean; camOn: boolean; sharing: boolean }) => {
   347→      const store = useCallStore.getState()
   348→      store.patchPeer(peerId, { micOn, camOn, sharing })
   349→      // Follow whoever is presenting, so a remote screen never hides in a small tile
   350→      if (sharing && store.spotlight !== peerId) store.setSpotlight(peerId)
   351→      else if (!sharing && store.spotlight === peerId) store.setSpotlight(null)
   352→    })
   353→    socket.on('chat:message', (msg: { id: string; from: string; name: string; text: string; ts: number }) => {
   354→      useCallStore.getState().addChat(msg)
   355→    })
   356→
   357→    /* ---- security & host controls ---- */
   358→    socket.on('host:changed', ({ hostId }: { hostId: string }) => {
   359→      useCallStore.getState().setHostId(hostId)
   360→      if (hostId === this.myId) {
   361→        toast('You are now the host', { duration: 2400 })
   362→        useCallStore.getState().addChat({
   363→          id: 'sys-' + Math.random(),
   364→          from: 'system',
   365→          name: 'System',
   366→          text: 'You are now the host',
   367→          ts: Date.now(),
   368→          system: true,
   369→        })
   370→      }
   371→    })
   372→    socket.on('host:force-mute', () => {
   373→      if (this.micTrack) this.micTrack.enabled = false
   374→      useCallStore.getState().setSelf({ micOn: false })
   375→      this.socket?.emit('state:update', { micOn: false })
   376→      toast('The host muted your microphone', { duration: 2800 })
   377→    })
   378→    socket.on('room:locked', ({ locked }: { locked: boolean }) => {
   379→      useCallStore.getState().setLocked(locked)
   380→      useCallStore.getState().addChat({
   381→        id: 'sys-' + Math.random(),
   382→        from: 'system',
   383→        name: 'System',
   384→        text: locked ? 'The host locked this call' : 'The host unlocked this call',
   385→        ts: Date.now(),
   386→        system: true,
   387→      })
   388→    })
   389→    socket.on('room:kicked', () => {
   390→      useCallStore.getState().setKicked(true)
   391→    })
   392→
   393→    /* ---- listen together ---- */
   394→    socket.on('music:state', (data: { peerId: string; action: 'start' | 'meta' | 'stop'; streamId?: string; title?: string; playing?: boolean; position?: number; duration?: number }) => {
   395→      const store = useCallStore.getState()
   396→      if (data.action === 'stop' || !data.peerId) {
   397→        this.remoteMusicStreamIds.clear()
   398→        detachRemoteMusic()
   399→        store.setMusic({ ...INITIAL_MUSIC })
   400→        return
   401→      }
   402→      if (data.action === 'start' && data.streamId) {
   403→        this.remoteMusicStreamIds.add(data.streamId)
   404→        store.setMusic({
   405→          peerId: data.peerId,
   406→          title: data.title ?? 'Music',
   407→          playing: !!data.playing,
   408→          position: data.position ?? 0,
   409→          duration: data.duration ?? 0,
   410→        })
   411→      } else if (data.action === 'meta') {
   412→        store.setMusic({
   413→          title: data.title ?? store.music.title,
   414→          playing: !!data.playing,
   415→          position: data.position ?? 0,
   416→          duration: data.duration ?? store.music.duration,
   417→        })
   418→      }
   419→    })
   420→    socket.on('music:busy', () => {
   421→      toast('Someone is already sharing music', { duration: 2400 })
   422→    })
   423→
   424→    socket.on('rtc:offer', ({ from, sdp }: { from: string; sdp: string }) => {
   425→      const pc = this.pcs.get(from)
   426→      if (!pc) return
   427→      const polite = this.myId < from
   428→      this.enqueueSig(from, async () => {
   429→        try {
   430→          // Re-evaluate glare inside the queue — the state may have moved while
   431→          // this op waited for the previous one.
   432→          const offerCollision = this.makingOffer.get(from) || pc.signalingState !== 'stable'
   433→          this.ignoreOffer.set(from, !polite && offerCollision)
   434→          if (this.ignoreOffer.get(from)) return
   435→          await pc.setRemoteDescription({ type: 'offer', sdp: this.munge(sdp) })
   436→          await this.flushIce(from)
   437→          const answer = await pc.createAnswer()
   438→          await pc.setLocalDescription({ type: 'answer', sdp: this.munge(answer.sdp ?? '', pc.remoteDescription?.sdp ?? '') })
   439→          const desc = pc.localDescription
   440→          if (desc) this.socket?.emit('rtc:answer', { to: from, sdp: desc.sdp })
   441→        } catch (err) {
   442→          // A peer that rebuilt its connection mid-session can offer a
   443→          // different m-line layout than ours; rebuild our side so the next
   444→          // offer lands on a matching PC instead of failing forever.
   445→          if (err instanceof Error && /m-line/i.test(err.message) && this.pcs.get(from) === pc) {
   446→            const info = useCallStore.getState().peers[from]
   447→            this.dropPeer(from)
   448→            this.ensurePeer(from, info?.name ?? 'Guest', info?.micOn ?? true, info?.camOn ?? true, info?.sharing ?? false)
   449→          }
   450→        }
   451→      })
   452→    })
   453→
   454→    socket.on('rtc:answer', ({ from, sdp }: { from: string; sdp: string }) => {
   455→      const pc = this.pcs.get(from)
   456→      if (!pc) return
   457→      this.enqueueSig(from, async () => {
   458→        try {
   459→          if (pc.signalingState !== 'have-local-offer') return
   460→          await pc.setRemoteDescription({ type: 'answer', sdp: this.munge(sdp, pc.localDescription?.sdp ?? '') })
   461→          await this.flushIce(from)
   462→        } catch {
   463→          /* the renegotiation watchdog re-offers if the pair stays stuck */
   464→        }
   465→      })
   466→    })
   467→
   468→    socket.on('rtc:ice', ({ from, candidate }: { from: string; candidate: RTCIceCandidateInit }) => {
   469→      const pc = this.pcs.get(from)
   470→      if (!pc) return
   471→      if (pc.remoteDescription) {
   472→        pc.addIceCandidate(candidate).catch(() => {
   473→          if (!this.ignoreOffer.get(from)) {
   474→            this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   475→          }
   476→        })
   477→      } else {
   478→        this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   479→      }
   480→    })
   481→
   482→    // ---- Stats loop ----
   483→    this.statsTimer = window.setInterval(() => void this.pollStats(), STATS_INTERVAL_MS)
   484→
   485→    return { mode: this.mediaMode }
   486→  }

   490→  private async acquireMedia(): Promise<MediaMode> {
   491→    if (this.testMode) {
   492→      this.localStream = await createTestStream()
   493→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   494→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   495→      putStream('self', this.localStream)
   496→      return 'full'
   497→    }
   498→    const audio: MediaTrackConstraints = {
   499→      echoCancellation: true,
   500→      noiseSuppression: true,
   501→      autoGainControl: true,
   502→      channelCount: 1,
   503→    }
   504→    const video = this.videoConstraints(this.resolveProfile())
   505→    // Full quality first
   506→    try {
   507→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio, video })
   508→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   509→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   510→      this.attachLocalStream()
   511→      return 'full'
   512→    } catch {
   513→      /* fall through */
   514→    }
   515→    // Audio only
   516→    try {
   517→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio })

   519→      this.cameraTrack = null
   520→      this.attachLocalStream()
   521→      return 'audio'
   522→    } catch {
   523→      /* fall through */
   524→    }
   525→    // Listener mode
   526→    this.localStream = null
   527→    return 'none'
   528→  }

   551→  /* ---------------- peers ---------------- */
   552→
   553→  private ensurePeer(
   554→    peerId: string,
   555→    name: string,
   556→    micOn = true,
   557→    camOn = true,
   558→    sharing = false,
   559→  ) {
   560→    const store = useCallStore.getState()
   561→    if (!store.peers[peerId]) {
   562→      store.upsertPeer({ id: peerId, name, micOn, camOn, sharing, level: 3, hasStream: false, streamVersion: 0 })
   563→    }
   564→    if (this.pcs.has(peerId)) return
   565→
   566→    const pc = new RTCPeerConnection({
   567→      iceServers: this.iceServers,
   568→      iceCandidatePoolSize: 4,
   569→      bundlePolicy: 'max-bundle',
   570→      rtcpMuxPolicy: 'require',
   571→    })
   572→    this.pcs.set(peerId, pc)
   573→    this.links.set(peerId, { lastCap: null })
   574→    this.makingOffer.set(peerId, false)
   575→    this.ignoreOffer.set(peerId, false)
   576→
   577→    // Local media into this connection
   578→    const senders: { audio?: RTCRtpSender; video?: RTCRtpSender; music?: RTCRtpSender } = {}
   579→    if (this.localStream) {
   580→      for (const track of this.localStream.getTracks()) {
   581→        try {
   582→          const sender = pc.addTrack(track, this.localStream)
   583→          if (track.kind === 'audio') senders.audio = sender
   584→          else senders.video = sender
   585→        } catch {
   586→          /* ignore */
   587→        }
   588→      }
   589→      // A rebuilt connection (reconnect, ICE restart) must carry the CURRENT
   590→      // outgoing video — if the screen is being shared, that is the screen
   591→      // track, not the camera that localStream still holds.
   592→      if (this.screenTrack && this.screenTrack.readyState === 'live' && senders.video) {
   593→        void senders.video.replaceTrack(this.screenTrack).catch(() => undefined)
   594→      }
   595→      // Keep camera-off cameras off on fresh connections too
   596→      if (!useCallStore.getState().self.camOn && !this.screenTrack) {
   597→        void this.setVideoSendersActive(false)
   598→      }
   599→    } else {
   600→      const at = pc.addTransceiver('audio', { direction: 'recvonly' })
   601→      const vt = pc.addTransceiver('video', { direction: 'recvonly' })
   602→      senders.audio = at.sender
   603→      senders.video = vt.sender
   604→    }
   605→    this.sendersMap.set(peerId, senders)
   606→
   607→    // Reserve the music m-line when this peer is the active listen-together
   608→    // sharer: that PC's layout is [audio, video, music], and a connection
   609→    // rebuilt mid-session must match it or every offer fails m-line order
   610→    // checks on the polite side (silent negotiation deadlock).
   611→    const musicPeerId = useCallStore.getState().music.peerId
   612→    if (musicPeerId && musicPeerId === peerId && !senders.music) {
   613→      try {
   614→        const t = pc.addTransceiver('audio', { direction: 'recvonly' })
   615→        senders.music = t.sender
   616→      } catch {
   617→        /* ignore */
   618→      }
   619→    }
   620→
   621→    pc.onicecandidate = (e) => {
   622→      if (e.candidate) this.socket?.emit('rtc:ice', { to: peerId, candidate: e.candidate.toJSON() })
   623→    }
   624→
   625→    pc.ontrack = (e) => {
   626→      const stream = e.streams[0]
   627→      if (!stream) return
   628→      // Listen-together streams ride their own MediaStream — recognize them by
   629→      // the stream id the sharer announced before the track ever lands.
   630→      if (this.remoteMusicStreamIds.has(stream.id)) {
   631→        void attachRemoteMusic(stream)
   632→        return
   633→      }
   634→      putStream(peerId, stream)
   635→      trackAudio(peerId, stream)
   636→      useCallStore.getState().bumpPeerStream(peerId, true)
   637→    }
   638→
   639→    pc.onnegotiationneeded = () => {
   640→      this.enqueueSig(peerId, () => this.makeOffer(peerId, pc))
   641→    }
   642→
   643→    pc.onconnectionstatechange = () => {
   644→      if (pc.connectionState === 'failed') {
   645→        try {
   646→          pc.restartIce()
   647→        } catch {
   648→          /* ignore */
   649→        }
   650→      }
   651→    }
   652→
   653→    // Prime encoding params on the fresh senders
   654→    this.capAudioSender(pc)
   655→    const videoSender = senders.video
   656→    if (videoSender) {
   657→      try {
   658→        const params = videoSender.getParameters()
   659→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   660→        params.encodings[0].maxBitrate = this.resolveProfile().maxBitrate
   661→        void videoSender.setParameters(params).catch(() => undefined)
   662→      } catch {
   663→        /* ignore */
   664→      }
   665→    }
   666→    // Someone joining mid-song gets the music track immediately
   667→    if (this.musicSource) this.ensureMusicSender(peerId, pc)
   668→  }
   669→
   670→  private dropPeer(peerId: string) {
   671→    const pc = this.pcs.get(peerId)
   672→    if (pc) {
   673→      pc.close()
   674→    }
   675→    this.pcs.delete(peerId)
   676→    this.sendersMap.delete(peerId)
   677→    this.pendingIce.delete(peerId)
   678→    this.links.delete(peerId)
   679→    this.makingOffer.delete(peerId)
   680→    this.ignoreOffer.delete(peerId)
   681→    this.sigQueue.delete(peerId)
   682→    this.lastOfferAt.delete(peerId)
   683→    trackAudio(peerId, null)
   684→    putStream(peerId, null)
   685→    useCallStore.getState().bumpPeerStream(peerId, false)
   686→    useCallStore.getState().removePeer(peerId)
   687→    // If the music sharer left, the session is over
   688→    if (useCallStore.getState().music.peerId === peerId) {
   689→      detachRemoteMusic()
   690→      this.remoteMusicStreamIds.clear()
   691→      useCallStore.getState().setMusic({ ...INITIAL_MUSIC })
   692→    }
   693→  }

   695→  private reconcilePeers(
   696→    peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>,
   697→  ) {
   698→    const store = useCallStore.getState()
   699→    const seen = new Set<string>()
   700→    for (const p of peers) {
   701→      if (p.id === this.myId) continue
   702→      seen.add(p.id)
   703→      this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   704→    }
   705→    for (const id of Object.keys(store.peers)) {
   706→      if (!seen.has(id)) this.dropPeer(id)
   707→    }
   708→    // Follow whoever is presenting — also for late joiners and reconnects,
   709→    // not just on live peer:state events.
   710→    if (!store.spotlight) {
   711→      const presenter = peers.find((p) => p.sharing && p.id !== this.myId)
   712→      if (presenter) useCallStore.getState().setSpotlight(presenter.id)
   713→    }
   714→    // After a reconnect, old connections may be stale — rebuild them
   715→    for (const [peerId, pc] of this.pcs) {
   716→      if (pc.connectionState === 'failed' || pc.connectionState === 'closed') {
   717→        this.dropPeer(peerId)
   718→        const p = peers.find((x) => x.id === peerId)
   719→        if (p) this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   720→      }
   721→    }
   722→  }

   724→  private async flushIce(peerId: string) {
   725→    const pc = this.pcs.get(peerId)
   726→    const queued = this.pendingIce.get(peerId) ?? []
   727→    if (!pc || queued.length === 0) return
   728→    this.pendingIce.set(peerId, [])
   729→    for (const c of queued) {
   730→      try {
   731→        await pc.addIceCandidate(c)
   732→      } catch {
   733→        /* ignore */
   734→      }
   735→    }
   736→  }
   737→
   738→  /**
   739→   * Serialize signaling ops per peer. Perfect negotiation only works when
   740→   * setRemoteDescription / setLocalDescription never interleave — during
   741→   * simultaneous reconnects both peers offer at once, and an in-flight
   742→   * setLocalDescription makes the other side's implicit rollback reject.
   743→   * Without this queue that rejection was silently swallowed and the pair
   744→   * sat in have-local-offer forever.
   745→   */
   746→  private enqueueSig(peerId: string, op: () => Promise<void>): Promise<void> {
   747→    const prev = this.sigQueue.get(peerId) ?? Promise.resolve()
   748→    const next = prev.then(op, op)
   749→    this.sigQueue.set(peerId, next)
   750→    next.catch(() => undefined) // branch: keep the chain usable, log nothing
   751→    return next
   752→  }
   753→
   754→  private async makeOffer(peerId: string, pc: RTCPeerConnection) {
   755→    this.makingOffer.set(peerId, true)
   756→    try {
   757→      const offer = await pc.createOffer()
   758→      await pc.setLocalDescription({ type: 'offer', sdp: this.munge(offer.sdp ?? '') })
   759→      const desc = pc.localDescription
   760→      if (desc) {
   761→        this.lastOfferAt.set(peerId, Date.now())
   762→        this.socket?.emit('rtc:offer', { to: peerId, sdp: desc.sdp })
   763→      }
   764→    } finally {
   765→      this.makingOffer.set(peerId, false)
   766→    }
   767→  }

   771→  private musicIds(): Set<string> {
   772→    const ids = new Set<string>()

   778→  /** mids of sections that carry a music msid in a reference (offer) SDP. */
   779→  private midsWithMusicSdp(referenceSdp: string): Set<string> {
   780→    const ids = this.musicIds()
   781→    const mids = new Set<string>()
   782→    if (!referenceSdp || ids.size === 0) return mids
   783→    let mid = ''
   784→    for (const line of referenceSdp.split('\r\n')) {
   785→      if (line.startsWith('m=')) mid = ''
   786→      const m1 = line.match(/^a=mid:(.+)$/)
   787→      if (m1) mid = m1[1].trim()
   788→      const m2 = line.match(/^a=msid:(\S+)/)
   789→      if (m2 && mid && ids.has(m2[1])) mids.add(mid)
   790→    }
   791→    return mids
   792→  }

   794→  private munge(sdp: string, referenceSdp = ''): string {
   795→    const ids = this.musicIds()
   796→    if (ids.size === 0) return mungeOpus(sdp, ids, new Set())
   797→    return mungeOpus(sdp, ids, this.midsWithMusicSdp(referenceSdp))
   798→  }

  1010→      }
  1011→      return true
  1012→    }
  1013→
  1014→    if (this.cameraTrack) this.cameraTrack.enabled = next
  1015→    await this.setVideoSendersActive(next && !this.screenTrack)
  1016→    return next
  1017→  }

  1019→  private publishLocalTrack(track: MediaStreamTrack, owner: MediaStream) {
  1020→    for (const [peerId, pc] of this.pcs) {
  1021→      const senders = this.sendersMap.get(peerId) ?? {}
  1022→      try {
  1023→        if (track.kind === 'video' && senders.video && senders.video.track?.kind === 'video') {
  1024→          // Swap the outgoing source — no renegotiation, seamless for viewers
  1025→          void senders.video.replaceTrack(track).catch(() => {
  1026→            try {
  1027→              const s = pc.addTrack(track, owner)
  1028→              senders.video = s
  1029→              this.sendersMap.set(peerId, senders)
  1030→            } catch {
  1031→              /* ignore */
  1032→            }
  1033→          })
  1034→        } else {
  1035→          const s = pc.addTrack(track, owner)
  1036→          if (track.kind === 'audio') senders.audio = s
  1037→          else senders.video = s
  1038→          this.sendersMap.set(peerId, senders)
  1039→        }
  1040→      } catch {
  1041→        /* ignore */
  1042→      }
  1043→    }
  1044→  }

  1046→  async toggleShare(): Promise<'started' | 'stopped' | 'cancelled'> {
  1047→    const sharing = useCallStore.getState().self.sharing
  1048→    if (!sharing) {
  1049→      try {
  1050→        const display = this.testMode
  1051→          ? await createTestScreenStream()
  1052→          : await navigator.mediaDevices.getDisplayMedia({ video: true, audio: true })
  1053→        const track = display.getVideoTracks()[0]
  1054→        if (!track) return 'cancelled'
  1055→        this.screenTrack = track
  1056→        track.addEventListener('ended', () => {
  1057→          void this.stopShare()
  1058→        })
  1059→        this.publishLocalTrack(track, display)
  1060→        // mirror the shared screen into the local preview tile
  1061→        putStream('self', display)
  1062→        useCallStore.getState().bumpPeerStream('self', true)
  1063→        useCallStore.getState().setSelf({ sharing: true })
  1064→        this.socket?.emit('state:update', { sharing: true })
  1065→        useCallStore.getState().setSpotlight('self')
  1066→        // Tab / system audio, when the browser hands it over, rides the
  1067→        // music channel (dedicated Opus stereo sender) — not the mic.
  1068→        const audioTrack = display.getAudioTracks()[0]
  1069→        if (audioTrack && !this.musicSource) {
  1070→          this.startMusicShare(
  1071→            {
  1072→              stream: new MediaStream([audioTrack]),
  1073→              title: 'Shared tab audio',
  1074→              duration: 0,
  1075→              getPosition: () => 0,
  1076→              isPlaying: () => true,
  1077→            },
  1078→            'tab',
  1079→          )
  1080→        }
  1081→        return 'started'
  1082→      } catch {
  1083→        return 'cancelled'
  1084→      }
  1085→    } else {
  1086→      await this.stopShare()
  1087→      return 'stopped'
  1088→    }
  1089→  }

  1091→  private async stopShare() {
  1092→    this.screenTrack?.stop()
  1093→    this.screenTrack = null
  1094→    if (this.musicOrigin === 'tab' && this.musicSource) {
  1095→      this.stopMusicShare()
  1096→    }
  1097→    if (this.cameraTrack && this.cameraTrack.readyState === 'live') {
  1098→      // restore the camera as the outgoing video source and the local preview

  1101→      await this.setVideoSendersActive(useCallStore.getState().self.camOn)
  1102→    } else {
  1103→      // no camera available: stop sending video entirely
  1104→      await this.setVideoSendersActive(false)
  1105→    }
  1106→    useCallStore.getState().setSelf({ sharing: false })
  1107→    this.socket?.emit('state:update', { sharing: false })

  1109→    if (spotlight === 'self') useCallStore.getState().setSpotlight(null)
  1110→  }

  1163→  private async applyProfile() {
  1164→    const profile = this.resolveProfile()
  1165→    // 1) Capture resolution / fps (no renegotiation needed)
  1166→    if (this.cameraTrack && !this.screenTrack && this.cameraTrack.readyState === 'live') {
  1167→      try {
  1168→        await this.cameraTrack.applyConstraints(this.videoConstraints(profile))
  1169→      } catch {
  1170→        /* some cameras ignore constraints */
  1171→      }
  1172→    }
  1173→    // 2) Bitrate caps per sender, respecting per-link adaptive caps
  1174→    for (const [peerId, senders] of this.sendersMap) {
  1175→      const sender = senders.video
  1176→      if (!sender) continue
  1177→      const link = this.links.get(peerId)
  1178→      const cap = link?.lastCap ?? null
  1179→      const target = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
  1180→      try {
  1181→        const params = sender.getParameters()
  1182→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
  1183→        params.encodings[0].maxBitrate = target
  1184→        await sender.setParameters(params)
  1185→      } catch {
  1186→        /* ignore */
  1187→      }
  1188→    }
  1189→  }

  1191→  /* ---------------- stats / adaptation ---------------- */

  1193→  private async pollStats() {
  1194→    if (this.pcs.size === 0) return
  1195→
  1196→    // Signaling watchdog: an offer left hanging for 9s+ (a rare race between
  1197→    // simultaneous reconnects) self-heals with a fresh offer.
  1198→    for (const [peerId, pc] of this.pcs) {
  1199→      if (pc.signalingState !== 'have-local-offer') continue
  1200→      if (Date.now() - (this.lastOfferAt.get(peerId) ?? 0) < 9000) continue
  1201→      if (!this.socket?.connected) continue
  1202→      this.lastOfferAt.set(peerId, Date.now())
  1203→      this.enqueueSig(peerId, () => this.makeOffer(peerId, pc))
  1204→    }
  1205→
  1206→    let overall: NetworkLevel = 3
  1207→    let totalKbps = 0
  1208→    const nextPrev = new Map<string, number>()
  1209→
  1210→    for (const [peerId, pc] of this.pcs) {
  1211→      if (pc.connectionState !== 'connected') {
  1212→        overall = Math.min(overall, 1) as NetworkLevel
  1213→        continue
  1214→      }
  1215→      let rtt: number | null = null
  1216→      let loss: number | null = null
  1217→      let bytesDelta = 0
  1218→      try {
  1219→        const stats = await pc.getStats()
  1220→        stats.forEach((report) => {
  1221→          const r = report as RTCStats & {
  1222→            currentRoundTripTime?: number
  1223→            fractionLost?: number
  1224→            type: string
  1225→            state?: string
  1226→            nominated?: boolean
  1227→            bytesSent?: number
  1228→            kind?: string
  1229→          }
  1230→          if (r.type === 'candidate-pair' && r.state === 'succeeded' && typeof r.currentRoundTripTime === 'number') {
  1231→            rtt = rtt === null ? r.currentRoundTripTime : Math.min(rtt, r.currentRoundTripTime)
  1232→          }
  1233→          if (r.type === 'remote-inbound-rtp' && typeof r.fractionLost === 'number') {
  1234→            loss = loss === null ? r.fractionLost : Math.max(loss, r.fractionLost)
  1235→          }
  1236→          if (r.type === 'outbound-rtp' && typeof r.bytesSent === 'number' && (r.kind === 'video' || r.kind === 'audio')) {
  1237→            const prevB = this.prevBytes.get(r.id) ?? r.bytesSent
  1238→            bytesDelta += Math.max(0, r.bytesSent - prevB)
  1239→            nextPrev.set(r.id, r.bytesSent)
  1240→          }
  1241→        })
  1242→      } catch {
  1243→        continue
  1244→      }
  1245→
  1246→      const level = classifyLink(rtt, loss)
  1247→      const prevLevel = useCallStore.getState().peers[peerId]?.level
  1248→      if (prevLevel !== level) useCallStore.getState().patchPeer(peerId, { level })
  1249→      overall = Math.min(overall, level) as NetworkLevel
  1250→      totalKbps += (bytesDelta * 8) / (STATS_INTERVAL_MS * 1000)
  1251→
  1252→      // Adaptive per-link video cap
  1253→      const link = this.links.get(peerId)
  1254→      const cap = LINK_CAPS[level]
  1255→      if (link && link.lastCap !== cap) {
  1256→        link.lastCap = cap
  1257→        const sender = this.sendersMap.get(peerId)?.video
  1258→        if (sender) {
  1259→          try {
  1260→            const params = sender.getParameters()
  1261→            if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
  1262→            const profile = this.resolveProfile()
  1263→            params.encodings[0].maxBitrate = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
  1264→            await sender.setParameters(params)
  1265→          } catch {
  1266→            /* ignore */
  1267→          }
  1268→        }
  1269→      }
  1270→    }
  1271→
  1272→    this.prevBytes = nextPrev
  1273→
  1274→    const prevOverall = useCallStore.getState().overallLevel
  1275→    if (prevOverall !== overall) {
  1276→      useCallStore.getState().setOverallLevel(overall)
  1277→      useCallStore.getState().patchPeer(this.myId, { level: overall })
  1278→    }
  1279→    useCallStore.setState({ bitrateKbps: Math.round(totalKbps) })
  1280→
  1281→    // Music quality follows the network instead of fighting it
  1282→    if (this.musicSource) this.applyMusicBitrate(overall)
  1283→
  1284→    // Auto quality profile with hysteresis
  1285→    if (this.preset === 'auto') {
  1286→      if (overall <= 1) {
  1287→        this.badCount++
  1288→        this.goodCount = 0
  1289→      } else if (overall === 3) {
  1290→        this.goodCount++
  1291→        this.badCount = 0
  1292→      } else {
  1293→        this.badCount = 0
  1294→        this.goodCount = 0
  1295→      }
  1296→      const order: Array<Exclude<QualityPreset, 'auto'>> = ['saver', 'balanced', 'sharp']
  1297→      const idx = order.indexOf(this.adaptiveProfile)
  1298→      if (this.badCount >= 2 && idx > 0) {
  1299→        this.adaptiveProfile = order[idx - 1]
  1300→        this.badCount = 0
  1301→        this.goodCount = 0
  1302→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
  1303→        void this.applyProfile()
  1304→      } else if (this.goodCount >= 8 && idx < order.length - 1) {
  1305→        this.adaptiveProfile = order[idx + 1]
  1306→        this.badCount = 0
  1307→        this.goodCount = 0
  1308→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
  1309→        void this.applyProfile()
  1310→      }
  1311→    }
  1312→  }

  1322→  private rejoin() {
  1323→    if (!this.roomId || !this.socket) return
  1324→    this.socket.emit(
  1325→      'room:join',
  1326→      { roomId: this.roomId, name: useCallStore.getState().self.name, passwordHash: this.passwordHash },
  1327→      (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; hostId?: string; locked?: boolean; hasPassword?: boolean; music?: { peerId: string; title: string; playing: boolean; position: number; duration: number; streamId?: string } | null }) => {
  1328→        // The server hands out a fresh socket id after a reconnect — adopt it,
  1329→        // otherwise our own messages and presence updates stop matching.
  1330→        if (res?.ok) {
  1331→          if (res.self?.id && res.self.id !== this.myId) {
  1332→            this.myId = res.self.id
  1333→            useCallStore.getState().setSelf({ id: res.self.id })
  1334→          }
  1335→          if (res.hostId) useCallStore.getState().setHostId(res.hostId)
  1336→          useCallStore.getState().setLocked(!!res.locked)
  1337→          useCallStore.getState().setPasscodeProtected(!!res.hasPassword)
  1338→          // Restore the listen-together session (if it survived the reconnect)
  1339→          // so the incoming music track is routed to the player, not a tile.
  1340→          if (res.music?.peerId && res.music.peerId !== this.myId) {
  1341→            this.remoteMusicStreamIds.clear()
  1342→            if (res.music.streamId) this.remoteMusicStreamIds.add(res.music.streamId)
  1343→            useCallStore.getState().setMusic({
  1344→              peerId: res.music.peerId,
  1345→              title: res.music.title,
  1346→              playing: res.music.playing,
  1347→              position: res.music.position,
  1348→              duration: res.music.duration,
  1349→            })
  1350→          }
  1351→          if (res.peers) this.reconcilePeers(res.peers)
  1352→          // Re-announce our live state: the server rebuilt our peer record
  1353→          // with defaults, so everyone else must re-learn mic/cam/sharing —
  1354→          // otherwise a mid-share reconnect drops the remote spotlight.
  1355→          this.socket?.emit('state:update', {
  1356→            micOn: useCallStore.getState().self.micOn,
  1357→            camOn: useCallStore.getState().self.camOn,
  1358→            sharing: !!(this.screenTrack && this.screenTrack.readyState === 'live'),
  1359→          })
  1360→          // Same for an active listen-together session: the server dropped it
  1361→          // when our old socket left, so re-announce to restore playback for
  1362→          // everyone (and our own late-join mapping on the next ontrack).
  1363→          if (this.musicSource && this.musicStreamId) {
  1364→            this.socket?.emit('music:state', {
  1365→              action: 'start',
  1366→              streamId: this.musicStreamId,
  1367→              title: this.musicSource.title,
  1368→              playing: this.musicSource.isPlaying(),
  1369→              position: this.musicSource.getPosition(),
  1370→              duration: this.musicSource.duration,
  1371→            })
  1372→          }
  1373→        }
  1374→      },
  1375→    )
  1376→  }

  1378→  async leave(silent = false) {
  1379→    if (this.statsTimer !== null) {
  1380→      window.clearInterval(this.statsTimer)
  1381→      this.statsTimer = null
  1382→    }
  1383→    this.stopMusicShare()
  1384→    detachRemoteMusic()
  1385→    this.socket?.emit('room:leave')
  1386→    for (const peerId of Array.from(this.pcs.keys())) this.dropPeer(peerId)
  1387→    this.screenTrack?.stop()
  1388→    this.localStream?.getTracks().forEach((t) => {
  1389→      try {
  1390→        t.stop()
  1391→      } catch {
  1392→        /* ignore */
  1393→      }
  1394→    })
  1395→    const selfStreamTest = (this.localStream as (MediaStream & { __stop?: () => void }) | null)?.__stop
  1396→    if (selfStreamTest) selfStreamTest()
  1397→    this.localStream = null
  1398→    this.cameraTrack = null
  1399→    this.micTrack = null
  1400→    clearAllAudio()
  1401→    putStream('self', null)
  1402→    this.pcs.clear()
  1403→    this.pendingIce.clear()
  1404→    this.links.clear()
  1405→    this.prevBytes.clear()
  1406→    this.socket?.disconnect()
  1407→    this.socket = null
  1408→    this.myId = ''
  1409→    this.started = false
  1410→    if (!silent) {
  1411→      useCallStore.getState().reset()
  1412→    }
  1413→  }
