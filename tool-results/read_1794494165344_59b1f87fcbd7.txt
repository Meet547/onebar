    15→    15→/**
    16→    16→ * Pico call engine — peer-to-peer WebRTC mesh.
    17→    17→ *
    18→    18→ * Design goals:
    19→    19→ *  - Media never touches a server: every audio/video packet travels directly
    20→    20→ *    between participants (mesh topology). The signaling socket only carries
    21→    21→ *    SDP/ICE/chat text.
    22→    22→ *  - Renegotiation-free quality control: resolution/fps changes ride on
    23→    23→ *    applyConstraints(), bitrate changes on setParameters(), screen share on
    24→    24→ *    replaceTrack(). No expensive offer/answer churn.
    25→    25→ *  - Opus DTX + 24 kbps cap: silence costs (almost) nothing on the wire.
    26→    26→ *  - Per-link adaptive bitrate: getStats() every 3s classifies each
    27→    27→ *    connection (RTT + packet loss) and caps outgoing video per link, so one
    28→    28→ *    weak participant never drags the whole call down.
    29→    29→ *  - Perfect-negotiation pattern (polite/impolite) for glare-free offers.
    30→    30→ */

    59→    59→/** Add Opus DTX + low bitrate cap to SDP — near-zero cost for silence. */
    60→    60→function mungeOpus(sdp: string): string {
    61→    61→  const lines = sdp.split('\r\n')
    62→    62→  const opusPTs = new Set<string>()
    63→    63→  for (const line of lines) {
    64→    64→    const m = line.match(/^a=rtpmap:(\d+) opus\/48000/i)
    65→    65→    if (m) opusPTs.add(m[1])
    66→    66→  }
    67→    67→  const out = lines.map((line) => {
    68→    68→    const m = line.match(/^a=fmtp:(\d+)(.*)$/)
    69→    69→    if (m && opusPTs.has(m[1])) {
    70→    70→      let params = m[2] ?? ''
    71→    71→      if (!/usedtx/i.test(params)) params += ';usedtx=1'
    72→    72→      if (!/maxaveragebitrate/i.test(params)) params += ';maxaveragebitrate=24000'
    73→    73→      return `a=fmtp:${m[1]}${params}`
    74→    74→    }
    75→    75→    return line
    76→    76→  })
    77→    77→  return out.join('\r\n')
    78→    78→}
    79→    79→
    80→    80→function classifyLink(rttSec: number | null, lossFrac: number | null): NetworkLevel {
    81→    81→  const rtt = rttSec ?? 0.4
    82→    82→  const loss = lossFrac ?? 0.05
    83→    83→  if (rtt > 0.5 || loss > 0.08) return 0
    84→    84→  if (rtt > 0.3 || loss > 0.05) return 1
    85→    85→  if (rtt > 0.15 || loss > 0.02) return 2
    86→    86→  return 3
    87→    87→}

   103→   103→  private pendingIce = new Map<string, RTCIceCandidateInit[]>()
   104→   104→  private links = new Map<string, LinkInfo>()
   105→   105→  private statsTimer: number | null = null
   106→   106→  private makingOffer = new Map<string, boolean>()
   107→   107→  private ignoreOffer = new Map<string, boolean>()
   108→   108→  private preset: QualityPreset = 'balanced'
   109→   109→  private adaptiveProfile: Exclude<QualityPreset, 'auto'> = 'balanced'
   110→   110→  private badCount = 0
   111→   111→  private goodCount = 0
   112→   112→  private started = false

   117→   117→  async start(opts: StartOptions): Promise<{ mode: MediaMode }> {
   118→   118→    if (this.started) await this.leave(true)
   119→   119→    this.started = true
   120→   120→    this.roomId = opts.roomId
   121→   121→    this.preset = opts.preset
   122→   122→    this.adaptiveProfile = opts.preset === 'auto' ? 'balanced' : opts.preset
   123→   123→    this.testMode = !!opts.testMode
   124→   124→
   125→   125→    const store = useCallStore.getState()
   126→   126→    store.setSelf({ name: opts.name })
   127→   127→    store.setPreset(opts.preset)
   128→   128→    store.setEffectiveLabel(this.effectiveLabel())
   129→   129→
   130→   130→    // ---- Signaling ----
   131→   131→    const socket = io('/?XTransformPort=3003', {
   132→   132→      path: '/',
   133→   133→      transports: ['websocket', 'polling'],
   134→   134→      reconnection: true,
   135→   135→      reconnectionAttempts: Infinity,
   136→   136→      reconnectionDelay: 1000,
   137→   137→      reconnectionDelayMax: 4000,
   138→   138→      timeout: 8000,
   139→   139→    })
   140→   140→    this.socket = socket
   141→   141→    // Dev/debug handle (harmless in prod, invaluable for live diagnostics)
   142→   142→    if (typeof window !== 'undefined') {
   143→   143→      ;(window as unknown as { __pico?: unknown }).__pico = { engine: this, socket }
   144→   144→    }
   145→   145→    socket.on('connect', () => {
   146→   146→      useCallStore.getState().setSignalingConnected(true)
   147→   147→      if (this.myId) this.rejoin()
   148→   148→    })
   149→   149→    socket.on('disconnect', () => useCallStore.getState().setSignalingConnected(false))
   150→   150→    socket.on('connect_error', () => useCallStore.getState().setSignalingConnected(false))
   151→   151→
   152→   152→    // ---- Local media ----
   153→   153→    this.localStream = opts.initialStream ?? null
   154→   154→    if (this.localStream) {
   155→   155→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   156→   156→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   157→   157→      this.mediaMode = this.cameraTrack ? 'full' : this.micTrack ? 'audio' : 'none'
   158→   158→      // Register the handed-over preview stream so tiles can attach to it
   159→   159→      putStream('self', this.localStream)
   160→   160→    }
   161→   161→    if (!this.localStream || (!this.cameraTrack && !this.micTrack)) {
   162→   162→      this.mediaMode = await this.acquireMedia()
   163→   163→    }
   164→   164→    this.applyInitialTrackState()
   165→   165→    trackAudio('self', this.localStream)
   166→   166→
   167→   167→    // ---- Room ----
   168→   168→    // The ack races a timeout so a dead signaling server can never hang the UI.
   169→   169→    await new Promise<void>((resolve) => {
   170→   170→      let settled = false
   171→   171→      const finish = () => {
   172→   172→        if (!settled) {
   173→   173→          settled = true
   174→   174→          resolve()
   175→   175→        }
   176→   176→      }
   177→   177→      const guard = window.setTimeout(finish, 12000)
   178→   178→      socket.emit(
   179→   179→        'room:join',
   180→   180→        { roomId: this.roomId, name: opts.name },
   181→   181→        (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>; reason?: string }) => {
   182→   182→          window.clearTimeout(guard)
   183→   183→          if (!res?.ok) {
   184→   184→            useCallStore.getState().addChat({
   185→   185→              id: 'sys-' + Date.now(),
   186→   186→              from: 'system',
   187→   187→              name: 'System',
   188→   188→              text: res?.reason ?? 'Could not join the room',
   189→   189→              ts: Date.now(),
   190→   190→              system: true,
   191→   191→            })
   192→   192→          }
   193→   193→          this.myId = res?.self?.id ?? socket.id!
   194→   194→          useCallStore.getState().setSelf({ id: this.myId })
   195→   195→          if (res?.peers) this.reconcilePeers(res.peers)
   196→   196→          finish()
   197→   197→        },
   198→   198→      )
   199→   199→    })
   200→   200→
   201→   201→    // ---- Events ----
   202→   202→    socket.on('peer:joined', ({ peer }: { peer: { id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean } }) => {
   203→   203→      this.ensurePeer(peer.id, peer.name, peer.micOn, peer.camOn, peer.sharing)
   204→   204→      useCallStore.getState().addChat({
   205→   205→        id: 'sys-' + Math.random(),
   206→   206→        from: 'system',
   207→   207→        name: 'System',
   208→   208→        text: `${peer.name} joined`,
   209→   209→        ts: Date.now(),
   210→   210→        system: true,
   211→   211→      })
   212→   212→    })
   213→   213→    socket.on('peer:left', ({ peerId }: { peerId: string }) => this.dropPeer(peerId))
   214→   214→    socket.on('room:peers', ({ peers }: { peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   215→   215→      // Only meaningful after a reconnect — reconcile full presence
   216→   216→      if (this.myId) this.reconcilePeers(peers)
   217→   217→    })
   218→   218→    socket.on('peer:state', ({ peerId, micOn, camOn, sharing }: { peerId: string; micOn: boolean; camOn: boolean; sharing: boolean }) => {
   219→   219→      const store = useCallStore.getState()
   220→   220→      store.patchPeer(peerId, { micOn, camOn, sharing })
   221→   221→      // Follow whoever is presenting, so a remote screen never hides in a small tile
   222→   222→      if (sharing && store.spotlight !== peerId) store.setSpotlight(peerId)
   223→   223→      else if (!sharing && store.spotlight === peerId) store.setSpotlight(null)
   224→   224→    })
   225→   225→    socket.on('chat:message', (msg: { id: string; from: string; name: string; text: string; ts: number }) => {
   226→   226→      useCallStore.getState().addChat(msg)
   227→   227→    })
   228→   228→
   229→   229→    socket.on('rtc:offer', async ({ from, sdp }: { from: string; sdp: string }) => {
   230→   230→      try {
   231→   231→        const pc = this.pcs.get(from)
   232→   232→        if (!pc) return
   233→   233→        const polite = this.myId < from
   234→   234→        const offerCollision = this.makingOffer.get(from) || pc.signalingState !== 'stable'
   235→   235→        this.ignoreOffer.set(from, !polite && offerCollision)
   236→   236→        if (this.ignoreOffer.get(from)) return
   237→   237→        await pc.setRemoteDescription({ type: 'offer', sdp: mungeOpus(sdp) })
   238→   238→        await this.flushIce(from)
   239→   239→        const answer = await pc.createAnswer()
   240→   240→        answer.sdp = mungeOpus(answer.sdp ?? '')
   241→   241→        await pc.setLocalDescription(answer)
   242→   242→        this.socket?.emit('rtc:answer', { to: from, sdp: pc.localDescription!.sdp })
   243→   243→      } catch {
   244→   244→        /* negotiation raced — next round will fix it */
   245→   245→      }
   246→   246→    })
   247→   247→
   248→   248→    socket.on('rtc:answer', async ({ from, sdp }: { from: string; sdp: string }) => {
   249→   249→      try {
   250→   250→        const pc = this.pcs.get(from)
   251→   251→        if (!pc || pc.signalingState !== 'have-local-offer') return
   252→   252→        await pc.setRemoteDescription({ type: 'answer', sdp: mungeOpus(sdp) })
   253→   253→        await this.flushIce(from)
   254→   254→      } catch {
   255→   255→        /* ignore */
   256→   256→      }
   257→   257→    })
   258→   258→
   259→   259→    socket.on('rtc:ice', ({ from, candidate }: { from: string; candidate: RTCIceCandidateInit }) => {
   260→   260→      const pc = this.pcs.get(from)
   261→   261→      if (!pc) return
   262→   262→      if (pc.remoteDescription) {
   263→   263→        pc.addIceCandidate(candidate).catch(() => {
   264→   264→          if (!this.ignoreOffer.get(from)) {
   265→   265→            this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   266→   266→          }
   267→   267→        })
   268→   268→      } else {
   269→   269→        this.pendingIce.set(from, [...(this.pendingIce.get(from) ?? []), candidate])
   270→   270→      }
   271→   271→    })
   272→   272→
   273→   273→    // ---- Stats loop ----
   274→   274→    this.statsTimer = window.setInterval(() => void this.pollStats(), STATS_INTERVAL_MS)
   275→   275→
   276→   276→    return { mode: this.mediaMode }
   277→   277→  }
   278→   278→
   279→   279→  /* ---------------- media ---------------- */
   280→   280→
   281→   281→  private async acquireMedia(): Promise<MediaMode> {
   282→   282→    if (this.testMode) {
   283→   283→      this.localStream = await createTestStream()
   284→   284→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   285→   285→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   286→   286→      putStream('self', this.localStream)
   287→   287→      return 'full'
   288→   288→    }
   289→   289→    const audio: MediaTrackConstraints = {
   290→   290→      echoCancellation: true,
   291→   291→      noiseSuppression: true,
   292→   292→      autoGainControl: true,
   293→   293→      channelCount: 1,
   294→   294→    }
   295→   295→    const video = this.videoConstraints(this.resolveProfile())
   296→   296→    // Full quality first
   297→   297→    try {
   298→   298→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio, video })
   299→   299→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   300→   300→      this.cameraTrack = this.localStream.getVideoTracks()[0] ?? null
   301→   301→      this.attachLocalStream()
   302→   302→      return 'full'
   303→   303→    } catch {
   304→   304→      /* fall through */
   305→   305→    }
   306→   306→    // Audio only
   307→   307→    try {
   308→   308→      this.localStream = await navigator.mediaDevices.getUserMedia({ audio })
   309→   309→      this.micTrack = this.localStream.getAudioTracks()[0] ?? null
   310→   310→      this.cameraTrack = null
   311→   311→      this.attachLocalStream()
   312→   312→      return 'audio'
   313→   313→    } catch {
   314→   314→      /* fall through */
   315→   315→    }
   316→   316→    // Listener mode
   317→   317→    this.localStream = null
   318→   318→    return 'none'
   319→   319→  }

   325→   325→      frameRate: { ideal: profile.frameRate, max: profile.frameRate },
   326→   326→      facingMode: 'user',
   327→   327→    }
   328→   328→  }

   330→   330→  private attachLocalStream() {

   335→   335→  private applyInitialTrackState() {
   336→   336→    const s = useCallStore.getState().self
   337→   337→    if (this.cameraTrack) this.cameraTrack.enabled = s.camOn
   338→   338→    if (this.micTrack) this.micTrack.enabled = s.micOn
   339→   339→    if (!s.camOn) void this.setVideoSendersActive(false)

   342→   342→  /* ---------------- peers ---------------- */
   343→   343→
   344→   344→  private ensurePeer(
   345→   345→    peerId: string,
   346→   346→    name: string,
   347→   347→    micOn = true,
   348→   348→    camOn = true,
   349→   349→    sharing = false,
   350→   350→  ) {
   351→   351→    const store = useCallStore.getState()
   352→   352→    if (!store.peers[peerId]) {
   353→   353→      store.upsertPeer({ id: peerId, name, micOn, camOn, sharing, level: 3, hasStream: false, streamVersion: 0 })
   354→   354→    }
   355→   355→    if (this.pcs.has(peerId)) return
   356→   356→
   357→   357→    const pc = new RTCPeerConnection({
   358→   358→      iceServers: ICE_SERVERS,
   359→   359→      iceCandidatePoolSize: 4,
   360→   360→      bundlePolicy: 'max-bundle',
   361→   361→      rtcpMuxPolicy: 'require',
   362→   362→    })
   363→   363→    this.pcs.set(peerId, pc)
   364→   364→    this.links.set(peerId, { lastCap: null })
   365→   365→    this.makingOffer.set(peerId, false)
   366→   366→    this.ignoreOffer.set(peerId, false)
   367→   367→
   368→   368→    // Local media into this connection
   369→   369→    const senders: { audio?: RTCRtpSender; video?: RTCRtpSender } = {}
   370→   370→    if (this.localStream) {
   371→   371→      for (const track of this.localStream.getTracks()) {
   372→   372→        try {
   373→   373→          const sender = pc.addTrack(track, this.localStream)
   374→   374→          if (track.kind === 'audio') senders.audio = sender
   375→   375→          else senders.video = sender
   376→   376→        } catch {
   377→   377→          /* ignore */
   378→   378→        }
   379→   379→      }
   380→   380→    } else {
   381→   381→      const at = pc.addTransceiver('audio', { direction: 'recvonly' })
   382→   382→      const vt = pc.addTransceiver('video', { direction: 'recvonly' })
   383→   383→      senders.audio = at.sender
   384→   384→      senders.video = vt.sender
   385→   385→    }
   386→   386→    this.sendersMap.set(peerId, senders)
   387→   387→
   388→   388→    pc.onicecandidate = (e) => {
   389→   389→      if (e.candidate) this.socket?.emit('rtc:ice', { to: peerId, candidate: e.candidate.toJSON() })
   390→   390→    }
   391→   391→
   392→   392→    pc.ontrack = (e) => {
   393→   393→      const stream = e.streams[0]
   394→   394→      if (!stream) return
   395→   395→      putStream(peerId, stream)
   396→   396→      trackAudio(peerId, stream)
   397→   397→      useCallStore.getState().bumpPeerStream(peerId, true)
   398→   398→    }
   399→   399→
   400→   400→    pc.onnegotiationneeded = async () => {
   401→   401→      try {
   402→   402→        this.makingOffer.set(peerId, true)
   403→   403→        const offer = await pc.createOffer()
   404→   404→        await pc.setLocalDescription({ type: 'offer', sdp: mungeOpus(offer.sdp ?? '') })
   405→   405→        const desc = pc.localDescription
   406→   406→        if (desc) this.socket?.emit('rtc:offer', { to: peerId, sdp: desc.sdp })
   407→   407→      } catch {
   408→   408→        /* ignore */
   409→   409→      } finally {
   410→   410→        this.makingOffer.set(peerId, false)
   411→   411→      }
   412→   412→    }
   413→   413→
   414→   414→    pc.onconnectionstatechange = () => {
   415→   415→      if (pc.connectionState === 'failed') {
   416→   416→        try {
   417→   417→          pc.restartIce()
   418→   418→        } catch {
   419→   419→          /* ignore */
   420→   420→        }
   421→   421→      }
   422→   422→    }
   423→   423→
   424→   424→    // Prime encoding params on the fresh senders
   425→   425→    this.capAudioSender(pc)
   426→   426→    const videoSender = senders.video
   427→   427→    if (videoSender) {
   428→   428→      try {
   429→   429→        const params = videoSender.getParameters()
   430→   430→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   431→   431→        params.encodings[0].maxBitrate = this.resolveProfile().maxBitrate
   432→   432→        void videoSender.setParameters(params).catch(() => undefined)
   433→   433→      } catch {
   434→   434→        /* ignore */
   435→   435→      }
   436→   436→    }
   437→   437→  }
   438→   438→
   439→   439→  private dropPeer(peerId: string) {
   440→   440→    const pc = this.pcs.get(peerId)
   441→   441→    if (pc) {
   442→   442→      pc.close()
   443→   443→    }
   444→   444→    this.pcs.delete(peerId)
   445→   445→    this.sendersMap.delete(peerId)
   446→   446→    this.pendingIce.delete(peerId)
   447→   447→    this.links.delete(peerId)
   448→   448→    trackAudio(peerId, null)
   449→   449→    putStream(peerId, null)
   450→   450→    useCallStore.getState().bumpPeerStream(peerId, false)
   451→   451→    useCallStore.getState().removePeer(peerId)
   452→   452→  }
   453→   453→
   454→   454→  private reconcilePeers(
   455→   455→    peers: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }>,
   456→   456→  ) {
   457→   457→    const store = useCallStore.getState()
   458→   458→    const seen = new Set<string>()
   459→   459→    for (const p of peers) {
   460→   460→      if (p.id === this.myId) continue
   461→   461→      seen.add(p.id)
   462→   462→      this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   463→   463→    }
   464→   464→    for (const id of Object.keys(store.peers)) {
   465→   465→      if (!seen.has(id)) this.dropPeer(id)
   466→   466→    }
   467→   467→    // After a reconnect, old connections may be stale — rebuild them
   468→   468→    for (const [peerId, pc] of this.pcs) {
   469→   469→      if (pc.connectionState === 'failed' || pc.connectionState === 'closed') {
   470→   470→        this.dropPeer(peerId)
   471→   471→        const p = peers.find((x) => x.id === peerId)
   472→   472→        if (p) this.ensurePeer(p.id, p.name, p.micOn, p.camOn, p.sharing)
   473→   473→      }
   474→   474→    }
   475→   475→  }

   477→   477→  private async flushIce(peerId: string) {
   478→   478→    const pc = this.pcs.get(peerId)
   479→   479→    const queued = this.pendingIce.get(peerId) ?? []
   480→   480→    if (!pc || queued.length === 0) return
   481→   481→    this.pendingIce.set(peerId, [])
   482→   482→    for (const c of queued) {
   483→   483→      try {
   484→   484→        await pc.addIceCandidate(c)
   485→   485→      } catch {
   486→   486→        /* ignore */
   487→   487→      }
   488→   488→    }
   489→   489→  }

   501→   501→  async toggleCam(): Promise<boolean> {
   502→   502→    const next = !useCallStore.getState().self.camOn
   503→   503→    useCallStore.getState().setSelf({ camOn: next })
   504→   504→    this.socket?.emit('state:update', { camOn: next })
   505→   505→
   506→   506→    if (next && !this.cameraTrack && this.mediaMode !== 'none') {
   507→   507→      // try to acquire a camera mid-call (video track added -> renegotiation fires)
   508→   508→      try {
   509→   509→        const profile = this.resolveProfile()
   510→   510→        const stream = await navigator.mediaDevices.getUserMedia({ video: this.videoConstraints(profile) })
   511→   511→        const track = stream.getVideoTracks()[0]
   512→   512→        this.cameraTrack = track
   513→   513→        track.enabled = true
   514→   514→        if (this.localStream) this.localStream.addTrack(track)
   515→   515→        else this.localStream = stream
   516→   516→        this.publishLocalTrack(track, this.localStream)
   517→   517→        putStream('self', this.localStream)
   518→   518→        await this.setVideoSendersActive(true)
   519→   519→      } catch {
   520→   520→        useCallStore.getState().setSelf({ camOn: false })
   521→   521→        this.socket?.emit('state:update', { camOn: false })
   522→   522→        return false
   523→   523→      }
   524→   524→      return true
   525→   525→    }
   526→   526→
   527→   527→    if (this.cameraTrack) this.cameraTrack.enabled = next
   528→   528→    await this.setVideoSendersActive(next && !this.screenTrack)
   529→   529→    return next
   530→   530→  }

   532→   532→  private publishLocalTrack(track: MediaStreamTrack, owner: MediaStream) {
   533→   533→    for (const [peerId, pc] of this.pcs) {
   534→   534→      const senders = this.sendersMap.get(peerId) ?? {}
   535→   535→      try {
   536→   536→        if (track.kind === 'video' && senders.video && senders.video.track?.kind === 'video') {
   537→   537→          // Swap the outgoing source — no renegotiation, seamless for viewers
   538→   538→          void senders.video.replaceTrack(track).catch(() => {
   539→   539→            try {
   540→   540→              const s = pc.addTrack(track, owner)
   541→   541→              senders.video = s
   542→   542→              this.sendersMap.set(peerId, senders)
   543→   543→            } catch {
   544→   544→              /* ignore */
   545→   545→            }
   546→   546→          })
   547→   547→        } else {
   548→   548→          const s = pc.addTrack(track, owner)
   549→   549→          if (track.kind === 'audio') senders.audio = s
   550→   550→          else senders.video = s
   551→   551→          this.sendersMap.set(peerId, senders)
   552→   552→        }
   553→   553→      } catch {
   554→   554→        /* ignore */
   555→   555→      }
   556→   556→    }
   557→   557→  }

   559→   559→  async toggleShare(): Promise<boolean> {
   560→   560→    const sharing = !useCallStore.getState().self.sharing
   561→   561→    if (sharing) {
   562→   562→      try {
   563→   563→        const display = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: false })
   564→   564→        const track = display.getVideoTracks()[0]
   565→   565→        this.screenTrack = track
   566→   566→        track.addEventListener('ended', () => {
   567→   567→          void this.stopShare()
   568→   568→        })
   569→   569→        this.publishLocalTrack(track, display)
   570→   570→        // mirror the shared screen into the local preview tile
   571→   571→        putStream('self', display)
   572→   572→        useCallStore.getState().bumpPeerStream('self', true)
   573→   573→        useCallStore.getState().setSelf({ sharing: true })
   574→   574→        this.socket?.emit('state:update', { sharing: true })
   575→   575→        useCallStore.getState().setSpotlight('self')
   576→   576→        return true
   577→   577→      } catch {
   578→   578→        return false
   579→   579→      }
   580→   580→    } else {
   581→   581→      await this.stopShare()
   582→   582→      return false
   583→   583→    }
   584→   584→  }

   586→   586→  private async stopShare() {
   587→   587→    this.screenTrack?.stop()
   588→   588→    this.screenTrack = null
   589→   589→    if (this.cameraTrack && this.cameraTrack.readyState === 'live') {
   590→   590→      // restore the camera as the outgoing video source and the local preview
   591→   591→      this.publishLocalTrack(this.cameraTrack, this.localStream ?? new MediaStream([this.cameraTrack]))
   592→   592→      if (this.localStream) putStream('self', this.localStream)
   593→   593→      await this.setVideoSendersActive(useCallStore.getState().self.camOn)
   594→   594→    } else {
   595→   595→      // no camera available: stop sending video entirely
   596→   596→      await this.setVideoSendersActive(false)
   597→   597→    }
   598→   598→    useCallStore.getState().setSelf({ sharing: false })
   599→   599→    this.socket?.emit('state:update', { sharing: false })
   600→   600→    const spotlight = useCallStore.getState().spotlight
   601→   601→    if (spotlight === 'self') useCallStore.getState().setSpotlight(null)
   602→   602→  }

   605→   605→    for (const [, senders] of this.sendersMap) {

   610→   610→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   611→   611→        params.encodings[0].active = active
   612→   612→        await sender.setParameters(params)
   613→   613→      } catch {
   614→   614→        /* browser may not support active toggle */
   615→   615→      }
   616→   616→    }

   648→   648→    this.preset = p
   649→   649→    useCallStore.getState().setPreset(p)
   650→   650→    if (p !== 'auto') this.adaptiveProfile = p

   654→   654→
   655→   655→  private async applyProfile() {
   656→   656→    const profile = this.resolveProfile()
   657→   657→    // 1) Capture resolution / fps (no renegotiation needed)
   658→   658→    if (this.cameraTrack && !this.screenTrack && this.cameraTrack.readyState === 'live') {
   659→   659→      try {
   660→   660→        await this.cameraTrack.applyConstraints(this.videoConstraints(profile))
   661→   661→      } catch {
   662→   662→        /* some cameras ignore constraints */
   663→   663→      }
   664→   664→    }
   665→   665→    // 2) Bitrate caps per sender, respecting per-link adaptive caps
   666→   666→    for (const [peerId, senders] of this.sendersMap) {
   667→   667→      const sender = senders.video
   668→   668→      if (!sender) continue
   669→   669→      const link = this.links.get(peerId)
   670→   670→      const cap = link?.lastCap ?? null
   671→   671→      const target = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   672→   672→      try {
   673→   673→        const params = sender.getParameters()
   674→   674→        if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   675→   675→        params.encodings[0].maxBitrate = target
   676→   676→        await sender.setParameters(params)
   677→   677→      } catch {
   678→   678→        /* ignore */
   679→   679→      }
   680→   680→    }
   681→   681→  }

   683→   683→  /* ---------------- stats / adaptation ---------------- */
   684→   684→
   685→   685→  private async pollStats() {
   686→   686→    if (this.pcs.size === 0) return
   687→   687→    let overall: NetworkLevel = 3
   688→   688→    let totalKbps = 0
   689→   689→    const nextPrev = new Map<string, number>()
   690→   690→
   691→   691→    for (const [peerId, pc] of this.pcs) {
   692→   692→      if (pc.connectionState !== 'connected') {
   693→   693→        overall = Math.min(overall, 1) as NetworkLevel
   694→   694→        continue
   695→   695→      }
   696→   696→      let rtt: number | null = null
   697→   697→      let loss: number | null = null
   698→   698→      let bytesDelta = 0
   699→   699→      try {
   700→   700→        const stats = await pc.getStats()
   701→   701→        stats.forEach((report) => {
   702→   702→          const r = report as RTCStats & {
   703→   703→            currentRoundTripTime?: number
   704→   704→            fractionLost?: number
   705→   705→            type: string
   706→   706→            state?: string
   707→   707→            nominated?: boolean
   708→   708→            bytesSent?: number
   709→   709→            kind?: string
   710→   710→          }
   711→   711→          if (r.type === 'candidate-pair' && r.state === 'succeeded' && typeof r.currentRoundTripTime === 'number') {
   712→   712→            rtt = rtt === null ? r.currentRoundTripTime : Math.min(rtt, r.currentRoundTripTime)
   713→   713→          }
   714→   714→          if (r.type === 'remote-inbound-rtp' && typeof r.fractionLost === 'number') {
   715→   715→            loss = loss === null ? r.fractionLost : Math.max(loss, r.fractionLost)
   716→   716→          }
   717→   717→          if (r.type === 'outbound-rtp' && r.kind === 'video' && typeof r.bytesSent === 'number') {
   718→   718→            const prevB = this.prevBytes.get(r.id) ?? r.bytesSent
   719→   719→            bytesDelta += Math.max(0, r.bytesSent - prevB)
   720→   720→            nextPrev.set(r.id, r.bytesSent)
   721→   721→          }
   722→   722→        })
   723→   723→      } catch {
   724→   724→        continue
   725→   725→      }
   726→   726→
   727→   727→      const level = classifyLink(rtt, loss)
   728→   728→      const prevLevel = useCallStore.getState().peers[peerId]?.level
   729→   729→      if (prevLevel !== level) useCallStore.getState().patchPeer(peerId, { level })
   730→   730→      overall = Math.min(overall, level) as NetworkLevel
   731→   731→      totalKbps += (bytesDelta * 8) / (STATS_INTERVAL_MS * 1000)
   732→   732→
   733→   733→      // Adaptive per-link video cap
   734→   734→      const link = this.links.get(peerId)
   735→   735→      const cap = LINK_CAPS[level]
   736→   736→      if (link && link.lastCap !== cap) {
   737→   737→        link.lastCap = cap
   738→   738→        const sender = this.sendersMap.get(peerId)?.video
   739→   739→        if (sender) {
   740→   740→          try {
   741→   741→            const params = sender.getParameters()
   742→   742→            if (!params.encodings || params.encodings.length === 0) params.encodings = [{}]
   743→   743→            const profile = this.resolveProfile()
   744→   744→            params.encodings[0].maxBitrate = cap !== null ? Math.min(cap, profile.maxBitrate) : profile.maxBitrate
   745→   745→            await sender.setParameters(params)
   746→   746→          } catch {
   747→   747→            /* ignore */
   748→   748→          }
   749→   749→        }
   750→   750→      }
   751→   751→    }
   752→   752→
   753→   753→    this.prevBytes = nextPrev
   754→   754→
   755→   755→    const prevOverall = useCallStore.getState().overallLevel
   756→   756→    if (prevOverall !== overall) {
   757→   757→      useCallStore.getState().setOverallLevel(overall)
   758→   758→      useCallStore.getState().patchPeer(this.myId, { level: overall })
   759→   759→    }
   760→   760→    useCallStore.setState({ bitrateKbps: Math.round(totalKbps) })
   761→   761→
   762→   762→    // Auto quality profile with hysteresis
   763→   763→    if (this.preset === 'auto') {
   764→   764→      if (overall <= 1) {
   765→   765→        this.badCount++
   766→   766→        this.goodCount = 0
   767→   767→      } else if (overall === 3) {
   768→   768→        this.goodCount++
   769→   769→        this.badCount = 0
   770→   770→      } else {
   771→   771→        this.badCount = 0
   772→   772→        this.goodCount = 0
   773→   773→      }
   774→   774→      const order: Array<Exclude<QualityPreset, 'auto'>> = ['saver', 'balanced', 'sharp']
   775→   775→      const idx = order.indexOf(this.adaptiveProfile)
   776→   776→      if (this.badCount >= 2 && idx > 0) {
   777→   777→        this.adaptiveProfile = order[idx - 1]
   778→   778→        this.badCount = 0
   779→   779→        this.goodCount = 0
   780→   780→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   781→   781→        void this.applyProfile()
   782→   782→      } else if (this.goodCount >= 8 && idx < order.length - 1) {
   783→   783→        this.adaptiveProfile = order[idx + 1]
   784→   784→        this.badCount = 0
   785→   785→        this.goodCount = 0
   786→   786→        useCallStore.getState().setEffectiveLabel(this.effectiveLabel())
   787→   787→        void this.applyProfile()
   788→   788→      }
   789→   789→    }
   790→   790→  }

   800→   800→  private rejoin() {
   801→   801→    if (!this.roomId || !this.socket) return
   802→   802→    this.socket.emit(
   803→   803→      'room:join',
   804→   804→      { roomId: this.roomId, name: useCallStore.getState().self.name },
   805→   805→      (res: { ok: boolean; self?: { id: string }; peers?: Array<{ id: string; name: string; micOn: boolean; camOn: boolean; sharing: boolean }> }) => {
   806→   806→        // The server hands out a fresh socket id after a reconnect — adopt it,
   807→   807→        // otherwise our own messages and presence updates stop matching.
   808→   808→        if (res?.ok) {
   809→   809→          if (res.self?.id && res.self.id !== this.myId) {
   810→   810→            this.myId = res.self.id
   811→   811→            useCallStore.getState().setSelf({ id: res.self.id })
   812→   812→          }
   813→   813→          if (res.peers) this.reconcilePeers(res.peers)
   814→   814→        }
   815→   815→      },
   816→   816→    )
   817→   817→  }

   819→   819→  async leave(silent = false) {
   820→   820→    if (this.statsTimer !== null) {
   821→   821→      window.clearInterval(this.statsTimer)
   822→   822→      this.statsTimer = null
   823→   823→    }
   824→   824→    this.socket?.emit('room:leave')
   825→   825→    for (const peerId of Array.from(this.pcs.keys())) this.dropPeer(peerId)
   826→   826→    this.screenTrack?.stop()
   827→   827→    this.localStream?.getTracks().forEach((t) => {
   828→   828→      try {
   829→   829→        t.stop()
   830→   830→      } catch {
   831→   831→        /* ignore */
   832→   832→      }
   833→   833→    })
   834→   834→    const selfStreamTest = (this.localStream as (MediaStream & { __stop?: () => void }) | null)?.__stop
   835→   835→    if (selfStreamTest) selfStreamTest()
   836→   836→    this.localStream = null
   837→   837→    this.cameraTrack = null
   838→   838→    this.micTrack = null
   839→   839→    clearAllAudio()
   840→   840→    putStream('self', null)
   841→   841→    this.pcs.clear()
   842→   842→    this.pendingIce.clear()
   843→   843→    this.links.clear()
   844→   844→    this.prevBytes.clear()
   845→   845→    this.socket?.disconnect()
   846→   846→    this.socket = null
   847→   847→    this.myId = ''
   848→   848→    this.started = false
   849→   849→    if (!silent) {
   850→   850→      useCallStore.getState().reset()
   851→   851→    }
   852→   852→  }
